### peek_char() 辅助方法

[packages/sz-orm-graphql/src/query_ir.rs:536](file:///E:/vue/test/鲜视达/rust/sz-orm/packages/sz-orm-graphql/src/query_ir.rs#L536)
```rust
fn peek_char(&self) -> Option<char> {
    if self.pos >= self.input.len() {
        return None;
    }
    let first = self.input[self.pos];
    let len = if first < 0x80 { 1 }
        else if first < 0xC0 { 1 }
        else if first < 0xE0 { 2 }
        else if first < 0xF0 { 3 }
        else { 4 };
    if self.pos + len > self.input.len() {
        return None;
    }
    std::str::from_utf8(&self.input[self.pos..self.pos + len])
        .ok()?
        .chars()
        .next()
}
```
