并发安全布隆过滤器

use sz_orm_core::bloom::BloomFilter;

let mut filter = BloomFilter::new(100, 0.01);
filter.add("key-1").unwrap();
assert!(filter.might_contain("key-1"));
assert!(!filter.might_contain("key-2")); // 不存在一定返回 false（不漏判）
