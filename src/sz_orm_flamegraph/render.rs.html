1//! 火焰图渲染
2//!
3//! - [`to_brendan_gregg`]：折叠栈格式（`flamegraph.pl` 兼容）
4//! - [`to_svg`]：自绘内联 SVG（无外部依赖，浏览器可直接打开）
5
6use crate::QueryPhaseTiming;
7
8/// 折叠栈格式：每阶段一行 `query;phase duration_us`，可直接喂给 `flamegraph.pl`。
9///
10/// ```text
11/// query_execute;query.build 200
12/// query_execute;pool.acquire 150
13/// query_execute;db.execute 4200
14/// ```
15pub fn to_brendan_gregg(timings: &[QueryPhaseTiming]) -> String {
16    let mut out = String::new();
17    for t in timings {
18        let us = t.duration_ms.saturating_mul(1000);
19        out.push_str(&format!("query_execute;{} {}\n", t.phase.as_str(), us));
20    }
21    out
22}
23
24/// 内联 SVG 火焰图：单层横向分块，块宽与耗时成正比，颜色按阶段区分。
25///
26/// 输出为完整 `<svg>` 文档（无外部依赖），宽度固定 800，高度 60 + 标注行。
27pub fn to_svg(timings: &[QueryPhaseTiming]) -> String {
28    const WIDTH: u64 = 800;
29    const HEIGHT: u64 = 48;
30    const ROW_HEIGHT: u64 = 16;
31    const MARGIN: u64 = 4;
32
33    let total: u64 = timings.iter().map(|t| t.duration_ms).sum();
34    let total = if total == 0 { 1 } else { total };
35
36    let mut rects = String::new();
37    let mut x: u64 = 0;
38    for t in timings {
39        let w = (t.duration_ms as u128 * WIDTH as u128 / total as u128) as u64;
40        let w = w.max(if t.duration_ms > 0 { 1 } else { 0 });
41        let color = phase_color(t.phase.as_str());
42        let title = format!("{}: {}ms", t.phase.as_str(), t.duration_ms);
43        rects.push_str(&format!(
44            r#"<rect x="{}" y="{}" width="{}" height="{}" fill="{}" rx="1"><title>{}</title></rect>"#,
45            x, MARGIN, w, ROW_HEIGHT, color, title
46        ));
47        x += w;
48    }
49
50    let legend: Vec<String> = timings
51        .iter()
52        .map(|t| format!("{} {}ms", t.phase.as_str(), t.duration_ms))
53        .collect();
54
55    format!(
56        r#"<svg xmlns="http://www.w3.org/2000/svg" width="{WIDTH}" height="{HEIGHT}" viewBox="0 0 {WIDTH} {HEIGHT}">
57<!-- sz-orm query flame graph (total: {total}ms) -->
58{rects}
59<text x="4" y="{HEIGHT}" font-family="monospace" font-size="11">{}</text>
60</svg>"#,
61        legend
62            .join(" | ")
63            .replace('&', "&amp;")
64            .replace('<', "&lt;")
65    )
66}
67
68/// 阶段颜色（稳定映射，同阶段同色）
69fn phase_color(phase: &str) -> &'static str {
70    match phase {
71        "query.build" => "#d95f02",
72        "query.bind" => "#7570b3",
73        "pool.acquire" => "#e7298a",
74        "db.execute" => "#1b9e77",
75        "result.map" => "#66a61e",
76        _ => "#999999",
77    }
78}

106    fn brendan_gregg_format_is_folded() {
107        let out = to_brendan_gregg(&sample());
108        let lines: Vec<&str> = out.lines().collect();
109        assert_eq!(lines.len(), 3);
