1//! 断路器抽象（P1-4：抽象提升到核心层，供连接池执行路径集成）
2//!
3//! 连接池在 `circuit-breaker` feature 下通过 `configure_circuit_breaker()` 配置
4//! [`DefaultCircuitBreaker`]，在获取连接/执行查询前调用 [`CircuitBreaker::can_execute`]
5//! 拦截失败请求，成功/失败时记录反馈，防止故障级联。
6//!
7//! 本模块为自包含实现（不依赖 sz-orm-health），消除核心层对上层 crate 的反向依赖。
8
9use std::time::{Duration, Instant};
10
11/// 断路器状态机状态。
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub enum CircuitState {
14    /// 正常运行：请求放行。
15    Closed,
16    /// 已熔断：在 `reset_timeout` 内拦截所有请求。
17    Open,
18    /// 试探：熔断超时后放行单个试探请求。
19    HalfOpen,
20}
21
22/// 断路器抽象 trait。
23///
24/// 实现方需维护失败计数与状态机；连接池通过该 trait 统一驱动
25/// （不依赖具体实现类型，便于测试替换）。
26pub trait CircuitBreaker: Send + Sync {
27    /// 请求是否可执行。`Open` 且熔断超时后自动转为 `HalfOpen` 并放行。
28    fn can_execute(&mut self) -> bool;
29    /// 记录一次成功（复位失败计数，回到 `Closed`）。
30    fn record_success(&mut self);
31    /// 记录一次失败（达到阈值后熔断为 `Open`）。
32    fn record_failure(&mut self);
33    /// 当前状态。
34    fn state(&self) -> CircuitState;
35    /// 手动重置到 `Closed`。返回是否实际发生了状态变更。
36    fn reset(&mut self) -> bool;
37}
38
39/// 默认断路器实现：连续失败 `failure_threshold` 次后熔断，
40/// `reset_timeout` 过后进入 `HalfOpen` 试探，成功恢复 `Closed`，失败回到 `Open`。
41pub struct DefaultCircuitBreaker {
42    failure_threshold: usize,
43    reset_timeout: Duration,
44    state: CircuitState,
45    consecutive_failures: usize,
46    last_failure_at: Option<Instant>,
47}
48
49impl DefaultCircuitBreaker {
50    /// 创建断路器。
51    ///
52    /// - `failure_threshold`：连续失败多少次后熔断（`Open`）；
53    /// - `reset_timeout`：熔断后等待多久进入 `HalfOpen` 试探。
54    pub fn new(failure_threshold: usize, reset_timeout: Duration) -> Self {
55        Self {
56            failure_threshold,
57            reset_timeout,
58            state: CircuitState::Closed,
59            consecutive_failures: 0,
60            last_failure_at: None,
61        }
62    }
63}
64
65impl CircuitBreaker for DefaultCircuitBreaker {
66    fn can_execute(&mut self) -> bool {
67        match self.state {
68            CircuitState::Closed => true,
69            CircuitState::HalfOpen => true,
70            CircuitState::Open => {
71                let elapsed = self
72                    .last_failure_at
73                    .map(|t| t.elapsed())
74                    .unwrap_or(Duration::ZERO);
75                if elapsed >= self.reset_timeout {
76                    self.state = CircuitState::HalfOpen;
77                    true
78                } else {
79                    false
80                }
81            }
82        }
83    }
84
85    fn record_success(&mut self) {
86        self.consecutive_failures = 0;
87        self.state = CircuitState::Closed;
88        self.last_failure_at = None;
89    }
90
91    fn record_failure(&mut self) {
92        self.consecutive_failures += 1;
93        self.last_failure_at = Some(Instant::now());
94        if self.consecutive_failures >= self.failure_threshold {
95            self.state = CircuitState::Open;
96        }
97    }
98
99    fn state(&self) -> CircuitState {
100        self.state
101    }
102
103    fn reset(&mut self) -> bool {
104        let changed = self.state != CircuitState::Closed || self.consecutive_failures != 0;
105        self.state = CircuitState::Closed;
106        self.consecutive_failures = 0;
107        self.last_failure_at = None;
108        changed
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    #[test]
117    fn test_circuit_breaker_starts_closed() {
118        let mut cb = DefaultCircuitBreaker::new(3, Duration::from_secs(60));
119        assert_eq!(cb.state(), CircuitState::Closed);
120        assert!(cb.can_execute());
121    }
122
123    #[test]
124    fn test_circuit_breaker_trips_after_threshold() {
125        let mut cb = DefaultCircuitBreaker::new(3, Duration::from_secs(60));
126        cb.record_failure();
127        cb.record_failure();
128        assert_eq!(cb.state(), CircuitState::Closed);
129        cb.record_failure();
130        assert_eq!(cb.state(), CircuitState::Open);
131        assert!(!cb.can_execute());
132    }
133
134    #[test]
135    fn test_circuit_breaker_success_resets() {
136        let mut cb = DefaultCircuitBreaker::new(2, Duration::from_secs(60));
137        cb.record_failure();
138        cb.record_success();
139        assert_eq!(cb.state(), CircuitState::Closed);
140        // 成功后失败计数已清零，需重新积累阈值
141        cb.record_failure();
142        assert_eq!(cb.state(), CircuitState::Closed);
143        cb.record_failure();
144        assert_eq!(cb.state(), CircuitState::Open);
145    }
146
147    #[test]
148    fn test_circuit_breaker_half_open_after_timeout() {
149        let mut cb = DefaultCircuitBreaker::new(1, Duration::from_millis(10));
150        cb.record_failure();
151        assert_eq!(cb.state(), CircuitState::Open);
152        // 超时后 can_execute 自动进入 HalfOpen 并放行
153        std::thread::sleep(Duration::from_millis(20));
154        assert!(cb.can_execute());
155        assert_eq!(cb.state(), CircuitState::HalfOpen);
156    }
157
158    #[test]
159    fn test_circuit_breaker_half_open_success_closes() {
160        let mut cb = DefaultCircuitBreaker::new(1, Duration::from_millis(10));
161        cb.record_failure();
162        std::thread::sleep(Duration::from_millis(20));
163        assert!(cb.can_execute()); // HalfOpen
164        cb.record_success();
165        assert_eq!(cb.state(), CircuitState::Closed);
166    }
167
168    #[test]
169    fn test_circuit_breaker_half_open_failure_reopens() {
170        let mut cb = DefaultCircuitBreaker::new(1, Duration::from_millis(10));
171        cb.record_failure();
172        std::thread::sleep(Duration::from_millis(20));
173        assert!(cb.can_execute()); // HalfOpen
174        cb.record_failure();
175        assert_eq!(cb.state(), CircuitState::Open);
176        assert!(!cb.can_execute());
177    }
178
179    #[test]
180    fn test_circuit_breaker_reset() {
181        let mut cb = DefaultCircuitBreaker::new(1, Duration::from_secs(60));
182        cb.record_failure();
183        assert_eq!(cb.state(), CircuitState::Open);
184        assert!(cb.reset());
185        assert_eq!(cb.state(), CircuitState::Closed);
186        assert!(cb.can_execute());
187        // 已关闭时 reset 返回 false
188        assert!(!cb.reset());
189    }
