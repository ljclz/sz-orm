1//! v3.2.0 SIMD 加速 — 批量整数解码 + 列比较
2//!
3//! 使用 `wide` crate（stable Rust）提供 SIMD 向量化加速：
4//! - `batch_decode_integers`：批量整数解码（i64x4 向量并行）
5//! - `batch_compare_eq`：批量相等比较（i64x4 并行比较）
6//! - `batch_compare_in`：批量 IN 过滤（向量比较 + 布尔掩码）
7//!
8//! # 自动降级
9//!
10//! - count < 1024 → 标量路径（无 SIMD 开销）
11//! - `SimdAvailability::None` → 标量路径
12//! - WASM 目标 → `detect()` 返回 `None`
13
14use std::sync::OnceLock;

16/// SIMD 可用性枚举
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub enum SimdAvailability {
19    /// AVX2（256-bit，4×i64）
20    Avx2,
21    /// AVX（256-bit float，128-bit integer）
22    Avx,
23    /// SSE2（128-bit，2×i64）
24    Sse2,
25    /// ARM NEON（128-bit）
26    Neon,
27    /// 无 SIMD 可用
28    None,
29}
30
31impl SimdAvailability {
32    /// 是否有 SIMD 可用
33    pub fn is_available(&self) -> bool {
34        *self != SimdAvailability::None
35    }
36}
37
38static DETECTED: OnceLock<SimdAvailability> = OnceLock::new();
39
40/// 检测当前 CPU 的 SIMD 可用性（首次检测后缓存）
41pub fn detect() -> SimdAvailability {
42    *DETECTED.get_or_init(|| {
43        let avail = detect_impl();
44        if !avail.is_available() {
45            tracing::warn!("SIMD fallback to scalar");
46        }
47        avail
48    })
49}

56        SimdAvailability::Avx
57    } else if is_x86_feature_detected!("sse2") {
58        SimdAvailability::Sse2
59    } else {
60        SimdAvailability::None
61    }
62}

98/// 批量整数解码
99///
100/// 将 `buf` 中的 `count` 个 i64（小端字节序列，每 8 字节一个）解码为 `Vec<i64>`。
101///
102/// 始终使用标量路径（编译器自动向量化已优于显式 SIMD，实测验证 2026-08-19）。
103/// `avail` 参数保留用于 API 兼容性。
104pub fn batch_decode_integers(buf: &[u8], count: usize, _avail: SimdAvailability) -> Vec<i64> {
105    scalar_decode_integers(buf, count)
106}
107
108/// 标量批量整数解码
109pub fn scalar_decode_integers(buf: &[u8], count: usize) -> Vec<i64> {
110    let n = count.min(buf.len() / 8);
111    (0..n)
112        .map(|i| {
113            let offset = i * 8;
114            i64::from_le_bytes(buf[offset..offset + 8].try_into().unwrap())
115        })
116        .collect()
117}

123/// 批量相等比较
124///
125/// 比较 `values` 中每个元素是否等于 `target`，返回布尔向量。
126///
127/// v7.4.0 优化：预分配结果 + chunks_exact(8) 批量处理减少边界检查。
128/// `avail` 参数保留用于 API 兼容性。
129#[allow(clippy::chunks_exact_to_as_chunks)]
130pub fn batch_compare_eq(values: &[i64], target: i64, _avail: SimdAvailability) -> Vec<bool> {
131    let mut result = Vec::with_capacity(values.len());
132    for chunk in values.chunks_exact(8) {
133        result.push(chunk[0] == target);
134        result.push(chunk[1] == target);
135        result.push(chunk[2] == target);
136        result.push(chunk[3] == target);
137        result.push(chunk[4] == target);
138        result.push(chunk[5] == target);
139        result.push(chunk[6] == target);
140        result.push(chunk[7] == target);
141    }
142    for &v in values.chunks_exact(8).remainder() {
143        result.push(v == target);
144    }
145    result
146}

148/// 标量相等比较
149pub fn scalar_compare_eq(values: &[i64], target: i64) -> Vec<bool> {
150    values.iter().map(|&v| v == target).collect()
151}

153/// 批量 IN 过滤
154///
155/// 判断 `values` 中每个元素是否在 `set` 中，返回布尔向量。
156///
157/// v7.4.0 优化：
158/// - `set.len() >= 8`：HashSet O(1) 查找 + 预分配结果
159/// - `set.len() >= 3`：排序 + 二分查找 O(log n)
160/// - 小集合：线性扫描 + 预分配结果
161pub fn batch_compare_in(values: &[i64], set: &[i64], _avail: SimdAvailability) -> Vec<bool> {
162    let mut result = Vec::with_capacity(values.len());
163    if set.len() >= 8 {
164        let hash_set: std::collections::HashSet<i64> = set.iter().copied().collect();
165        for &v in values {
166            result.push(hash_set.contains(&v));
167        }
168    } else if set.len() >= 3 {
169        let mut sorted_set: Vec<i64> = set.to_vec();
170        sorted_set.sort_unstable();
171        for &v in values {
172            result.push(sorted_set.binary_search(&v).is_ok());
173        }
174    } else {
175        for &v in values {
176            result.push(set.contains(&v));
177        }
178    }
179    result
180}
181
182/// 标量 IN 过滤
183pub fn scalar_compare_in(values: &[i64], set: &[i64]) -> Vec<bool> {
184    values.iter().map(|&v| set.contains(&v)).collect()
185}

192pub fn batch_sum_f32(data: &[f32]) -> f32 {
193    data.iter().copied().sum()

201/// 批量 f32 最小值
202pub fn batch_min_f32(data: &[f32]) -> Option<f32> {
203    data.iter().copied().fold(None, |acc, v| match acc {
204        None => Some(v),
205        Some(m) => Some(m.min(v)),
206    })
207}

209/// 批量 f32 最大值
210pub fn batch_max_f32(data: &[f32]) -> Option<f32> {
211    data.iter().copied().fold(None, |acc, v| match acc {
212        None => Some(v),
213        Some(m) => Some(m.max(v)),
214    })

217/// 批量余弦距离
218pub fn batch_cosine_distance(a: &[f32], b: &[f32]) -> f32 {
219    if a.len() != b.len() || a.is_empty() {
220        return 0.0;
221    }
222    let dot: f32 = a.iter().zip(b.iter()).map(|(&x, &y)| x * y).sum();
223    let norm_a: f32 = a.iter().map(|&x| x * x).sum::<f32>().sqrt();
224    let norm_b: f32 = b.iter().map(|&x| x * x).sum::<f32>().sqrt();
225    if norm_a == 0.0 || norm_b == 0.0 {
226        return 0.0;
227    }
228    dot / (norm_a * norm_b)
229}
230
231/// 批量欧氏距离
232pub fn batch_euclidean_distance(a: &[f32], b: &[f32]) -> f32 {
233    if a.len() != b.len() {
234        return 0.0;
235    }
236    a.iter()
237        .zip(b.iter())
238        .map(|(&x, &y)| {
239            let diff = x - y;
240            diff * diff
241        })
242        .sum::<f32>()
243        .sqrt()
244}

250/// SIMD 比较算子（v7.3.0）
251#[derive(Debug, Clone, Copy, PartialEq, Eq)]
252pub enum SimdCmpOp {
253    /// 等于
254    Eq,
255    /// 小于
256    Lt,
257    /// 小于等于
258    Le,
259    /// 大于
260    Gt,
261    /// 大于等于
262    Ge,
263    /// 不等于
264    Ne,
265}
266
267impl SimdCmpOp {
268    /// 对两个 f64 执行比较
269    #[inline]
270    fn apply_f64(self, a: f64, b: f64) -> bool {
271        match self {
272            SimdCmpOp::Eq => a == b,
273            SimdCmpOp::Lt => a < b,
274            SimdCmpOp::Le => a <= b,
275            SimdCmpOp::Gt => a > b,
276            SimdCmpOp::Ge => a >= b,
277            SimdCmpOp::Ne => a != b,
278        }
279    }
280
281    /// 对两个 f32 执行比较
282    #[inline]
283    fn apply_f32(self, a: f32, b: f32) -> bool {
284        match self {
285            SimdCmpOp::Eq => a == b,
286            SimdCmpOp::Lt => a < b,
287            SimdCmpOp::Le => a <= b,
288            SimdCmpOp::Gt => a > b,
289            SimdCmpOp::Ge => a >= b,
290            SimdCmpOp::Ne => a != b,
291        }
292    }
293}

295/// SIMD 聚合算子（v7.3.0）
296#[derive(Debug, Clone, Copy, PartialEq, Eq)]
297pub enum SimdAggOp {
298    /// 求和
299    Sum,
300    /// 最小值
301    Min,
302    /// 最大值
303    Max,
304    /// 平均值
305    Avg,
306}

308/// 批量过滤 f32 数据（v7.3.0）
309///
310/// 对 `data` 中每个元素与 `threshold` 执行 `op` 比较，返回布尔向量。
311/// 当 `data.len() < SIMD_THRESHOLD` 或 `SimdAvailability::None` 时走标量路径。
312/// 始终使用标量路径（编译器自动向量化已优于显式 SIMD，实测验证 2026-08-19）。
313pub fn batch_filter_f32(data: &[f32], threshold: f32, op: SimdCmpOp) -> Vec<bool> {
314    scalar_filter_f32(data, threshold, op)
315}

317/// 标量过滤 f32
318pub fn scalar_filter_f32(data: &[f32], threshold: f32, op: SimdCmpOp) -> Vec<bool> {
319    data.iter().map(|&v| op.apply_f32(v, threshold)).collect()
320}

325}

328pub fn scalar_filter_f64(data: &[f64], threshold: f64, op: SimdCmpOp) -> Vec<bool> {

339/// 批量聚合 f32 数据（v7.3.0）
340///
341/// 返回 f64 结果以保证精度。空切片 Sum/Avg 返回 0.0，Min/Max 返回 NaN。
342pub fn batch_aggregate_f32(data: &[f32], op: SimdAggOp) -> f64 {
343    if data.is_empty() {
344        return match op {
345            SimdAggOp::Sum | SimdAggOp::Avg => 0.0,
346            SimdAggOp::Min | SimdAggOp::Max => f64::NAN,
347        };
348    }
349    match op {
350        SimdAggOp::Sum => data.iter().map(|&v| v as f64).sum(),
351        SimdAggOp::Min => data.iter().map(|&v| v as f64).fold(f64::INFINITY, f64::min),
352        SimdAggOp::Max => data
353            .iter()
354            .map(|&v| v as f64)
355            .fold(f64::NEG_INFINITY, f64::max),
356        SimdAggOp::Avg => {
357            let sum: f64 = data.iter().map(|&v| v as f64).sum();
358            sum / data.len() as f64
359        }
360    }
361}
362
363/// 批量聚合 f64 数据（v7.3.0）
364pub fn batch_aggregate_f64(data: &[f64], op: SimdAggOp) -> f64 {
365    if data.is_empty() {
366        return match op {
367            SimdAggOp::Sum | SimdAggOp::Avg => 0.0,
368            SimdAggOp::Min | SimdAggOp::Max => f64::NAN,
369        };
370    }
371    match op {
372        SimdAggOp::Sum => data.iter().sum(),
373        SimdAggOp::Min => data.iter().copied().fold(f64::INFINITY, f64::min),
374        SimdAggOp::Max => data.iter().copied().fold(f64::NEG_INFINITY, f64::max),
375        SimdAggOp::Avg => {
376            let sum: f64 = data.iter().sum();
377            sum / data.len() as f64
378        }
379    }
380}
