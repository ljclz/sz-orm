1//! v8.8.0 BatchSizeAdvisor — 批量操作自适应分批建议
2//!
3//! 根据数据库类型、列数、总行数和参数上限计算最优批量大小，
4//! 避免超过数据库参数绑定上限，同时最大化批量吞吐量。
5
6use crate::dialect::DbType;
7
8/// 数据库参数绑定上限
9fn param_limit(db_type: DbType) -> usize {
10    match db_type {
11        DbType::MySQL => 65535,
12        DbType::PostgreSQL => 32767,
13        DbType::Oracle => 1000,
14        DbType::Sqlite => 999,
15        DbType::SqlServer => 2100,
16        DbType::ClickHouse => 65535,
17        DbType::Db2 => 32767,
18        _ => 1000,
19    }
20}
21
22/// 批量大小建议器
23pub struct BatchSizeAdvisor;
24
25impl BatchSizeAdvisor {
26    /// 根据数据库类型、列数、总行数计算最优批量大小
27    ///
28    /// 约束：
29    /// - batch_size * columns <= param_limit（不超参数上限）
30    /// - batch_size <= total_rows（不超过总行数）
31    /// - batch_size >= 1（至少 1 行）
32    /// - 优先 500 行，其次受参数上限限制
33    pub fn advise(db_type: DbType, columns: usize, total_rows: usize) -> usize {
34        if columns == 0 || total_rows == 0 {
35            return 1;
36        }
37        let limit = param_limit(db_type);
38        let max_by_params = limit / columns;
39        let ideal = 500usize;
40        let batch_size = ideal.min(max_by_params).min(total_rows);
41        batch_size.max(1)
42    }
43
44    /// 将行列表按建议的批量大小分片，返回每片的 (start, end) 索引范围
45    pub fn batch_ranges(db_type: DbType, columns: usize, total_rows: usize) -> Vec<(usize, usize)> {
46        let batch_size = Self::advise(db_type, columns, total_rows);
47        let mut ranges = Vec::with_capacity(total_rows.div_ceil(batch_size));
48        let mut start = 0;
49        while start < total_rows {
50            let end = (start + batch_size).min(total_rows);
51            ranges.push((start, end));
52            start = end;
53        }
54        ranges
55    }
56}

58#[cfg(test)]
59mod tests {
60    use super::*;
61
62    #[test]
63    fn test_advise_respects_param_limit() {
64        let batch = BatchSizeAdvisor::advise(DbType::Oracle, 10, 10000);
65        assert!(
66            batch * 10 <= 1000,
67            "batch * columns must not exceed param limit"
68        );
69    }
70
71    #[test]
72    fn test_advise_respects_total_rows() {
73        let batch = BatchSizeAdvisor::advise(DbType::MySQL, 3, 100);
74        assert!(batch <= 100, "batch must not exceed total rows");
75    }
76
77    #[test]
78    fn test_advise_minimum_one() {
79        let batch = BatchSizeAdvisor::advise(DbType::Sqlite, 0, 100);
80        assert_eq!(batch, 1, "zero columns should return 1");
81    }
82
83    #[test]
84    fn test_advise_ideal_500() {
85        let batch = BatchSizeAdvisor::advise(DbType::MySQL, 2, 10000);
86        assert_eq!(batch, 500, "should use ideal 500 when no constraint hit");
87    }
88
89    #[test]
90    fn test_batch_ranges_cover_all() {
91        let ranges = BatchSizeAdvisor::batch_ranges(DbType::MySQL, 2, 1000);
92        let covered: usize = ranges.iter().map(|(s, e)| e - s).sum();
93        assert_eq!(covered, 1000, "ranges must cover all rows");
94    }
95}
