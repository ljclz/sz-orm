1//! v6.6.0 Saga 分布式事务协调器
2//!
3//! Saga 模式：正向执行 + 补偿编排，保证最终一致性。
4//!
5//! # 执行流程
6//! 1. 依次执行正向步骤
7//! 2. 步骤 i 失败 → 逆序执行已完成步骤的补偿事务
8//! 3. 全部成功 → Saga 完成
9//!
10//! # 超时处理
11//! - 全局超时：整个 Saga 的最大执行时间
12//! - 单步骤超时：每个步骤的最大执行时间

90/// Saga 状态存储（内存实现）
91pub struct SagaStore {
92    instances: Mutex<HashMap<SagaId, StoredInstance>>,
93}
94
95struct StoredInstance {
96    status: SagaStatus,
97    completed_steps: Vec<StepIndex>,
98    failed_step: Option<StepIndex>,
99}
100
101impl SagaStore {
102    pub fn new() -> Self {
103        Self {
104            instances: Mutex::new(HashMap::new()),
105        }
106    }
107
108    pub fn create(&self, saga_id: &str) {
109        let mut instances = self.instances.lock().unwrap();
110        instances.insert(
111            saga_id.to_string(),
112            StoredInstance {
113                status: SagaStatus::Running,
114                completed_steps: Vec::new(),
115                failed_step: None,
116            },
117        );
118    }
119
120    pub fn mark_step_complete(&self, saga_id: &str, step: StepIndex) {
121        let mut instances = self.instances.lock().unwrap();
122        if let Some(inst) = instances.get_mut(saga_id) {
123            inst.completed_steps.push(step);
124        }
125    }
126
127    pub fn mark_failed(&self, saga_id: &str, step: StepIndex) {
128        let mut instances = self.instances.lock().unwrap();
129        if let Some(inst) = instances.get_mut(saga_id) {
130            inst.failed_step = Some(step);
131            inst.status = SagaStatus::Failed;
132        }
133    }
134
135    pub fn update_status(&self, saga_id: &str, status: SagaStatus) {
136        let mut instances = self.instances.lock().unwrap();
137        if let Some(inst) = instances.get_mut(saga_id) {
138            inst.status = status;
139        }
140    }
141
142    pub fn get_status(&self, saga_id: &str) -> Option<SagaStatus> {
143        let instances = self.instances.lock().unwrap();
144        instances.get(saga_id).map(|i| i.status.clone())
145    }
146
147    pub fn get_completed_steps(&self, saga_id: &str) -> Vec<StepIndex> {
148        let instances = self.instances.lock().unwrap();
149        instances
150            .get(saga_id)
151            .map(|i| i.completed_steps.clone())
152            .unwrap_or_default()
153    }
154}
155
156impl Default for SagaStore {
157    fn default() -> Self {
158        Self::new()
159    }
160}

194    /// 执行 Saga
195    ///
196    /// 依次执行正向步骤，失败时逆序补偿。
197    pub async fn execute(&self, saga_id: &str, steps: Vec<SagaStep>) -> SagaResult {
198        let start = Instant::now();
199        let _n = steps.len();
200        self.store.create(saga_id);
201
202        let mut completed: Vec<StepIndex> = Vec::new();
203        let mut compensates: Vec<(
204            StepIndex,
205            Box<dyn FnOnce() -> BoxFuture<'static, Result<(), String>> + Send>,
206        )> = Vec::new();
207
208        for (i, step) in steps.into_iter().enumerate() {
209            if let Some(overall) = self.timeout_config.overall {
210                if start.elapsed() >= overall {
211                    self.store.update_status(saga_id, SagaStatus::TimedOut);
212                    return SagaResult {
213                        saga_id: saga_id.to_string(),
214                        status: SagaStatus::TimedOut,
215                        completed_steps: completed,
216                        failed_step: None,
217                        error: Some("overall timeout".into()),
218                        elapsed: start.elapsed(),
219                    };
220                }
221            }
222
223            let action = step.action;
224            if let Some(c) = step.compensate {
225                compensates.push((i, c));
226            }
227
228            let step_result = if let Some(per_step) = self.timeout_config.per_step {
229                match tokio::time::timeout(per_step, action()).await {
230                    Ok(r) => r,
231                    Err(_) => Err(format!("step {} timed out", step.name)),
232                }
233            } else {
234                action().await
235            };
236
237            match step_result {
238                Ok(()) => {
239                    completed.push(i);
240                    self.store.mark_step_complete(saga_id, i);
241                }
242                Err(e) => {
243                    self.store.mark_failed(saga_id, i);
244                    let _ = self.compensate(saga_id, &mut compensates).await;
245                    let status = self.store.get_status(saga_id).unwrap_or(SagaStatus::Failed);
246                    return SagaResult {
247                        saga_id: saga_id.to_string(),
248                        status,
249                        completed_steps: completed,
250                        failed_step: Some(i),
251                        error: Some(e),
252                        elapsed: start.elapsed(),
253                    };
254                }
255            }
256        }
257
258        self.store.update_status(saga_id, SagaStatus::Completed);
259        SagaResult {
260            saga_id: saga_id.to_string(),
261            status: SagaStatus::Completed,
262            completed_steps: completed,
263            failed_step: None,
264            error: None,
265            elapsed: start.elapsed(),
266        }
267    }
268
269    async fn compensate(
270        &self,
271        saga_id: &str,
272        compensates: &mut Vec<(
273            StepIndex,
274            Box<dyn FnOnce() -> BoxFuture<'static, Result<(), String>> + Send>,
275        )>,
276    ) -> Result<(), String> {
277        let mut all_ok = true;
278        while let Some((idx, comp)) = compensates.pop() {
279            let result = if let Some(per_step) = self.timeout_config.per_step {
280                match tokio::time::timeout(per_step, comp()).await {
281                    Ok(r) => r,
282                    Err(_) => Err(format!("compensate step {} timed out", idx)),
283                }
284            } else {
285                comp().await
286            };
287            if result.is_err() {
288                all_ok = false;
289            }
290        }
291        if all_ok {
292            self.store.update_status(saga_id, SagaStatus::Compensated);
293            Ok(())
294        } else {
295            self.store.update_status(saga_id, SagaStatus::Failed);
296            Err("compensation failed".into())
297        }
298    }

310/// 事务重试错误
311#[derive(Debug)]
312pub enum TxRetryError {
313    /// 不可重试错误
314    NonRetryable(DbError),
315    /// 重试耗尽
316    RetryExhausted { attempts: u32, last_error: DbError },
317    /// 风暴 guard 拒绝（并发重试过多）
318    StormGuardRejected,
319    /// 补偿失败（需人工介入）
320    CompensateFailed(String),
321}

346/// 重试风暴 guard
347///
348/// 限制并发重试数，防止重试风暴。
349pub struct RetryStormGuard {
350    max_concurrent_retries: u32,
351    current_retries: AtomicU32,
352}
353
354impl RetryStormGuard {
355    /// 创建风暴 guard
356    pub fn new(max_concurrent_retries: u32) -> Self {
357        Self {
358            max_concurrent_retries,
359            current_retries: AtomicU32::new(0),
360        }
361    }
362
363    /// 尝试获取重试槽位
364    ///
365    /// 返回 true 表示获取成功，false 表示并发重试已满。
366    pub fn try_acquire(&self) -> bool {
367        let current = self.current_retries.load(AtomicOrdering::Relaxed);
368        if current >= self.max_concurrent_retries {
369            return false;
370        }
371        self.current_retries
372            .compare_exchange(
373                current,
374                current + 1,
375                AtomicOrdering::AcqRel,
376                AtomicOrdering::Relaxed,
377            )
378            .is_ok()
379    }
380
381    /// 释放重试槽位
382    pub fn release(&self) {
383        self.current_retries.fetch_sub(1, AtomicOrdering::Relaxed);
384    }
385
386    /// 当前并发重试数
387    pub fn current_retries(&self) -> u32 {
388        self.current_retries.load(AtomicOrdering::Relaxed)
389    }
390}

427/// 事务重试策略
428///
429/// 可重试错误自动重试（指数退避 + 抖动 + 最大重试次数），
430/// 不可重试错误立即失败，重试失败触发 Saga 补偿回滚，
431/// 重试抖动限流防风暴。
432pub struct TxRetryPolicy {
433    retry_policy: RetryPolicy,
434    storm_guard: RetryStormGuard,
435    compensate_trigger: SagaCompensateTrigger,
436    /// 是否幂等（幂等事务可重试，非幂等跳过重试）
437    idempotent: bool,
438}
439
440impl TxRetryPolicy {
441    /// 创建事务重试策略
442    pub fn new(retry_policy: RetryPolicy, max_concurrent_retries: u32, idempotent: bool) -> Self {
443        Self {
444            retry_policy,
445            storm_guard: RetryStormGuard::new(max_concurrent_retries),
446            compensate_trigger: SagaCompensateTrigger::new(),
447            idempotent,
448        }
449    }
450
451    /// 带重试执行操作
452    ///
453    /// - 幂等事务：可重试错误自动重试，不可重试立即失败
454    /// - 非幂等事务：任何错误立即失败（不重试）
455    /// - 重试耗尽：触发 Saga 补偿
456    pub fn execute_with_retry<F, T>(&self, operation: F) -> Result<T, TxRetryError>
457    where
458        F: FnMut() -> Result<T, DbError>,
459    {
460        if !self.idempotent {
461            let mut op = operation;
462            return op().map_err(TxRetryError::NonRetryable);
463        }
464
465        if !self.storm_guard.try_acquire() {
466            return Err(TxRetryError::StormGuardRejected);
467        }
468
469        let mut op = operation;
470        let mut last_error: Option<DbError> = None;
471        let mut attempts = 0u32;
472
473        for attempt in 0..=self.retry_policy.max_retries {
474            attempts = attempt + 1;
475            match op() {
476                Ok(result) => {
477                    self.storm_guard.release();
478                    return Ok(result);
479                }
480                Err(e) => {
481                    if !RetryableClassifier::is_retryable(&e) {
482                        self.storm_guard.release();
483                        return Err(TxRetryError::NonRetryable(e));
484                    }
485                    if attempt == self.retry_policy.max_retries {
486                        last_error = Some(e);
487                        break;
488                    }
489                    last_error = Some(e);
490                    let delay = self.retry_policy.delay(attempt);
491                    std::thread::sleep(delay);
492                }
493            }
494        }
495
496        self.storm_guard.release();
497
498        let last_err = last_error.unwrap_or(DbError::Internal("retry exhausted".into()));
499        self.compensate_trigger.trigger_compensate("saga")?;
500        Err(TxRetryError::RetryExhausted {
501            attempts,
502            last_error: last_err,
503        })
504    }
505
506    /// 重试策略引用
507    pub fn retry_policy(&self) -> &RetryPolicy {
508        &self.retry_policy
509    }
510
511    /// 风暴 guard 引用
512    pub fn storm_guard(&self) -> &RetryStormGuard {
513        &self.storm_guard
514    }
515
516    /// 补偿触发器引用
517    pub fn compensate_trigger(&self) -> &SagaCompensateTrigger {
518        &self.compensate_trigger
519    }
520
521    /// 是否幂等
522    pub fn is_idempotent(&self) -> bool {
523        self.idempotent
524    }
525}
