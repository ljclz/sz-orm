1//! 强类型 AST 支持模块
2//!
3//! 为 `typed_query!` 宏生成的代码提供 trait 基础。
4//!
5//! # 设计
6//!
7//! `typed_query!` 宏在编译期为每张表生成：
8//! - 一个 `table` 标记类型，实现 [`TypedTable`]
9//! - 每列一个零大小标记类型，实现 [`TypedColumn`]
10//!
11//! 这样可以把列名 + Rust 类型提升到类型系统，
12//! 让 SQL 列名错误在编译期被捕获，而非运行时。
13//!
14//! # 用法
15//!
16//! ```ignore
17//! use sz_orm_macros::typed_query;
18//! use sz_orm_core::typed::{TypedTable, TypedColumn};
19//!
20//! // 1. 声明表 schema
21//! typed_query! {
22//!     table users {
23//!         id: i64,
24//!         name: String,
25//!     }
26//! }
27//!
28//! // 2. 通过标记类型引用列名（类型安全）
29//! fn select_id() -> &'static str {
30//!     <users::col_id as TypedColumn>::NAME
31//! }
32//! ```
33
34/// 类型安全的表标记 trait
35///
36/// 由 `typed_query!(table <name> { ... })` 宏生成的 `table` 结构体实现。
37///
38/// # 实现约定
39///
40/// - `NAME` 为表名（与数据库表名一致）
41/// - 实现者应为零大小类型（unit struct）
42pub trait TypedTable: 'static {
43    /// 表名
44    const NAME: &'static str;
45}
46
47/// 类型安全的列标记 trait
48///
49/// 由 `typed_query!` 宏为每列生成的 `col_<name>` 结构体实现。
50///
51/// # 实现约定
52///
53/// - `NAME` 为列名（与数据库列名一致）
54/// - `Table` 为该列所属的表标记类型
55/// - `RustType` 为该列映射到的 Rust 类型（如 `i64`、`String`）
56/// - `SqlType` 为该列对应的 SQL 类型标记（默认 `Untyped`，宏生成的列使用默认值）
57/// - 实现者应为零大小类型（unit struct）
58pub trait TypedColumn: 'static {
59    /// 列名
60    const NAME: &'static str;
61
62    /// 该列所属的表
63    type Table: TypedTable;
64
65    /// 该列对应的 Rust 类型
66    type RustType;
67
68    /// 该列对应的 SQL 类型标记
69    ///
70    /// 宏生成的列默认使用 `crate::typed_ast::Untyped`。
71    /// 需要强类型 SQL 检查的场景（如 `typed_ast`）应显式指定此类型。
72    type SqlType: crate::typed_ast::SqlType;
73}
74
75/// 在编译期查找表 schema 常量
76///
77/// `typed_query!` 宏在生成表声明时，会同时生成一个常量
78/// `__SZ_ORM_TYPED_SCHEMA_<TABLE>: &[(&str, &str)]`，
79/// 包含该表所有列的 `(列名, 类型名)` 对。
80///
81/// 此函数在 SELECT 表达式分支中被宏自动调用，
82/// 用于在编译期校验 SELECT 列名是否存在于表声明中。
83///
84/// # 注意
85///
86/// 这是一个常量函数（const fn），但实际上宏生成的代码会直接
87/// 通过 `const` 上下文进行列名匹配，不会真正调用此函数。
88/// 此函数仅作为占位，便于文档说明和未来扩展。
89pub const fn _schema_lookup(_schema: &[(&str, &str)], _col: &str) -> Option<&'static str> {
90    // 编译期 schema 查找由宏直接展开为 `match` 表达式完成
91    // 此函数仅作为类型签名参考保留
92    None
93}

99/// 类型安全多列 SELECT 宏（M3 验收标准）
100///
101/// 解决 Rust 无法将异构类型放入同一数组的限制，
102/// 允许一次性指定多个 `TypedColumn` 类型作为 SELECT 列。
103///
104/// # 用法
105///
106/// ```ignore
107/// use sz_orm_core::select_typed;
108/// use sz_orm_core::typed::{TypedColumn, TypedTable};
109///
110/// // 假设已定义 UserColId / UserColName / UserColEmail
111/// let (sql, params) = select_typed!(query, UserColId, UserColName, UserColEmail)
112///     .build_select_with_params();
113/// // 生成：SELECT id, name, email FROM users
114/// ```
115///
116/// # 编译期保证
117///
118/// 每个参数必须是实现了 `TypedColumn` 的类型。
119/// 若传入非 `TypedColumn` 类型，编译期即报错。
120///
121/// # 与 `select_typed` 方法的关系
122///
123/// `QueryBuilder::select_typed::<Col>()` 是单列版本，需链式调用：
124/// ```ignore
125/// query.select_typed::<Col1>().select_typed::<Col2>().select_typed::<Col3>()
126/// ```
127/// 本宏将其简化为：
128/// ```ignore
129/// select_typed!(query, Col1, Col2, Col3)
130/// ```
131#[macro_export]
132macro_rules! select_typed {
133    ($query:expr, $col:ty) => {
134        $query.select_typed::<$col>()
135    };
136    ($query:expr, $col:ty, $($cols:ty),+) => {
137        $crate::select_typed!($query.select_typed::<$col>(), $($cols),+)
138    };
139}
