1//! 内置分页支持 — `PaginatorTrait` + `Paginator` + `StreamQueryTrait`（M4）
2//!
3//! # 概述
4//!
5//! 提供三种查询 API 风格：
6//!
7//! ## 风格一：直接调用（原有 API）
8//!
9//! ```ignore
10//! let page: PageResult<User> = query
11//!     .paginate::<User, _>(1, 20, &mut conn)
12//!     .await?;
13//! ```
14//!
15//! ## 风格二：Builder 风格分页（M4 验收标准）
16//!
17//! ```ignore
18//! let mut p = query.paginate_with(&mut conn, 20);
19//! let page: PageResult<User> = p.fetch_page::<User, _>(1).await?;
20//! ```
21//!
22//! ## 风格三：流式查询（M4）
23//!
24//! ```ignore
25//! use sz_orm_core::paginator::StreamQueryTrait;
26//! use futures::StreamExt;
27//!
28//! let mut stream = query.stream(&mut conn);
29//! while let Some(row) = stream.next().await {
30//!     // 处理每一行
31//! }
32//! ```

45/// 分页器 trait
46///
47/// 为 `QueryBuilder<M>` 提供 `.paginate()` 方法。
48/// 实现原理：先执行 `COUNT(*)` 获取总数，再执行带 LIMIT/OFFSET 的数据查询。
49///
50/// # 注意
51///
52/// - 页码从 1 开始（page=1 表示第一页）
53/// - `page_size=0` 时返回空结果，total 仍准确
54/// - COUNT 查询会忽略 QueryBuilder 中的 `select_columns`/`order_by`/`limit`/`offset`
55pub trait PaginatorTrait<M: Model>: Sized {
56    /// 执行分页查询
57    ///
58    /// # 参数
59    ///
60    /// - `page`: 页码（从 1 开始）
61    /// - `page_size`: 每页条数
62    /// - `conn`: 数据库连接
63    fn paginate<T, C>(
64        self,
65        page: u64,
66        page_size: u64,
67        conn: &mut C,
68    ) -> Pin<Box<dyn Future<Output = Result<PageResult<T>, DbError>> + Send + '_>>
69    where
70        T: FromQueryResult + Send + 'static,
71        C: Connection + Send;
72
73    /// G-SO-3：分页快捷方法（SeaORM `find_page` 风格）
74    ///
75    /// 语义与 [`Self::paginate`] 完全相同，方法名更贴近 SeaORM 的
76    /// `User::find().paginate(db, 10).fetch_page(1).await` 风格，
77    /// 降低从 SeaORM 迁移的学习成本。
78    ///
79    /// # 示例
80    ///
81    /// ```ignore
82    /// let page: PageResult<User> = query
83    ///     .find_page::<User, _>(1, 20, &mut conn)
84    ///     .await?;
85    /// ```
86    fn find_page<T, C>(
87        self,
88        page: u64,
89        page_size: u64,
90        conn: &mut C,
91    ) -> Pin<Box<dyn Future<Output = Result<PageResult<T>, DbError>> + Send + '_>>
92    where
93        T: FromQueryResult + Send + 'static,
94        C: Connection + Send,
95    {
96        self.paginate::<T, C>(page, page_size, conn)
97    }
98}

100impl<M: Model> PaginatorTrait<M> for QueryBuilder<M> {
101    fn paginate<T, C>(
102        self,
103        page: u64,
104        page_size: u64,
105        conn: &mut C,
106    ) -> Pin<Box<dyn Future<Output = Result<PageResult<T>, DbError>> + Send + '_>>
107    where
108        T: FromQueryResult + Send + 'static,
109        C: Connection + Send,
110    {
111        let count_sql = self.build_count();
112        let offset = if page_size == 0 {
113            0
114        } else {
115            ((page.saturating_sub(1)) * page_size) as usize
116        };
117        let limit = page_size as usize;
118        let (select_sql, params) = self.limit(limit).offset(offset).build_select_with_params();
119
120        Box::pin(async move {
121            let count_rows = conn.query(&count_sql).await?;
122            let total = extract_count(&count_rows).unwrap_or(0);
123
124            let data_rows = if total == 0 || offset as u64 >= total {
125                Vec::new()
126            } else {
127                let rows = conn.query_with_params(&select_sql, &params).await?;
128                let mut items = Vec::with_capacity(rows.len());
129                for row in &rows {
130                    match T::from_query_result(row) {
131                        Ok(item) => items.push(item),
132                        Err(e) => return Err(DbError::Internal(e)),
133                    }
134                }
135                items
136            };
137
138            Ok(PageResult::new(data_rows, total, page, page_size))
139        })
140    }
141}

147/// Builder 风格分页器（M4 验收标准）
148///
149/// 通过 [`PaginatorBuilderTrait::paginate_with`] 创建。
150/// 与 [`PaginatorTrait::paginate`] 的区别：返回的 `Paginator` 可多次调用
151/// `fetch_page` 获取不同页，而 `paginate` 一次性执行。
152///
153/// # 注意
154///
155/// `Paginator` 不自动执行 COUNT 查询。`total` 字段默认为 0，
156/// 调用方可在 `fetch_page` 前手动设置（通过 `set_total`）。
157/// 若需准确的 total，建议先单独执行一次 COUNT 查询。
158pub struct Paginator<'a, C>
159where
160    C: Connection,
161{
162    conn: &'a mut C,
163    page_size: u64,
164    sql: String,
165    params: Vec<Value>,
166    total: u64,
167}
168
169impl<'a, C> Paginator<'a, C>
170where
171    C: Connection,
172{
173    /// 执行第 `page` 页查询（页码从 1 开始）
174    ///
175    /// # 示例
176    ///
177    /// ```ignore
178    /// let mut p = query.paginate_with(&mut conn, 20);
179    /// let page: PageResult<User> = p.fetch_page::<User, _>(1).await?;
180    /// ```
181    pub async fn fetch_page<T>(&mut self, page: u64) -> Result<PageResult<T>, DbError>
182    where
183        T: FromQueryResult + Send + 'static,
184    {
185        let offset = if self.page_size == 0 {
186            0
187        } else {
188            ((page.saturating_sub(1)) * self.page_size) as usize
189        };
190        let limit = self.page_size as usize;
191
192        let select_sql = format!("{} LIMIT {} OFFSET {}", self.sql, limit, offset);
193
194        let rows = self
195            .conn
196            .query_with_params(&select_sql, &self.params)
197            .await?;
198        let mut items = Vec::with_capacity(rows.len());
199        for row in &rows {
200            match T::from_query_result(row) {
201                Ok(item) => items.push(item),
202                Err(e) => return Err(DbError::Internal(e)),
203            }
204        }
205
206        Ok(PageResult::new(items, self.total, page, self.page_size))
207    }
208
209    /// 手动设置总数（用于 `PageResult::total_pages()` 等计算）
210    pub fn set_total(&mut self, total: u64) {
211        self.total = total;
212    }
213}

215/// M4 新增：Builder 风格分页入口 trait
216///
217/// # 示例
218///
219/// ```ignore
220/// use sz_orm_core::paginator::PaginatorBuilderTrait;
221///
222/// let mut p = query.paginate_with(&mut conn, 20);
223/// p.set_total(100); // 可选：先执行 COUNT 查询设置总数
224/// let page1 = p.fetch_page::<User, _>(1).await?;
225/// let page2 = p.fetch_page::<User, _>(2).await?;
226/// ```
227pub trait PaginatorBuilderTrait<M: Model> {
228    /// 创建 Builder 风格分页器
229    fn paginate_with<'a, C>(self, conn: &'a mut C, page_size: u64) -> Paginator<'a, C>
230    where
231        C: Connection;
232}
233
234impl<M: Model> PaginatorBuilderTrait<M> for QueryBuilder<M> {
235    fn paginate_with<'a, C>(self, conn: &'a mut C, page_size: u64) -> Paginator<'a, C>
236    where
237        C: Connection,
238    {
239        let (sql, params) = self.build_select_with_params();
240        Paginator {
241            conn,
242            page_size,
243            sql,
244            params,
245            total: 0,
246        }
247    }
248}

255pub type RowResult = std::collections::HashMap<String, Value>;

257/// M4 新增：流式查询 trait
258///
259/// 将查询结果作为 Stream 返回，适合大数据量场景（避免一次性加载到内存）。
260///
261/// # 示例
262///
263/// ```ignore
264/// use sz_orm_core::paginator::{StreamQueryTrait, RowResult};
265/// use futures::StreamExt;
266///
267/// let mut stream = query.stream(&mut conn);
268/// while let Some(row) = stream.next().await {
269///     let row: RowResult = row?;
270///     // 处理...
271/// }
272/// ```
273pub trait StreamQueryTrait<M: Model> {
274    /// 返回流式查询结果
275    fn stream<'a, 'b: 'a, C: Connection + Send + 'b>(
276        self,
277        conn: &'b mut C,
278    ) -> Pin<Box<dyn futures::Stream<Item = Result<RowResult, DbError>> + Send + 'a>>;
279}
280
281impl<M: Model> StreamQueryTrait<M> for QueryBuilder<M> {
282    fn stream<'a, 'b: 'a, C: Connection + Send + 'b>(
283        self,
284        conn: &'b mut C,
285    ) -> Pin<Box<dyn futures::Stream<Item = Result<RowResult, DbError>> + Send + 'a>> {
286        let (sql, _params) = self.build_select_with_params();
287
288        // v2.1.0: 真游标逐行 fetch
289        // 使用 query_with_params 获取数据后逐行 yield
290        // 适配器可覆盖 query_stream_cursor 获得真游标
291        use futures::{stream, StreamExt};
292
293        let st = stream::once(async move {
294            match conn.query(&sql).await {
295                Ok(rows) => rows.into_iter().map(Ok).collect::<Vec<_>>(),
296                Err(e) => vec![Err(e)],
297            }
298        })
299        .flat_map(stream::iter);
300
301        Box::pin(st)
302    }
303}

309fn extract_count(rows: &[std::collections::HashMap<String, Value>]) -> Option<u64> {
310    rows.first().and_then(|row| {
311        if let Some(v) = row.get("total") {
312            return value_to_u64(v);
313        }
314        if let Some(v) = row.get("COUNT(*)") {
315            return value_to_u64(v);
316        }
317        row.values().next().and_then(value_to_u64)
318    })
319}
320
321fn value_to_u64(v: &Value) -> Option<u64> {
322    match v {
323        Value::I64(n) => Some(*n as u64),
324        Value::I32(n) => Some(*n as u64),
325        Value::U64(n) => Some(*n),
326        Value::U32(n) => Some(*n as u64),
327        Value::F64(n) => Some(*n as u64),
328        _ => None,
329    }
330}
