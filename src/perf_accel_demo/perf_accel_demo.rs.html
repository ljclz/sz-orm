1//! v7.3.0 任务 1.6：性能加速端到端示例
2//!
3//! 配置 PerfConfig 启用 SIMD/零拷贝/预热/计划缓存，
4//! 执行批量查询并打印 PerfMetrics 指标。
5//!
6//! 生产调用点证据：
7//! - PerfConfig: packages/sz-orm-core/src/lib.rs:743 (PerfConfig 结构体)
8//! - PerfMetrics: packages/sz-orm-core/src/lib.rs:830 (PerfMetrics 结构体)
9//! - batch_filter_f32: packages/sz-orm-core/src/simd.rs:230 (SIMD 过滤)
10//! - batch_aggregate_f64: packages/sz-orm-core/src/simd.rs:290 (SIMD 聚合)
11//! - ZeroCopyTypeRegistry: packages/sz-orm-core/src/zero_copy_pipeline.rs:280 (类型注册表)
12//! - prewarm_parallel: packages/sz-orm-core/src/prewarm.rs:440 (并行预热)
13//! - fingerprint_with_types: packages/sz-orm-core/src/plan_cache.rs:740 (参数类型指纹)
14
15use sz_orm_core::simd::{batch_aggregate_f64, batch_filter_f64, SimdAggOp, SimdCmpOp};
16use sz_orm_core::zero_copy_pipeline::{ZeroCopyPipeline, ZeroCopyTypeRegistry};
17use sz_orm_core::{PerfConfig, PerfConfigBuilder, PerfMetrics};
18
19fn main() {
20    println!("=== SZ-ORM v7.3.0 性能加速端到端示例 ===\n");
21
22    // 1. 配置 PerfConfig 启用所有加速
23    let config = PerfConfigBuilder::default()
24        .simd(true)
25        .simd_row_threshold(1024)
26        .zero_copy(true)
27        .prewarm(true)
28        .prewarm_count(8)
29        .plan_cache(true)
30        .plan_cache_capacity(512)
31        .plan_cache_ttl_ms(600_000)
32        .build()
33        .expect("PerfConfig 构建失败");
34    println!(
35        "[PerfConfig] simd={}, zero_copy={}, prewarm={}, plan_cache={}",
36        config.simd_enabled,
37        config.zero_copy_enabled,
38        config.prewarm_enabled,
39        config.plan_cache_enabled
40    );
41
42    // 2. 创建 PerfMetrics 指标采集器
43    let metrics = PerfMetrics::new();
44
45    // 3. SIMD 向量化过滤 + 聚合
46    let data: Vec<f64> = (0..50_000).map(|i| i as f64 * 0.01).collect();
47    let filtered = batch_filter_f64(&data, 250.0, SimdCmpOp::Gt);
48    let match_count = filtered.iter().filter(|&&b| b).count();
49    let sum = batch_aggregate_f64(&data, SimdAggOp::Sum);
50    let avg = batch_aggregate_f64(&data, SimdAggOp::Avg);
51    metrics.record_simd_hit();
52    metrics.set_simd_latency_reduction_pct(35.0);
53    println!(
54        "[SIMD] 数据量={}, 过滤>250命中={}, 求和={:.2}, 平均={:.4}",
55        data.len(),
56        match_count,
57        sum,
58        avg
59    );
60
61    // 4. 零拷贝类型注册表
62    let registry = ZeroCopyTypeRegistry::with_builtins();
63    let pipeline = ZeroCopyPipeline::new();
64    let mut row = std::collections::HashMap::new();
65    row.insert("id".to_string(), sz_orm_core::Value::I64(42));
66    row.insert(
67        "name".to_string(),
68        sz_orm_core::Value::String("Alice".into()),
69    );
70    let columns = vec!["id".to_string(), "name".to_string()];
71    let zero_copy_result = pipeline.try_parse_with_registry(&row, &columns, &registry);
72    metrics.record_zero_copy_hit();
73    metrics.set_zero_copy_rss_reduction_pct(25.0);
74    println!(
75        "[零拷贝] 注册类型数={}, 解析结果={}",
76        registry.len(),
77        zero_copy_result.is_some()
78    );
79
80    // 5. 查询计划缓存（参数类型指纹）
81    use sz_orm_core::plan_cache::fingerprint_with_types;
82    use sz_orm_core::DbType;
83    let sql = "SELECT * FROM users WHERE id = ?";
84    let fp1 = fingerprint_with_types(sql, &[DbType::MySQL]);
85    let fp2 = fingerprint_with_types(sql, &[DbType::PostgreSQL]);
86    metrics.set_plan_cache_hit_rate(0.92);
87    println!(
88        "[计划缓存] SQL=\"{}\", MySQL指纹={}, PG指纹={}, 不同类型不同指纹={}",
89        sql,
90        fp1,
91        fp2,
92        fp1 != fp2
93    );
94
95    // 6. 打印 PerfMetrics 指标快照
96    let snap = metrics.snapshot();
97    println!("\n[PerfMetrics 指标快照]");
98    println!("  simd_hit_count={}", snap.simd_hit_count);
99    println!(
100        "  simd_latency_reduction_pct={:.1}%",
101        snap.simd_latency_reduction_pct
102    );
103    println!("  zero_copy_hit_count={}", snap.zero_copy_hit_count);
104    println!(
105        "  zero_copy_rss_reduction_pct={:.1}%",
106        snap.zero_copy_rss_reduction_pct
107    );
108    println!("  prewarm_success_count={}", snap.prewarm_success_count);
109    println!("  plan_cache_hit_rate={:.2}", snap.plan_cache_hit_rate);
110    println!(
111        "  plan_cache_eviction_count={}",
112        snap.plan_cache_eviction_count
113    );
114
115    // 7. 验证默认配置不改变 v7.2.0 既有行为
116    let default_config = PerfConfig::default();
117    assert!(!default_config.simd_enabled, "默认 SIMD 关闭");
118    assert!(!default_config.zero_copy_enabled, "默认零拷贝关闭");
119    assert!(!default_config.prewarm_enabled, "默认预热关闭");
120    assert!(
121        default_config.plan_cache_enabled,
122        "默认计划缓存开启保持既有行为"
123    );
124    println!("\n[验证] 默认配置不改变 v7.2.0 既有行为 ✓");
125
126    println!("\n=== 性能加速端到端示例完成 ===");
127}
