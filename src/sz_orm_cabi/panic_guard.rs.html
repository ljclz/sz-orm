1//! FFI panic 捕获
2//!
3//! 捕获 Rust 侧 panic 转换为错误码，防止 panic 跨 FFI 边界导致 UB。
4
5use std::panic::UnwindSafe;
6
7/// panic 捕获结果
8pub enum FfiResult<T> {
9    Ok(T),
10    Panic(String),
11}
12
13/// 捕获可能 panic 的闭包，返回 `FfiResult`
14///
15/// SAFETY: 使用 `catch_unwind` 捕获 panic，防止 panic 跨 FFI 边界。
16pub fn catch_panic<F, T>(f: F) -> FfiResult<T>
17where
18    F: FnOnce() -> T + UnwindSafe,
19{
20    match std::panic::catch_unwind(f) {
21        Ok(result) => FfiResult::Ok(result),
22        Err(panic_payload) => {
23            let msg = if let Some(s) = panic_payload.downcast_ref::<&str>() {
24                s.to_string()
25            } else if let Some(s) = panic_payload.downcast_ref::<String>() {
26                s.clone()
27            } else {
28                "unknown panic".to_string()
29            };
30            FfiResult::Panic(msg)
31        }
32    }
33}
34
35/// 捕获可能 panic 的闭包，panic 时返回默认值
36pub fn catch_panic_or_default<F, T>(f: F, default: T) -> T
37where
38    F: FnOnce() -> T + UnwindSafe,
39{
40    match catch_panic(f) {
41        FfiResult::Ok(result) => result,
42        FfiResult::Panic(_) => default,
43    }
44}

46#[cfg(test)]
47mod tests {
48    use super::*;
49
50    #[test]
51    fn test_catch_panic_ok() {
52        let result = catch_panic(|| 42);
53        match result {
54            FfiResult::Ok(v) => assert_eq!(v, 42),
55            FfiResult::Panic(_) => panic!("expected Ok"),
56        }
57    }
58
59    #[test]
60    fn test_catch_panic_str() {
61        let result = catch_panic(|| panic!("test panic"));
62        match result {
63            FfiResult::Ok(_) => panic!("expected Panic"),
64            FfiResult::Panic(msg) => assert_eq!(msg, "test panic"),
65        }
66    }
67
68    #[test]
69    fn test_catch_panic_string() {
70        let result = catch_panic(|| panic!("{}", "formatted panic"));
71        match result {
72            FfiResult::Ok(_) => panic!("expected Panic"),
73            FfiResult::Panic(msg) => assert!(msg.contains("formatted panic")),
74        }
75    }
76
77    #[test]
78    fn test_catch_panic_or_default_ok() {
79        let result = catch_panic_or_default(|| 100, 0);
80        assert_eq!(result, 100);
81    }
82
83    #[test]
84    fn test_catch_panic_or_default_panic() {
