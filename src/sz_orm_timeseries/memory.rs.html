11/// 内存 TimescaleDB 实现
12pub struct MemoryTimeseries {
13    /// 指标存储：metric_name -> `Vec<Metric>`
14    storage: Mutex<HashMap<String, Vec<Metric>>>,
15    /// hypertable 注册表
16    hypertables: Mutex<Vec<(String, String)>>,
17    /// 连续聚合视图
18    continuous_aggregates: Mutex<Vec<(String, String)>>,
19}
20
21impl MemoryTimeseries {
22    pub fn new() -> Self {
23        Self {
24            storage: Mutex::new(HashMap::new()),
25            hypertables: Mutex::new(Vec::new()),
26            continuous_aggregates: Mutex::new(Vec::new()),
27        }
28    }
29
30    /// 解析时间桶字符串（如 "1h"、"5m"、"1d"）为秒数
31    fn parse_bucket(bucket: &str) -> Result<i64, TimescaleError> {
32        if bucket.is_empty() {
33            return Err(TimescaleError::InvalidConfig(
34                "bucket cannot be empty".to_string(),
35            ));
36        }
37        let (num_str, unit) = bucket.split_at(bucket.len() - 1);
38        let num: i64 = num_str.parse().map_err(|_| {
39            TimescaleError::InvalidConfig(format!("invalid bucket number: {}", num_str))
40        })?;
41        let seconds = match unit {
42            "s" => num,
43            "m" => num * 60,
44            "h" => num * 3600,
45            "d" => num * 86400,
46            _ => {
47                return Err(TimescaleError::InvalidConfig(format!(
48                    "unsupported bucket unit: {}",
49                    unit
50                )))
51            }
52        };
53        Ok(seconds)
54    }
55}

63#[async_trait]
64impl TimeseriesExt for MemoryTimeseries {
65    async fn create_hypertable(
66        &self,
67        table: &str,
68        time_column: &str,
69    ) -> Result<(), TimescaleError> {
70        self.hypertables
71            .lock()
72            .push((table.to_string(), time_column.to_string()));
73        // 确保 storage 中有该 metric 的 entry
74        self.storage.lock().entry(table.to_string()).or_default();
75        Ok(())
76    }
77
78    async fn insert_metric(&self, metric: &Metric) -> Result<(), TimescaleError> {
79        let mut storage = self.storage.lock();
80        storage
81            .entry(metric.name.clone())
82            .or_default()
83            .push(metric.clone());
84        Ok(())
85    }
86
87    async fn query_range(
88        &self,
89        metric: &str,
90        start: DateTime<Utc>,
91        end: DateTime<Utc>,
92    ) -> Result<Vec<Metric>, TimescaleError> {
93        // M-17 修复：校验时间范围（start < end 且跨度 <= MAX_QUERY_RANGE_SECS）
94        crate::validate_time_range(start, end)?;
95        let storage = self.storage.lock();
96        let data = storage.get(metric).cloned().unwrap_or_default();
97        Ok(data
98            .into_iter()
99            .filter(|m| m.timestamp >= start && m.timestamp < end)
100            .collect())
101    }
102
103    async fn time_bucket_aggregate(
104        &self,
105        metric: &str,
106        bucket: &str,
107        aggregation: Aggregation,
108        start: DateTime<Utc>,
109        end: DateTime<Utc>,
110    ) -> Result<Vec<TimeBucket>, TimescaleError> {
111        // M-17 修复：校验时间范围（start < end 且跨度 <= MAX_QUERY_RANGE_SECS）
112        crate::validate_time_range(start, end)?;
113        let bucket_secs = Self::parse_bucket(bucket)?;
114        let bucket_duration = chrono::Duration::seconds(bucket_secs);
115
116        // 获取范围内的数据
117        let data = self.query_range(metric, start, end).await?;
118
119        // 按桶分组
120        let mut buckets: HashMap<DateTime<Utc>, Vec<f64>> = HashMap::new();
121        for m in data {
122            let elapsed = (m.timestamp - start).num_seconds();
123            let bucket_idx = elapsed / bucket_secs;
124            let bucket_start = start + chrono::Duration::seconds(bucket_idx * bucket_secs);
125            buckets.entry(bucket_start).or_default().push(m.value);
126        }
127
128        // 生成连续的桶（即使没有数据）
129        let mut result = Vec::new();
130        let mut current = start;
131        while current < end {
132            let values = buckets.remove(&current).unwrap_or_default();
133            let mut bucket = TimeBucket::from_values(current, &values);
134            // 如果桶内无数据，按聚合类型填充 0 或跳过
135            if values.is_empty() {
136                // 空桶：count=0，其他字段保持 0.0
137                bucket = TimeBucket {
138                    bucket_start: current,
139                    count: 0,
140                    sum: 0.0,
141                    min: 0.0,
142                    max: 0.0,
143                    avg: 0.0,
144                };
145            }
146            // 应用聚合（用于验证聚合函数工作）
147            let _agg_value = aggregation.apply(&values);
148            result.push(bucket);
149            current += bucket_duration;
150        }
151
152        Ok(result)
153    }
154
155    async fn create_continuous_aggregate(
156        &self,
157        view_name: &str,
158        query: &str,
159    ) -> Result<(), TimescaleError> {
160        self.continuous_aggregates
161            .lock()
162            .push((view_name.to_string(), query.to_string()));
163        Ok(())
164    }
165
166    async fn downsample(&self, config: &DownsampleConfig) -> Result<(), TimescaleError> {
167        // 内存实现：从源指标读取所有数据，按 interval 聚合，写入目标指标
168        let storage = self.storage.lock();
169        let source_data = storage
170            .get(&config.source_metric)
171            .cloned()
172            .unwrap_or_default();
173        drop(storage);
174
175        if source_data.is_empty() {
176            return Ok(());
177        }
178
179        let bucket_secs = Self::parse_bucket(&config.interval)?;
180        let start = source_data
181            .iter()
182            .map(|m| m.timestamp)
183            .min()
184            .unwrap_or_else(Utc::now);
185
186        // 按桶分组并聚合
187        let mut buckets: HashMap<i64, Vec<f64>> = HashMap::new();
188        for m in &source_data {
189            let elapsed = (m.timestamp - start).num_seconds();
190            let bucket_idx = elapsed / bucket_secs;
191            buckets.entry(bucket_idx).or_default().push(m.value);
192        }
193
194        // 生成降采样后的指标
195        let mut downsampled = Vec::new();
196        for (idx, values) in buckets {
197            let bucket_start = start + chrono::Duration::seconds(idx * bucket_secs);
198            let agg_value = config.aggregation.apply(&values);
199            downsampled.push(Metric::new(
200                config.target_metric.clone(),
201                bucket_start,
202                agg_value,
203            ));
204        }
205
206        // 写入目标指标
207        let mut storage = self.storage.lock();
208        let target = storage.entry(config.target_metric.clone()).or_default();
209        target.extend(downsampled);
210
211        Ok(())
212    }
213
214    async fn drop_metric(&self, metric: &str) -> Result<(), TimescaleError> {
215        let mut storage = self.storage.lock();
216        if storage.remove(metric).is_none() {
217            return Err(TimescaleError::NotFound(metric.to_string()));
218        }
219        Ok(())
220    }
221}

321    fn test_m17_validate_time_range_function() {
322        let start = Utc.with_ymd_and_hms(2026, 1, 1, 0, 0, 0).unwrap();
323        // start == end：拒绝
324        assert!(crate::validate_time_range(start, start).is_err());
325        // start > end：拒绝
326        let end = start - chrono::Duration::seconds(1);
327        assert!(crate::validate_time_range(start, end).is_err());
328        // 正常范围：允许
329        let end = start + chrono::Duration::days(30);
330        assert!(crate::validate_time_range(start, end).is_ok());
331        // 366 天：允许
332        let end = start + chrono::Duration::seconds(crate::MAX_QUERY_RANGE_SECS);
333        assert!(crate::validate_time_range(start, end).is_ok());
334        // 366 天 + 1 秒：拒绝
335        let end = start + chrono::Duration::seconds(crate::MAX_QUERY_RANGE_SECS + 1);
336        assert!(crate::validate_time_range(start, end).is_err());
337    }

340    async fn test_time_bucket_aggregate() {
341        let ts = MemoryTimeseries::new();
342        // 每 10 分钟一个数据点，4 个点
343        for i in 0..4 {
344            ts.insert_metric(&make_metric("cpu", i * 10, 0.1 * (i + 1) as f64))
345                .await
346                .unwrap();
347        }
348
349        let start = Utc.with_ymd_and_hms(2026, 7, 20, 0, 0, 0).unwrap();
350        let end = Utc.with_ymd_and_hms(2026, 7, 20, 1, 0, 0).unwrap();
351        let buckets = ts
352            .time_bucket_aggregate("cpu", "30m", Aggregation::Avg, start, end)
353            .await
354            .unwrap();
355        // 1 小时按 30 分钟分桶 = 2 个桶
356        assert_eq!(buckets.len(), 2);
357        // 第一个桶（0-30min）：包含 0, 10, 20 分钟的数据
358        assert_eq!(buckets[0].count, 3);
359        // (0.1+0.2+0.3)/3 = 0.2
360        assert!((buckets[0].avg - 0.2).abs() < 1e-10);
361        // 第二个桶（30-60min）：包含 30 分钟的数据
362        assert_eq!(buckets[1].count, 1);
363        assert!((buckets[1].avg - 0.4).abs() < 1e-10);
364    }
