1//! N+1 查询检测规则引擎
2//!
3//! 可配置的规则引擎，按规则集检测 N+1 查询模式。
4//! 支持自定义规则、规则优先级、规则启用/禁用。
5
6use std::collections::HashMap;
7
8use crate::{N1Finding, N1Pattern, N1Severity};
9
10/// 检测规则
11#[derive(Debug, Clone)]
12pub struct DetectionRule {
13    /// 规则 ID
14    pub rule_id: String,
15    /// 规则描述
16    pub description: String,
17    /// 匹配的模式
18    pub pattern: N1Pattern,
19    /// 规则优先级（0=最高）
20    pub priority: u8,
21    /// 是否启用
22    pub enabled: bool,
23}
24
25impl DetectionRule {
26    /// 创建新规则
27    pub fn new(
28        rule_id: impl Into<String>,
29        description: impl Into<String>,
30        pattern: N1Pattern,
31    ) -> Self {
32        Self {
33            rule_id: rule_id.into(),
34            description: description.into(),
35            pattern,
36            priority: N1Severity::from_pattern(pattern) as u8,
37            enabled: true,
38        }
39    }
40
41    /// 设置优先级
42    pub fn with_priority(mut self, priority: u8) -> Self {
43        self.priority = priority;
44        self
45    }
46
47    /// 禁用规则
48    pub fn disabled(mut self) -> Self {
49        self.enabled = false;
50        self
51    }
52}
53
54/// 规则匹配结果
55#[derive(Debug, Clone)]
56pub struct RuleMatchResult {
57    /// 匹配的规则 ID
58    pub rule_id: String,
59    /// 匹配的检测结果
60    pub finding: N1Finding,
61    /// 规则优先级
62    pub priority: u8,
63}
64
65/// N+1 查询检测规则引擎
66///
67/// 按配置的规则集检测 N+1 查询，支持规则启用/禁用与优先级排序。
68pub struct N1DetectionRuleEngine {
69    rules: Vec<DetectionRule>,
70}
71
72impl N1DetectionRuleEngine {
73    /// 创建空规则引擎
74    pub fn new() -> Self {
75        Self { rules: Vec::new() }
76    }
77
78    /// 创建带默认规则集的引擎
79    pub fn with_default_rules() -> Self {
80        let mut engine = Self::new();
81        engine.add_rule(DetectionRule::new(
82            "query-in-loop",
83            "检测循环体内直接查询调用",
84            N1Pattern::QueryInLoop,
85        ));
86        engine.add_rule(DetectionRule::new(
87            "conditional-query-in-loop",
88            "检测循环体内条件分支中的查询调用",
89            N1Pattern::ConditionalQueryInLoop,
90        ));
91        engine.add_rule(DetectionRule::new(
92            "missing-eager-load",
93            "检测可批量替代的单条查询",
94            N1Pattern::MissingEagerLoadHint,
95        ));
96        engine
97    }
98
99    /// 添加规则
100    pub fn add_rule(&mut self, rule: DetectionRule) {
101        self.rules.push(rule);
102        self.rules.sort_by_key(|r| r.priority);
103    }
104
105    /// 按 ID 禁用规则
106    pub fn disable_rule(&mut self, rule_id: &str) -> bool {
107        let mut found = false;
108        for rule in &mut self.rules {
109            if rule.rule_id == rule_id {
110                rule.enabled = false;
111                found = true;
112            }
113        }
114        found
115    }
116
117    /// 按 ID 启用规则
118    pub fn enable_rule(&mut self, rule_id: &str) -> bool {
119        let mut found = false;
120        for rule in &mut self.rules {
121            if rule.rule_id == rule_id {
122                rule.enabled = true;
123                found = true;
124            }
125        }
126        found
127    }
128
129    /// 获取所有规则
130    pub fn rules(&self) -> &[DetectionRule] {
131        &self.rules
132    }
133
134    /// 启用的规则数
135    pub fn enabled_rule_count(&self) -> usize {
136        self.rules.iter().filter(|r| r.enabled).count()
137    }
138
139    /// 对检测结果应用规则
140    ///
141    /// 仅保留匹配启用规则的检测结果，并附加规则 ID 与优先级。
142    pub fn apply(&self, findings: &[N1Finding]) -> Vec<RuleMatchResult> {
143        let mut results = Vec::new();
144        for finding in findings {
145            for rule in &self.rules {
146                if rule.enabled && rule.pattern == finding.pattern {
147                    results.push(RuleMatchResult {
148                        rule_id: rule.rule_id.clone(),
149                        finding: finding.clone(),
150                        priority: rule.priority,
151                    });
152                    break;
153                }
154            }
155        }
156        results.sort_by_key(|r| r.priority);
157        results
158    }
159
160    /// 按规则 ID 分组统计
161    pub fn group_by_rule(&self, findings: &[N1Finding]) -> HashMap<String, Vec<N1Finding>> {
162        let mut groups: HashMap<String, Vec<N1Finding>> = HashMap::new();
163        for finding in findings {
164            for rule in &self.rules {
165                if rule.enabled && rule.pattern == finding.pattern {
166                    groups
167                        .entry(rule.rule_id.clone())
168                        .or_default()
169                        .push(finding.clone());
170                    break;
171                }
172            }
173        }
174        groups
175    }
176
177    /// 清空规则
178    pub fn clear(&mut self) {
179        self.rules.clear();
180    }
181}
182
183impl Default for N1DetectionRuleEngine {
184    fn default() -> Self {
185        Self::with_default_rules()
186    }
187}

189/// 规则引擎统计
190#[derive(Debug, Clone, Default)]
191pub struct RuleEngineStats {
192    /// 总匹配数
193    pub total_matches: usize,
194    /// 按规则 ID 统计
195    pub matches_by_rule: HashMap<String, usize>,
196    /// 按模式统计
197    pub matches_by_pattern: HashMap<N1Pattern, usize>,
198}
199
200impl RuleEngineStats {
201    /// 创建空统计
202    pub fn new() -> Self {
203        Self::default()
204    }
205
206    /// 从匹配结果构建统计
207    pub fn from_results(results: &[RuleMatchResult]) -> Self {
208        let mut stats = Self::new();
209        stats.total_matches = results.len();
210        for result in results {
211            *stats
212                .matches_by_rule
213                .entry(result.rule_id.clone())
214                .or_insert(0) += 1;
215            *stats
216                .matches_by_pattern
217                .entry(result.finding.pattern)
218                .or_insert(0) += 1;
219        }
220        stats
221    }
222
223    /// 某规则的匹配数
224    pub fn count_for_rule(&self, rule_id: &str) -> usize {
225        self.matches_by_rule.get(rule_id).copied().unwrap_or(0)
226    }
227
228    /// 某模式的匹配数
229    pub fn count_for_pattern(&self, pattern: N1Pattern) -> usize {
230        self.matches_by_pattern.get(&pattern).copied().unwrap_or(0)
231    }
232}
