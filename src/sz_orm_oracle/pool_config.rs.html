1//! Oracle 连接池配置模块
2//!
3//! 提供 [`OraclePoolConfig`] 用于配置连接池参数，包括大小上限、空闲超时、
4//! 健康检查间隔等；[`PoolStats`] 用于统计连接池运行指标。
5
6use std::fmt;
7use std::time::Duration;
8
9/// Oracle 连接池配置
10///
11/// 控制 `OraclePoolHandle` 的行为参数。所有字段均通过 builder 风格的
12/// `with_*` 方法设置，或通过 [`OraclePoolConfig::from_env`] 从环境变量加载。
13#[derive(Debug, Clone)]
14pub struct OraclePoolConfig {
15    /// 连接池大小上限（默认 10）
16    pub max_size: usize,
17    /// 空闲连接超时（默认 30 分钟），超时后连接将被关闭
18    pub idle_timeout: Duration,
19    /// 健康检查间隔（默认 60 秒），定期 ping 空闲连接
20    pub health_check_interval: Duration,
21    /// 获取连接超时（默认 5 秒），超时返回错误
22    pub acquire_timeout: Duration,
23    /// 连接建立超时（默认 10 秒）
24    pub connect_timeout: Duration,
25    /// 最大连接生命周期（默认 1 小时），超时后连接将被重建
26    pub max_lifetime: Duration,
27    /// 是否启用连接健康检查（默认 true）
28    pub health_check_enabled: bool,
29    /// 最小空闲连接数（默认 0），低于此数将预创建连接
30    pub min_idle: usize,
31}
32
33impl Default for OraclePoolConfig {
34    fn default() -> Self {
35        Self {
36            max_size: 10,
37            idle_timeout: Duration::from_secs(1800),
38            health_check_interval: Duration::from_secs(60),
39            acquire_timeout: Duration::from_secs(5),
40            connect_timeout: Duration::from_secs(10),
41            max_lifetime: Duration::from_secs(3600),
42            health_check_enabled: true,
43            min_idle: 0,
44        }
45    }
46}

107        self.min_idle = min_idle;
108        self
109    }
110
111    /// 验证配置有效性
112    ///
113    /// # Errors
114    ///
115    /// 若配置无效返回 `Err(String)` 描述错误原因。
116    pub fn validate(&self) -> Result<(), String> {
117        if self.max_size == 0 {
118            return Err("max_size must be greater than 0".to_string());
119        }
120        if self.min_idle > self.max_size {
121            return Err("min_idle cannot exceed max_size".to_string());
122        }
123        if self.idle_timeout.as_secs() == 0 {
124            return Err("idle_timeout must be greater than 0".to_string());
125        }
126        if self.acquire_timeout.as_secs() == 0 {
127            return Err("acquire_timeout must be greater than 0".to_string());
128        }
129        Ok(())
130    }
131
132    /// 从环境变量构建配置
133    ///
134    /// 支持的环境变量：
135    /// - `SZ_ORM_ORACLE_POOL_MAX_SIZE`：连接池大小上限
136    /// - `SZ_ORM_ORACLE_POOL_MIN_IDLE`：最小空闲连接数
137    /// - `SZ_ORM_ORACLE_POOL_IDLE_TIMEOUT`：空闲超时（秒）
138    /// - `SZ_ORM_ORACLE_POOL_ACQUIRE_TIMEOUT`：获取超时（秒）
139    /// - `SZ_ORM_ORACLE_POOL_MAX_LIFETIME`：最大生命周期（秒）
140    #[must_use]
141    pub fn from_env() -> Self {
142        let mut config = Self::default();
143        if let Ok(val) = std::env::var("SZ_ORM_ORACLE_POOL_MAX_SIZE") {
144            if let Ok(n) = val.parse::<usize>() {
145                if n > 0 {
146                    config.max_size = n;
147                }
148            }
149        }
150        if let Ok(val) = std::env::var("SZ_ORM_ORACLE_POOL_MIN_IDLE") {
151            if let Ok(n) = val.parse::<usize>() {
152                config.min_idle = n.min(config.max_size);
153            }
154        }
155        if let Ok(val) = std::env::var("SZ_ORM_ORACLE_POOL_IDLE_TIMEOUT") {
156            if let Ok(n) = val.parse::<u64>() {
157                config.idle_timeout = Duration::from_secs(n);
158            }
159        }
160        if let Ok(val) = std::env::var("SZ_ORM_ORACLE_POOL_ACQUIRE_TIMEOUT") {
161            if let Ok(n) = val.parse::<u64>() {
162                config.acquire_timeout = Duration::from_secs(n);
163            }
164        }
165        if let Ok(val) = std::env::var("SZ_ORM_ORACLE_POOL_MAX_LIFETIME") {
166            if let Ok(n) = val.parse::<u64>() {
167                config.max_lifetime = Duration::from_secs(n);
168            }
169        }
170        config
171    }
172
173    /// 计算有效连接数（`max(max_size, min_idle)`）
174    #[must_use]
175    pub fn effective_connections(&self) -> usize {
176        self.max_size.max(self.min_idle)
177    }
178
179    /// 是否需要预创建连接
180    #[must_use]
181    pub fn needs_preconnect(&self) -> bool {
182        self.min_idle > 0
183    }
184
185    /// 估算内存占用（字节）
186    ///
187    /// 每个连接约 64 KiB 缓冲区 + 4 KiB 元数据。
188    #[must_use]
189    pub fn estimated_memory_bytes(&self) -> u64 {
190        const PER_CONN_BYTES: u64 = 68 * 1024;
191        self.max_size as u64 * PER_CONN_BYTES
192    }
193}

195/// 连接池统计信息
196#[derive(Debug, Clone, Default)]
197pub struct PoolStats {
198    /// 当前空闲连接数
199    pub idle_count: usize,
200    /// 当前在用连接数
201    pub in_use_count: usize,
202    /// 累计创建连接数
203    pub total_created: u64,
204    /// 累计关闭连接数
205    pub total_closed: u64,
206    /// 累计获取连接次数
207    pub total_acquired: u64,
208    /// 累计归还连接次数
209    pub total_released: u64,
210    /// 累计获取超时次数
211    pub acquire_timeouts: u64,
212    /// 累计健康检查失败次数
213    pub health_check_failures: u64,
214}
215
216impl PoolStats {
217    /// 创建新的统计信息
218    #[must_use]
219    pub fn new() -> Self {
220        Self::default()
221    }
222
223    /// 当前总连接数（空闲 + 在用）
224    #[must_use]
225    pub fn total_connections(&self) -> usize {
226        self.idle_count + self.in_use_count
227    }
228
229    /// 连接利用率（在用 / 总数）
230    #[must_use]
231    pub fn utilization(&self) -> f64 {
232        let total = self.total_connections();
233        if total == 0 {
234            0.0
235        } else {
236            self.in_use_count as f64 / total as f64
237        }
238    }
239
240    /// 获取超时率
241    #[must_use]
242    pub fn timeout_rate(&self) -> f64 {
243        if self.total_acquired == 0 {
244            0.0
245        } else {
246            self.acquire_timeouts as f64 / self.total_acquired as f64
247        }
248    }
249
250    /// 健康检查失败率
251    #[must_use]
252    pub fn health_check_failure_rate(&self) -> f64 {
253        let total_checks = self.health_check_failures + self.total_acquired;
254        if total_checks == 0 {
255            0.0
256        } else {
257            self.health_check_failures as f64 / total_checks as f64
258        }
259    }
260
261    /// 合并两个统计信息（用于多池聚合）
262    #[must_use]
263    pub fn merge(&self, other: &PoolStats) -> PoolStats {
264        PoolStats {
265            idle_count: self.idle_count + other.idle_count,
266            in_use_count: self.in_use_count + other.in_use_count,
267            total_created: self.total_created + other.total_created,
268            total_closed: self.total_closed + other.total_closed,
269            total_acquired: self.total_acquired + other.total_acquired,
270            total_released: self.total_released + other.total_released,
271            acquire_timeouts: self.acquire_timeouts + other.acquire_timeouts,
272            health_check_failures: self.health_check_failures + other.health_check_failures,
273        }
274    }
275
276    /// 记录一次获取连接
277    pub fn record_acquire(&mut self) {
278        self.total_acquired += 1;
279        self.in_use_count += 1;
280        if self.idle_count > 0 {
281            self.idle_count -= 1;
282        }
283    }
284
285    /// 记录一次归还连接
286    pub fn record_release(&mut self) {
287        self.total_released += 1;
288        self.idle_count += 1;
289        if self.in_use_count > 0 {
290            self.in_use_count -= 1;
291        }
292    }
293
294    /// 记录一次获取超时
295    pub fn record_timeout(&mut self) {
296        self.acquire_timeouts += 1;
297    }
298
299    /// 记录一次健康检查失败
300    pub fn record_health_check_failure(&mut self) {
301        self.health_check_failures += 1;
302    }
303}
304
305impl fmt::Display for PoolStats {
306    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
307        write!(
308            f,
309            "PoolStats(idle={}, in_use={}, total={}, acquired={}, timeouts={})",
310            self.idle_count,
311            self.in_use_count,
312            self.total_connections(),
313            self.total_acquired,
314            self.acquire_timeouts
315        )
316    }
317}
