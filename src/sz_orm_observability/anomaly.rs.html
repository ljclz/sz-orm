1//! 异常检测模块（Anomaly Detection）
2//!
3//! 对应 v4.6.0 REQ-V46-004，tasks.md M6。
4//!
5//! # 核心概念
6//!
7//! - **AnomalyDetector**：异常检测器，复用 `MetricsRegistry` 指标数据
8//! - **AnomalyAlgorithm**：五种检测算法（Threshold/Trend/Statistical/ZScore/IQR）
9//! - **AnomalyAlert**：异常告警联动（Log/Webhook/Slo 三种通道）
10//! - **异常标注**：在查询日志上标注异常标记
11//!
12//! # 使用示例
13//!
14//! ```
15//! use sz_orm_observability::anomaly::{AnomalyDetector, AnomalyConfig, AnomalyAlgorithm};
16//! use sz_orm_observability::MetricsRegistry;
17//! use std::sync::Arc;
18//!
19//! let registry = Arc::new(MetricsRegistry::new());
20//! let config = AnomalyConfig::new().with_algorithm(AnomalyAlgorithm::Threshold).with_threshold(1.0);
21//! let detector = AnomalyDetector::new(Arc::clone(&registry), config);
22//! detector.record("query_duration", 0.5);
23//! detector.record("query_duration", 1.5);
24//! ```

38/// 异常检测算法
39#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
40pub enum AnomalyAlgorithm {
41    /// 阈值检测：指标值超过阈值即异常
42    Threshold,
43    /// 趋势检测：指标持续上升/下降即异常
44    Trend,
45    /// 统计检测：基于均值/方差检测异常
46    Statistical,
47    /// Z-Score 检测：Z-Score 超过阈值即异常
48    ZScore,
49    /// IQR 检测：四分位距检测异常
50    IQR,
51}

171// ============================================================================
172// AnomalyError — 异常检测错误
173// ============================================================================
174
175/// 异常检测错误
176#[derive(Debug, Clone, PartialEq, Eq)]
177pub enum AnomalyError {
178    /// 历史数据不足
179    InsufficientData,
180    /// 算法计算异常（除零/溢出）
181    CalculationError(String),
182    /// 告警通道不可用
183    AlertChannelUnavailable(String),
184}
185
186impl std::fmt::Display for AnomalyError {
187    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
188        match self {
189            AnomalyError::InsufficientData => write!(f, "insufficient history data"),
190            AnomalyError::CalculationError(msg) => write!(f, "calculation error: {}", msg),
191            AnomalyError::AlertChannelUnavailable(msg) => {
192                write!(f, "alert channel unavailable: {}", msg)
193            }
194        }
195    }
196}
197
198impl std::error::Error for AnomalyError {}

204/// 异常告警
205#[derive(Debug, Clone)]
206pub struct AnomalyAlert {
207    /// 异常信息
208    pub anomaly: Anomaly,
209    /// 告警通道
210    pub channel: AlertChannel,
211}
212
213impl AnomalyAlert {
214    /// 发送告警
215    pub async fn send(&self) -> Result<(), AnomalyError> {
216        match self.channel {
217            AlertChannel::Log => {
218                tracing::info!(
219                    "anomaly detected: metric={}, value={}, threshold={}, algorithm={:?}",
220                    self.anomaly.metric_name,
221                    self.anomaly.anomaly_value,
222                    self.anomaly.threshold,
223                    self.anomaly.algorithm
224                );
225                Ok(())
226            }
227            AlertChannel::Webhook => {
228                tracing::warn!(
229                    "webhook alert not implemented in offline mode, anomaly: {}",
230                    self.anomaly.description
231                );
232                Ok(())
233            }
234            AlertChannel::Slo => {
235                tracing::info!(
236                    "SLO anomaly alert: metric={}, value={}",
237                    self.anomaly.metric_name,
238                    self.anomaly.anomaly_value
239                );
240                Ok(())
241            }
242        }
243    }
244}

247// AnomalyDetector — 异常检测器

250/// 异常检测器
251///
252/// 复用 `MetricsRegistry` 指标注册中心，维护指标历史数据缓冲区，
253/// 按 `AnomalyAlgorithm` 检测指标异常，触发 `AnomalyAlert` 告警。
254pub struct AnomalyDetector {
255    /// 指标注册中心
256    metrics: Arc<MetricsRegistry>,
257    /// 检测配置
258    config: AnomalyConfig,
259    /// 历史数据缓冲区（metric_name → 历史值队列）
260    history: RwLock<HashMap<String, VecDeque<f64>>>,
261}
262
263impl AnomalyDetector {
264    /// 创建异常检测器
265    pub fn new(metrics: Arc<MetricsRegistry>, config: AnomalyConfig) -> Self {
266        Self {
267            metrics,
268            config,
269            history: RwLock::new(HashMap::new()),
270        }
271    }
272
273    /// 获取当前时间（Unix 毫秒）
274    fn now_ms() -> u64 {
275        SystemTime::now()
276            .duration_since(UNIX_EPOCH)
277            .unwrap_or_default()
278            .as_millis() as u64
279    }
280
281    /// 记录指标值到历史缓冲区，同时更新 MetricsRegistry 中的 Gauge
282    pub fn record(&self, metric_name: &str, value: f64) {
283        let gauge = self
284            .metrics
285            .register_gauge(metric_name, "anomaly detection metric");
286        gauge.set(value);
287        let mut history = self.history.write().unwrap();
288        let queue = history.entry(metric_name.to_string()).or_default();
289        queue.push_back(value);
290        while queue.len() > self.config.max_history {
291            queue.pop_front();
292        }
293    }
294
295    /// 检测指标异常
296    pub async fn detect(&self, metric_name: &str) -> Result<Vec<Anomaly>, AnomalyError> {
297        let data: Vec<f64> = {
298            let history = self.history.read().unwrap();
299            match history.get(metric_name) {
300                Some(q) => q.iter().copied().collect(),
301                None => return Err(AnomalyError::InsufficientData),
302            }
303        };
304        if data.len() < 2 {
305            return Err(AnomalyError::InsufficientData);
306        }
307        let anomalies = match self.config.algorithm {
308            AnomalyAlgorithm::Threshold => self.detect_threshold(metric_name, &data)?,
309            AnomalyAlgorithm::Trend => self.detect_trend(metric_name, &data)?,
310            AnomalyAlgorithm::Statistical => self.detect_statistical(metric_name, &data)?,
311            AnomalyAlgorithm::ZScore => self.detect_zscore(metric_name, &data)?,
312            AnomalyAlgorithm::IQR => self.detect_iqr(metric_name, &data)?,
313        };
314        for anomaly in &anomalies {
315            let _ = self.trigger_alert(anomaly).await;
316        }
317        Ok(anomalies)
318    }
319
320    /// 阈值检测
321    fn detect_threshold(
322        &self,
323        metric_name: &str,
324        data: &[f64],
325    ) -> Result<Vec<Anomaly>, AnomalyError> {
326        let now = Self::now_ms();
327        let mut anomalies = Vec::new();
328        for &value in data {
329            if value > self.config.threshold {
330                anomalies.push(Anomaly {
331                    metric_name: metric_name.to_string(),
332                    anomaly_value: value,
333                    threshold: self.config.threshold,
334                    window_ms: self.config.window_ms,
335                    algorithm: AnomalyAlgorithm::Threshold,
336                    detected_at: now,
337                    description: format!(
338                        "{} {} exceeds threshold {}",
339                        metric_name, value, self.config.threshold
340                    ),
341                });
342            }
343        }
344        Ok(anomalies)
345    }
346
347    /// 趋势检测：持续上升或下降
348    fn detect_trend(&self, metric_name: &str, data: &[f64]) -> Result<Vec<Anomaly>, AnomalyError> {
349        if data.len() < 3 {
350            return Err(AnomalyError::InsufficientData);
351        }
352        let now = Self::now_ms();
353        let mut anomalies = Vec::new();
354        let mut increasing = 0;
355        let mut decreasing = 0;
356        for i in 1..data.len() {
357            if data[i] > data[i - 1] {
358                increasing += 1;
359                decreasing = 0;
360            } else if data[i] < data[i - 1] {
361                decreasing += 1;
362                increasing = 0;
363            }
364        }
365        let trend_count = data.len() / 2;
366        if increasing >= trend_count {
367            anomalies.push(Anomaly {
368                metric_name: metric_name.to_string(),
369                anomaly_value: data[data.len() - 1],
370                threshold: trend_count as f64,
371                window_ms: self.config.window_ms,
372                algorithm: AnomalyAlgorithm::Trend,
373                detected_at: now,
374                description: format!(
375                    "{} sustained increasing trend over {} samples",
376                    metric_name, increasing
377                ),
378            });
379        }
380        if decreasing >= trend_count {
381            anomalies.push(Anomaly {
382                metric_name: metric_name.to_string(),
383                anomaly_value: data[data.len() - 1],
384                threshold: trend_count as f64,
385                window_ms: self.config.window_ms,
386                algorithm: AnomalyAlgorithm::Trend,
387                detected_at: now,
388                description: format!(
389                    "{} sustained decreasing trend over {} samples",
390                    metric_name, decreasing
391                ),
392            });
393        }
394        Ok(anomalies)
395    }
396
397    /// 统计检测：基于均值/方差
398    fn detect_statistical(
399        &self,
400        metric_name: &str,
401        data: &[f64],
402    ) -> Result<Vec<Anomaly>, AnomalyError> {
403        let now = Self::now_ms();
404        let mean = data.iter().sum::<f64>() / data.len() as f64;
405        let variance = data.iter().map(|x| (x - mean).powi(2)).sum::<f64>() / data.len() as f64;
406        let std_dev = variance.sqrt();
407        if std_dev == 0.0 {
408            return Ok(Vec::new());
409        }
410        let mut anomalies = Vec::new();
411        for &value in data {
412            let deviation = (value - mean).abs() / std_dev;
413            if deviation > self.config.threshold {
414                anomalies.push(Anomaly {
415                    metric_name: metric_name.to_string(),
416                    anomaly_value: value,
417                    threshold: self.config.threshold,
418                    window_ms: self.config.window_ms,
419                    algorithm: AnomalyAlgorithm::Statistical,
420                    detected_at: now,
421                    description: format!(
422                        "{} value {} deviates {} std devs from mean {}",
423                        metric_name, value, deviation, mean
424                    ),
425                });
426            }
427        }
428        Ok(anomalies)
429    }

431    /// Z-Score 检测
432    fn detect_zscore(&self, metric_name: &str, data: &[f64]) -> Result<Vec<Anomaly>, AnomalyError> {
433        let now = Self::now_ms();
434        let mean = data.iter().sum::<f64>() / data.len() as f64;
435        let variance = data.iter().map(|x| (x - mean).powi(2)).sum::<f64>() / data.len() as f64;
436        let std_dev = variance.sqrt();
437        if std_dev == 0.0 {
438            return Ok(Vec::new());
439        }
440        let mut anomalies = Vec::new();
441        for &value in data {
442            let zscore = (value - mean) / std_dev;
443            if zscore.abs() > self.config.zscore_threshold {
444                anomalies.push(Anomaly {
445                    metric_name: metric_name.to_string(),
446                    anomaly_value: value,
447                    threshold: self.config.zscore_threshold,
448                    window_ms: self.config.window_ms,
449                    algorithm: AnomalyAlgorithm::ZScore,
450                    detected_at: now,
451                    description: format!(
452                        "{} Z-Score {} indicates anomaly (threshold {})",
453                        metric_name, zscore, self.config.zscore_threshold
454                    ),
455                });
456            }
457        }
458        Ok(anomalies)
459    }
460
461    /// IQR 检测
462    fn detect_iqr(&self, metric_name: &str, data: &[f64]) -> Result<Vec<Anomaly>, AnomalyError> {
463        let now = Self::now_ms();
464        let mut sorted = data.to_vec();
465        sorted.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
466        let n = sorted.len();
467        let q1 = sorted[n / 4];
468        let q3 = sorted[3 * n / 4];
469        let iqr = q3 - q1;
470        let lower_bound = q1 - 1.5 * iqr;
471        let upper_bound = q3 + 1.5 * iqr;
472        let mut anomalies = Vec::new();
473        for &value in data {
474            if value < lower_bound || value > upper_bound {
475                anomalies.push(Anomaly {
476                    metric_name: metric_name.to_string(),
477                    anomaly_value: value,
478                    threshold: upper_bound,
479                    window_ms: self.config.window_ms,
480                    algorithm: AnomalyAlgorithm::IQR,
481                    detected_at: now,
482                    description: format!(
483                        "{} value {} outside IQR bounds [{}, {}]",
484                        metric_name, value, lower_bound, upper_bound
485                    ),
486                });
487            }
488        }
489        Ok(anomalies)
490    }

492    /// 触发告警
493    async fn trigger_alert(&self, anomaly: &Anomaly) -> Result<(), AnomalyError> {
494        let alert = AnomalyAlert {
495            anomaly: anomaly.clone(),
496            channel: self.config.alert_channel.clone(),
497        };
498        alert.send().await
499    }

501    /// 获取指标历史数据
502    pub fn history(&self, metric_name: &str) -> Vec<f64> {
503        let history = self.history.read().unwrap();
504        history
505            .get(metric_name)
506            .map(|q| q.iter().copied().collect())
507            .unwrap_or_default()
508    }
