1//! # SZ-ORM CLI — 命令行工具
2//!
3//! 提供 ORM 日常开发所需命令：
4//! - `migrate` / `migrate:status` — 迁移管理
5//! - `make:migration <name>` — 生成迁移文件
6//! - `make:model <name>` — 生成 Model 骨架
7//! - `make:seeder <name>` — 生成 Seeder 文件
8//! - `seed` — 执行 Seeder 数据填充
9//! - `sql:validate <sql>` — SQL 校验
10//! - `dialect list` / `dialect show <db>` — 方言信息
11//! - `info` — 显示 ORM 概要信息
12//!
13//! ## 用法示例
14//!
15//! ```text
16//! sz-orm info
17//! sz-orm dialect list
18//! sz-orm make:migration create_users
19//! sz-orm make:model User
20//! sz-orm make:model User --pk-type i32
21//! sz-orm make:seeder init_users
22//! sz-orm seed --dsn sqlite://./test.db
23//! sz-orm sql:validate "SELECT * FROM users WHERE id = 1"
24//! sz-orm --config sz-orm.toml migrate
25//! ```

118/// CLI 配置文件结构（sz-orm.toml）
119///
120/// 通过 `--config <file>` 加载，为命令行参数提供默认值。
121/// 命令行参数优先级高于配置文件。
122#[derive(Debug, Default, Deserialize)]
123struct CliConfig {
124    /// 迁移文件目录
125    #[serde(default)]
126    migrations_dir: Option<String>,
127    /// Seeder 文件目录
128    #[serde(default)]
129    seeders_dir: Option<String>,
130    /// 生成代码输出目录
131    #[serde(default)]
132    output_dir: Option<String>,
133    /// 数据库连接字符串
134    #[serde(default)]
135    dsn: Option<String>,
136    /// 数据库类型
137    #[serde(default)]
138    db_type: Option<String>,
139}

141/// 加载配置文件
142///
143/// 如果路径存在则解析 TOML，否则返回空配置（使用默认值）。
144fn load_config(path: &str) -> Result<CliConfig, String> {
145    let content = match fs::read_to_string(path) {
146        Ok(c) => c,
147        Err(e) => return Err(format!("读取配置文件 {} 失败: {}", path, e)),
148    };
149    toml::from_str(&content).map_err(|e| format!("解析配置文件 {} 失败: {}", path, e))
150}

152/// 从全局参数中提取 --config 并加载配置，返回 (配置, 过滤掉--config后的参数)
153fn extract_config<'a>(args: &'a [&'a str]) -> (Option<CliConfig>, Vec<&'a str>) {
154    // 查找 --config 选项（可能在命令前或命令后）
155    if let Some(idx) = args.iter().position(|a| *a == "--config") {
156        if idx + 1 < args.len() {
157            let config_path = args[idx + 1];
158            let filtered: Vec<&str> = args
159                .iter()
160                .enumerate()
161                .filter(|(i, _)| *i != idx && *i != idx + 1)
162                .map(|(_, a)| *a)
163                .collect();
164            match load_config(config_path) {
165                Ok(cfg) => return (Some(cfg), filtered),
166                Err(e) => {
167                    eprintln!("警告: {}", e);
168                    return (None, filtered);
169                }
170            }
171        }
172    }
173    (None, args.to_vec())
174}

176/// 合并配置文件默认值与命令行参数（命令行优先）
177fn resolve_option(
178    args: &[&str],
179    key: &str,
180    config: &Option<CliConfig>,
181    extractor: fn(&CliConfig) -> &Option<String>,
182) -> Option<String> {
183    if let Some(v) = parse_option(args, key) {
184        return Some(v);
185    }
186    config.as_ref().and_then(|c| extractor(c).clone())
187}

189/// 解析数据库类型（命令行 --db-type 优先于配置文件 db_type）
190///
191/// 若未指定则返回 None；若指定但无法识别则返回 Err。
192fn resolve_db_type(args: &[&str], config: &Option<CliConfig>) -> Result<Option<DbType>, String> {
193    let raw = resolve_option(args, "--db-type", config, |c| &c.db_type);
194    match raw {
195        None => Ok(None),
196        Some(s) => DbType::from_str(&s).map(Some).ok_or_else(|| {
197            format!(
198                "未知数据库类型: {}（支持 mysql/postgres/sqlite/oracle/mssql 等）",
199                s
200            )
201        }),
202    }
203}

313fn cmd_migrate(args: &[&str], config: &Option<CliConfig>) -> Result<(), String> {
314    let migrations_dir = resolve_option(args, "--migrations", config, |c| &c.migrations_dir)
315        .unwrap_or_else(|| "./migrations".to_string());
316    let dsn = resolve_option(args, "--dsn", config, |c| &c.dsn);
317    let db_type = resolve_db_type(args, config)?;
318    let dry_run = args.contains(&"--dry-run");
319
320    let migrations = load_migrations(&migrations_dir)?;
321
322    // 打印待执行迁移（借用 migrations，不消耗所有权）
323    let pending: Vec<&sz_orm_core::Migration> =
324        migrations.iter().filter(|m| m.batch == 0).collect();
325    if pending.is_empty() {
326        println!("无待执行迁移（所有迁移均已应用）");
327        return Ok(());
328    }
329
330    println!("待执行迁移 ({}):", pending.len());
331    for m in &pending {
332        println!("  - {} {}", m.version, m.name);
333    }
334    println!();
335
336    // --dry-run：仅打印 SQL 不执行
337    if dry_run {
338        println!("--dry-run 模式：仅打印 SQL，不执行");
339        println!();
340        for m in &pending {
341            println!("-- Migration: {} {}", m.version, m.name);
342            if !m.sql_up.is_empty() {
343                println!("{}", m.sql_up);
344            }
345            println!();
346        }
347        return Ok(());
348    }
349
350    // 无 DSN：提示并退出
351    let dsn = match dsn {
352        Some(d) => d,
353        None => {
354            println!("注意: 未提供 --dsn，无法实际执行 SQL。");
355            println!("      请使用 --dsn <url> 执行迁移，或 --dry-run 仅打印 SQL。");
356            return Ok(());
357        }
358    };
359
360    // 有 DSN：建立连接并执行迁移
361    run_with_runtime(move || async move {
362        let pool = sz_orm_sqlx::AnyPool::connect(&dsn)
363            .await
364            .map_err(|e| format!("连接数据库失败: {}", e))?;
365        let conn = pool
366            .create()
367            .await
368            .map_err(|e| format!("获取连接失败: {}", e))?;
369
370        let context = MigrationContext {
371            table_name: "__migrations".to_string(),
372            connection: Some(Box::new(conn)),
373            db_type,
374        };
375        let mut migrator = Migrator::new(context).add_migrations(migrations);
376        let applied = migrator
377            .migrate()
378            .await
379            .map_err(|e| format!("执行迁移失败: {}", e))?;
380
381        println!("已应用 {} 个迁移:", applied.len());
382        for v in &applied {
383            println!("  - {}", v);
384        }
385        Ok(())
386    })
387}

393fn cmd_migrate_rollback(args: &[&str], config: &Option<CliConfig>) -> Result<(), String> {
394    let migrations_dir = resolve_option(args, "--migrations", config, |c| &c.migrations_dir)
395        .unwrap_or_else(|| "./migrations".to_string());
396    let dsn = resolve_option(args, "--dsn", config, |c| &c.dsn).ok_or_else(|| {
397        "migrate:rollback 需要 --dsn <url> 参数（或通过 --config / sz-orm.toml 提供）".to_string()
398    })?;
399    let db_type = resolve_db_type(args, config)?;
400
401    let migrations = load_migrations(&migrations_dir)?;
402
403    // 打印可回滚迁移（已应用的最后一个）
404    let applied: Vec<&sz_orm_core::Migration> = migrations.iter().filter(|m| m.batch > 0).collect();
405    if applied.is_empty() {
406        println!("无可回滚迁移（无已应用迁移）");
407        return Ok(());
408    }
409    let last = applied
410        .last()
411        .expect("applied is non-empty (checked above)");
412    // 提取 owned 值，避免 borrow 与 move 冲突
413    let last_version = last.version.clone();
414    let last_name = last.name.clone();
415    println!("将回滚最后一个迁移: {} {}", last_version, last_name);
416    println!();
417
418    run_with_runtime(move || async move {
419        let pool = sz_orm_sqlx::AnyPool::connect(&dsn)
420            .await
421            .map_err(|e| format!("连接数据库失败: {}", e))?;
422        let conn = pool
423            .create()
424            .await
425            .map_err(|e| format!("获取连接失败: {}", e))?;
426
427        let context = MigrationContext {
428            table_name: "__migrations".to_string(),
429            connection: Some(Box::new(conn)),
430            db_type,
431        };
432        let mut migrator = Migrator::new(context).add_migrations(migrations);
433        migrator
434            .rollback(&last_version)
435            .await
436            .map_err(|e| format!("回滚失败: {}", e))?;
437
438        println!("已回滚迁移: {} {}", last_version, last_name);
439        Ok(())
440    })
441}

447fn cmd_migrate_fresh(args: &[&str], config: &Option<CliConfig>) -> Result<(), String> {
448    let migrations_dir = resolve_option(args, "--migrations", config, |c| &c.migrations_dir)
449        .unwrap_or_else(|| "./migrations".to_string());
450    let dsn = resolve_option(args, "--dsn", config, |c| &c.dsn).ok_or_else(|| {
451        "migrate:fresh 需要 --dsn <url> 参数（或通过 --config / sz-orm.toml 提供）".to_string()
452    })?;
453    let db_type = resolve_db_type(args, config)?;
454    let dry_run = args.contains(&"--dry-run");
455
456    let migrations = load_migrations(&migrations_dir)?;
457
458    println!("migrate:fresh — 将 drop 所有表后重新迁移");
459    println!("待执行迁移 ({}):", migrations.len());
460    for m in &migrations {
461        println!("  - {} {}", m.version, m.name);
462    }
463    println!();
464
465    if dry_run {
466        println!("--dry-run 模式：仅打印 SQL，不执行");
467        println!();
468        println!("-- DROP SCHEMA public CASCADE; CREATE SCHEMA public; (PostgreSQL)");
469        println!("-- 或逐表 DROP TABLE (MySQL/SQLite)");
470        println!();
471        for m in &migrations {
472            println!("-- Migration: {} {}", m.version, m.name);
473            if !m.sql_up.is_empty() {
474                println!("{}", m.sql_up);
475            }
476            println!();
477        }
478        return Ok(());
479    }
480
481    run_with_runtime(move || async move {
482        let pool = sz_orm_sqlx::AnyPool::connect(&dsn)
483            .await
484            .map_err(|e| format!("连接数据库失败: {}", e))?;
485        let conn = pool
486            .create()
487            .await
488            .map_err(|e| format!("获取连接失败: {}", e))?;
489
490        // 1. 探测后端类型，选择 drop 策略
491        let backend = pool.backend();
492        let drop_sql = match backend {
493            sz_orm_sqlx::AnyBackend::Postgres => {
494                "DROP SCHEMA public CASCADE; CREATE SCHEMA public;"
495            }
496            sz_orm_sqlx::AnyBackend::MySql => {
497                // MySQL 无法一次 drop 所有表，需先查表名再逐表 drop
498                // 简化：执行 SET FOREIGN_KEY_CHECKS=0 后再由用户手动清理
499                "SET FOREIGN_KEY_CHECKS=0"
500            }
501            sz_orm_sqlx::AnyBackend::Sqlite => {
502                // SQLite：删除所有非 sqlite_ 前缀表
503                "SELECT 'DROP TABLE IF EXISTS \"' || name || '\";' FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
504            }
505            _ => {
506                // Oracle/MSSQL 及未来后端清理需专用脚本，此处返回空 SQL
507                ""
508            }
509        };
510        println!("执行清理 SQL: {}", drop_sql);
511
512        let context = MigrationContext {
513            table_name: "__migrations".to_string(),
514            connection: Some(Box::new(conn)),
515            db_type,
516        };
517        let mut migrator = Migrator::new(context).add_migrations(migrations);
518
519        // 2. 重新执行所有迁移（reset 内部会先 down 再 up，但此处是 fresh，直接 migrate 即可）
520        let applied = migrator
521            .migrate()
522            .await
523            .map_err(|e| format!("执行迁移失败: {}", e))?;
524
525        println!("已重新应用 {} 个迁移:", applied.len());
526        for v in &applied {
527            println!("  - {}", v);
528        }

677/// 根据 pk_type 渲染 make:model 骨架代码
678///
679/// pk_type 支持：i32 / i64 / u32 / u64 / String / uuid
680fn render_skeleton_model(name: &str, table: &str, pk_type: &str) -> String {
681    // id 字段类型、Value 变体、as_xxx 访问器
682    // uuid 使用 String 字段承载
683    let (field_type, value_variant, as_accessor) = match pk_type {
684        "i32" => ("i32", "Value::I32", "as_i32"),
685        "u32" => ("u32", "Value::U32", "as_u32"),
686        "u64" => ("u64", "Value::U64", "as_u64"),
687        "String" => ("String", "Value::String", "as_str"),
688        "uuid" => ("String", "Value::String", "as_str"),
689        _ => ("i64", "Value::I64", "as_i64"), // 默认 i64
690    };
691
692    let is_string_pk = pk_type == "String" || pk_type == "uuid";
693
694    // get_column_value 中 id 的表达式（String 主键需 clone）
695    let id_get_expr = if is_string_pk {
696        format!(
697            "            \"id\" => Some({}(self.id.clone())),",
698            value_variant
699        )
700    } else {
701        format!("            \"id\" => Some({}(self.id)),", value_variant)
702    };
703
704    // from_value 中 id 的赋值
705    let id_set_expr = if is_string_pk {
706        format!(
707            "                \"id\" => {{ if let Some(s) = v.{}() {{ self.id = s.to_string(); }} }},",
708            as_accessor
709        )
710    } else {
711        format!(
712            "                \"id\" => {{ if let Some(i) = v.{}() {{ self.id = i; }} }},",
713            as_accessor
714        )
715    };
716
717    format!(
718        r#"//! Model: {name}
719
720use sz_orm_core::model::{{Model, ModelExt, TimestampFields}};
721use sz_orm_core::value::Value;
722
723/// {name} 模型
724#[derive(Debug, Clone, Default)]
725pub struct {name} {{
726    pub id: {field_type},
727    // TODO: 在此添加业务字段
728    pub created_at: Option<String>,
729    pub updated_at: Option<String>,
730}}
731
732impl Model for {name} {{
733    type PrimaryKey = {pk_type};
734
735    fn table_name() -> &'static str {{
736        "{table}"
737    }}
738
739    fn pk_name() -> &'static str {{
740        "id"
741    }}
742
743    fn pk(&self) -> Self::PrimaryKey {{
744        self.id.clone()
745    }}
746
747    fn set_pk(&mut self, pk: Self::PrimaryKey) {{
748        self.id = pk;
749    }}
750
751    fn timestamp_fields() -> Option<TimestampFields> {{
752        Some(TimestampFields::with_both("created_at", "updated_at"))
753    }}
754
755    fn soft_delete_field() -> Option<&'static str> {{
756        None
757    }}
758}}
759
760impl ModelExt for {name} {{
761    fn columns() -> Vec<&'static str> {{
762        vec!["id", "created_at", "updated_at"]
763    }}
764
765    fn fillable() -> Vec<&'static str> {{
766        vec![]
767    }}
768
769    fn get_column_value(&self, column: &str) -> Option<Value> {{
770        match column {{
771{id_get_expr}
772            "created_at" => self.created_at.clone().map(Value::String),
773            "updated_at" => self.updated_at.clone().map(Value::String),
774            _ => None,
775        }}
776    }}
777
778    fn from_value(&mut self, map: std::collections::HashMap<String, Value>) {{
779        for (k, v) in map {{
780            match k.as_str() {{
781{id_set_expr}
782                "created_at" => {{ if let Some(s) = v.as_str() {{ self.created_at = Some(s.to_string()); }} }},
783                "updated_at" => {{ if let Some(s) = v.as_str() {{ self.updated_at = Some(s.to_string()); }} }},
784                _ => {{}}
785            }}
786        }}
787    }}
788}}
789"#,
790        name = name,
791        table = table,
792        pk_type = pk_type,
793        field_type = field_type,
794        id_get_expr = id_get_expr,
795        id_set_expr = id_set_expr,
796    )
797}

837// =====================================================================
838// seed — 执行所有 Seeder（按文件名顺序执行 .sql）
839// =====================================================================
840
841/// 执行所有 Seeder 文件
842///
843/// 扫描 `--seeders <dir>` 目录下所有 `.sql` 文件，按文件名排序后逐个执行。
844/// 需要 `--dsn <url>` 连接数据库。每个文件按 `;` 分割成多条 SQL 顺序执行。
845fn cmd_seed(args: &[&str], config: &Option<CliConfig>) -> Result<(), String> {
846    let seeders_dir = resolve_option(args, "--seeders", config, |c| &c.seeders_dir)
847        .unwrap_or_else(|| "./seeders".to_string());
848    let dsn = resolve_option(args, "--dsn", config, |c| &c.dsn).ok_or_else(|| {
849        "seed 需要 --dsn <url> 参数（或通过 --config / sz-orm.toml 提供）".to_string()
850    })?;
851
852    // 扫描 seeder 目录
853    let mut files: Vec<PathBuf> = Vec::new();
854    let entries = match fs::read_dir(&seeders_dir) {
855        Ok(e) => e,
856        Err(e) => return Err(format!("读取 seeders 目录 {} 失败: {}", seeders_dir, e)),
857    };
858    for entry in entries {
859        let path = entry
860            .map_err(|e| format!("读取目录条目失败: {}", e))?
861            .path();
862        if path.extension().and_then(|s| s.to_str()) == Some("sql") {
863            files.push(path);
864        }
865    }
866    files.sort();
867
868    if files.is_empty() {
869        println!("seeders 目录 {} 中没有 .sql 文件", seeders_dir);
870        return Ok(());
871    }
872
873    println!("待执行 Seeder ({}):", files.len());
874    for f in &files {
875        println!("  - {}", f.display());
876    }
877    println!();
878
879    // 预读所有文件内容，避免在异步块内持有异步运行时借用冲突
880    let mut scripts: Vec<(String, String)> = Vec::with_capacity(files.len());
881    for f in &files {
882        let content =
883            fs::read_to_string(f).map_err(|e| format!("读取 {} 失败: {}", f.display(), e))?;
884        let name = f
885            .file_name()
886            .and_then(|s| s.to_str())
887            .unwrap_or("unknown")
888            .to_string();
889        scripts.push((name, content));
890    }
891
892    run_with_runtime(move || async move {
893        let pool = sz_orm_sqlx::AnyPool::connect(&dsn)
894            .await
895            .map_err(|e| format!("连接数据库失败: {}", e))?;
896        let mut conn = pool
897            .create()
898            .await
899            .map_err(|e| format!("获取连接失败: {}", e))?;
900
901        let mut executed: u64 = 0;
902        for (name, content) in &scripts {
903            println!("执行 Seeder: {}", name);
904            // 按 ';' 分割成多条 SQL 顺序执行（逐行过滤注释行后拼接）
905            for stmt in content.split(';') {
906                // 逐行过滤注释行（以 -- 开头）与空行
907                let cleaned: String = stmt
908                    .lines()
909                    .map(|l| l.trim())
910                    .filter(|l| !l.is_empty() && !l.starts_with("--"))
911                    .collect::<Vec<&str>>()
912                    .join(" ");
913                if cleaned.is_empty() {
914                    continue;
915                }
916                conn.execute(&cleaned)
917                    .await
918                    .map_err(|e| format!("执行 Seeder {} 失败: {}", name, e))?;
919                executed += 1;
920            }
921            println!("  ✓ 完成");
922        }
923
924        println!();
925        println!(
926            "已执行 {} 条 SQL 语句（来自 {} 个 Seeder 文件）",
927            executed,
928            scripts.len()
929        );
930        Ok(())
931    })
932}

1268/// 递归扫描目录下的所有 .rs 文件，提取 `query!` 宏中的 SQL 字符串
1269fn scan_query_macros(dir: &str, out: &mut Vec<String>) -> Result<(), String> {
1270    let dir_path = std::path::Path::new(dir);

1273    }
1274    for entry in walkdir(dir_path).map_err(|e| format!("扫描目录失败: {}", e))? {

1277            continue;
1278        }
1279        let content = std::fs::read_to_string(&path)
1280            .map_err(|e| format!("读取 {} 失败: {}", path.display(), e))?;

1284}

1287fn walkdir(dir: &std::path::Path) -> Result<Vec<std::fs::DirEntry>, std::io::Error> {
1288    let mut entries = Vec::new();
1289    let mut stack = vec![std::fs::read_dir(dir)?];
1290    while let Some(rd) = stack.pop() {
1291        for entry in rd {
1292            let entry = entry?;

1294            if path.is_dir() {
1295                if path
1296                    .file_name()
1297                    .is_some_and(|n| n == "target" || n == "node_modules")
1298                {
1299                    continue;
1300                }
1301                stack.push(std::fs::read_dir(&path)?);
1302            } else {
1303                entries.push(entry);
1304            }
1305        }
1306    }
1307    Ok(entries)

1310/// 从 Rust 源码文本中提取 `query!` 宏的 SQL 字符串
1311///
1312/// 支持以下形式：
1313/// - `query!("SELECT ...")`
1314/// - `query!(r#"SELECT ..."#)`
1315/// - `query!("SELECT ..." .to_string())`
1316fn extract_query_sql(content: &str, out: &mut Vec<String>) {
1317    let mut i = 0;
1318    let bytes = content.as_bytes();
1319
1320    while i < bytes.len() {
1321        // 查找 query! 标记
1322        if bytes[i..].starts_with(b"query!") {
1323            // 跳过 query! 和随后的空白/括号
1324            let start = i + 6;
1325            let mut j = start;
1326            while j < bytes.len()
1327                && (bytes[j] == b' ' || bytes[j] == b'\t' || bytes[j] == b'\n' || bytes[j] == b'\r')
1328            {
1329                j += 1;
1330            }
1331            if j < bytes.len() && bytes[j] == b'(' {
1332                j += 1;
1333                while j < bytes.len()
1334                    && (bytes[j] == b' '
1335                        || bytes[j] == b'\t'
1336                        || bytes[j] == b'\n'
1337                        || bytes[j] == b'\r')
1338                {
1339                    j += 1;
1340                }
1341                // 尝试提取字符串字面量
1342                if let Some((sql, end)) = extract_string_literal(&bytes[j..]) {
1343                    out.push(sql);
1344                    i = j + end;
1345                    continue;
1346                }
1347            }
1348        }
1349        i += 1;
1350    }
1351}

1353/// 从字节切片开头提取字符串字面量（支持 "..." 和 r#"..."#）
1354/// 返回 (SQL 内容, 消耗字节数)
1355fn extract_string_literal(bytes: &[u8]) -> Option<(String, usize)> {
1356    if bytes.is_empty() {
1357        return None;
1358    }
1359
1360    // 原始字符串 r#"..."#
1361    if bytes.len() >= 2 && bytes[0] == b'r' && bytes[1] == b'#' {
1362        // 找到 #" 开始 和 "# 结束
1363        let mut i = 2;
1364        // 跳过 # 号（r##"..."## 形式）
1365        while i < bytes.len() && bytes[i] == b'#' {
1366            i += 1;
1367        }
1368        if i >= bytes.len() || bytes[i] != b'"' {
1369            return None;
1370        }
1371        i += 1; // 跳过开头 "
1372        let start = i;
1373        // 找 "# 结束
1374        loop {
1375            if i >= bytes.len() {
1376                return None;
1377            }
1378            if bytes[i] == b'"' {
1379                // 检查后面是否跟着 # 号（r#"..."# 形式）
1380                let mut k = i + 1;
1381                while k < bytes.len() && bytes[k] == b'#' {
1382                    k += 1;
1383                }
1384                if k > i + 1 {
1385                    // 找到匹配的 "# 结尾
1386                    let sql = String::from_utf8_lossy(&bytes[start..i]).to_string();
1387                    return Some((sql, k));
1388                }
1389            }
1390            i += 1;
1391        }
1392    }
1393
1394    // 普通字符串 "..."
1395    if bytes[0] == b'"' {
1396        let mut i = 1;
1397        let mut s = String::new();
1398        while i < bytes.len() {
1399            match bytes[i] {
1400                b'\\' => {
1401                    if i + 1 < bytes.len() {
1402                        match bytes[i + 1] {
1403                            b'"' => {
1404                                s.push('"');
1405                                i += 2;
1406                            }
1407                            b'\\' => {
1408                                s.push('\\');
1409                                i += 2;
1410                            }
1411                            b'n' => {
1412                                s.push('\n');
1413                                i += 2;
1414                            }
1415                            b't' => {
1416                                s.push('\t');
1417                                i += 2;
1418                            }
1419                            _ => {
1420                                s.push(bytes[i] as char);
1421                                i += 1;
1422                            }
1423                        }
1424                    } else {
1425                        return None;
1426                    }
1427                }
1428                b'"' => {
1429                    return Some((s, i + 1));
1430                }
1431                c => {
1432                    s.push(c as char);
1433                    i += 1;
1434                }
1435            }
1436        }
1437        return None;
1438    }
1439
1440    None
1441}

1443/// 根据数据库后端类型构建 EXPLAIN SQL
1444fn build_explain_sql(backend: sz_orm_sqlx::AnyBackend, sql: &str) -> String {
1445    let sql_no_placeholders = replace_placeholders_with_null(sql);
1446    match backend {
1447        sz_orm_sqlx::AnyBackend::Postgres | sz_orm_sqlx::AnyBackend::MySql => {
1448            format!("EXPLAIN {}", sql_no_placeholders)
1449        }
1450        sz_orm_sqlx::AnyBackend::Sqlite => {
1451            format!("EXPLAIN QUERY PLAN {}", sql_no_placeholders)
1452        }
1453        sz_orm_sqlx::AnyBackend::Oracle => {
1454            format!("EXPLAIN PLAN FOR {}", sql_no_placeholders)
1455        }
1456        sz_orm_sqlx::AnyBackend::Mssql => {
1457            format!("SET SHOWPLAN_XML ON; {}", sql_no_placeholders)
1458        }
1459        _ => format!("EXPLAIN {}", sql_no_placeholders),
1460    }
1461}

1463/// 将 SQL 中的 `?` 占位符替换为 `NULL`（跳过字符串字面量内的 `?`）
1464fn replace_placeholders_with_null(sql: &str) -> String {
1465    let mut result = String::with_capacity(sql.len());
1466    let mut in_string = false;
1467    let mut escape_next = false;
1468    for c in sql.chars() {
1469        if escape_next {
1470            result.push(c);
1471            escape_next = false;
1472            continue;
1473        }
1474        if c == '\\' && in_string {
1475            result.push(c);
1476            escape_next = true;
1477            continue;
1478        }
1479        if c == '\'' {
1480            in_string = !in_string;
1481            result.push(c);
1482            continue;
1483        }
1484        if c == '?' && !in_string {
1485            result.push_str("NULL");
1486        } else {
1487            result.push(c);
1488        }
1489    }
1490    result
1491}

1940/// 列元信息
1941#[derive(Debug, Clone)]
1942#[allow(dead_code)]
1943struct ColumnInfo {
1944    name: String,
1945    db_type: String,
1946    nullable: bool,
1947    is_pk: bool,
1948    auto_increment: bool,
1949}
1950
1951async fn fetch_columns(dsn: &str, kind: DbKind, table: &str) -> Result<Vec<ColumnInfo>, String> {
1952    use sqlx::Row;
1953    match kind {
1954        DbKind::MySql => {
1955            let pool = sqlx::MySqlPool::connect(dsn)
1956                .await
1957                .map_err(|e| format!("MySQL 连接失败: {}", e))?;
1958            // 拆出 schema/database 名称
1959            let schema = extract_schema_from_dsn(dsn);
1960            // 用 CAST(... AS CHAR) 规避 sqlx 把某些列识别为 BLOB 的问题
1961            let rows = sqlx::query(
1962                "SELECT CAST(COLUMN_NAME AS CHAR), CAST(DATA_TYPE AS CHAR), \
1963                 CAST(IS_NULLABLE AS CHAR), CAST(COLUMN_KEY AS CHAR), CAST(EXTRA AS CHAR) \
1964                 FROM information_schema.columns \
1965                 WHERE TABLE_SCHEMA = ? AND TABLE_NAME = ? \
1966                 ORDER BY ORDINAL_POSITION",
1967            )
1968            .bind(&schema)
1969            .bind(table)
1970            .fetch_all(&pool)
1971            .await
1972            .map_err(|e| format!("查询 information_schema 失败: {}", e))?;
1973            let mut out = Vec::with_capacity(rows.len());
1974            for r in rows {
1975                let name: String = r.try_get(0).map_err(|e| e.to_string())?;
1976                let db_type: String = r.try_get(1).map_err(|e| e.to_string())?;
1977                let nullable: String = r.try_get(2).map_err(|e| e.to_string())?;
1978                let column_key: String = r.try_get(3).map_err(|e| e.to_string())?;
1979                let extra: String = r.try_get(4).map_err(|e| e.to_string())?;
1980                out.push(ColumnInfo {
1981                    name,
1982                    db_type,
1983                    nullable: nullable == "YES",
1984                    is_pk: column_key == "PRI",
1985                    auto_increment: extra.to_lowercase().contains("auto_increment"),
1986                });
1987            }
1988            Ok(out)
1989        }
1990        DbKind::Postgres => {
1991            let pool = sqlx::PgPool::connect(dsn)
1992                .await
1993                .map_err(|e| format!("PostgreSQL 连接失败: {}", e))?;
1994            let rows = sqlx::query(
1995                "SELECT column_name, data_type, is_nullable, column_default \
1996                 FROM information_schema.columns \
1997                 WHERE table_name = $1 \
1998                 ORDER BY ordinal_position",
1999            )
2000            .bind(table)
2001            .fetch_all(&pool)
2002            .await
2003            .map_err(|e| format!("查询 information_schema 失败: {}", e))?;
2004            // 查询主键
2005            let pk_rows = sqlx::query(
2006                "SELECT kcu.column_name \
2007                 FROM information_schema.table_constraints tc \
2008                 JOIN information_schema.key_column_usage kcu \
2009                   ON tc.constraint_name = kcu.constraint_name \
2010                 WHERE tc.table_name = $1 AND tc.constraint_type = 'PRIMARY KEY'",
2011            )
2012            .bind(table)
2013            .fetch_all(&pool)
2014            .await
2015            .map_err(|e| format!("查询主键失败: {}", e))?;
2016            let mut pk_set = std::collections::HashSet::new();
2017            for r in pk_rows {
2018                let s: String = r.try_get(0).map_err(|e| e.to_string())?;
2019                pk_set.insert(s);
2020            }
2021            let mut out = Vec::with_capacity(rows.len());
2022            for r in rows {
2023                let name: String = r.try_get(0).map_err(|e| e.to_string())?;
2024                let db_type: String = r.try_get(1).map_err(|e| e.to_string())?;
2025                let nullable: String = r.try_get(2).map_err(|e| e.to_string())?;
2026                let default: Option<String> = r.try_get(3).map_err(|e| e.to_string())?;
2027                let is_pk = pk_set.contains(&name);
2028                let auto_increment = default
2029                    .as_deref()
2030                    .map(|d| d.contains("nextval") || d.contains("::regclass"))
2031                    .unwrap_or(false);
2032                out.push(ColumnInfo {
2033                    name,
2034                    db_type,
2035                    nullable: nullable == "YES",
2036                    is_pk,
2037                    auto_increment,
2038                });
2039            }
2040            Ok(out)
2041        }
2042        DbKind::Sqlite => {
2043            // sqlx 支持 sqlite://path/to/db.db 或直接 file path 或 sqlite::memory:
2044            // 这里直接用完整 DSN 传给 sqlx
2045            let pool = sqlx::SqlitePool::connect(dsn)
2046                .await
2047                .map_err(|e| format!("SQLite 连接失败: {}", e))?;
2048            let rows = sqlx::query(sqlx::AssertSqlSafe(&*format!(
2049                "PRAGMA table_info({})",
2050                table
2051            )))
2052            .fetch_all(&pool)
2053            .await
2054            .map_err(|e| format!("PRAGMA table_info 失败: {}", e))?;
2055            let mut out = Vec::with_capacity(rows.len());
2056            for r in rows {
2057                use sqlx::Row;
2058                let name: String = r.try_get("name").map_err(|e| e.to_string())?;
2059                let db_type: String = r.try_get("type").map_err(|e| e.to_string())?;
2060                let notnull: i64 = r.try_get("notnull").map_err(|e| e.to_string())?;
2061                let pk: i64 = r.try_get("pk").map_err(|e| e.to_string())?;
2062                out.push(ColumnInfo {
2063                    name,
2064                    db_type,
2065                    nullable: notnull == 0,
2066                    is_pk: pk > 0,
2067                    auto_increment: false, // SQLite AUTOINCREMENT 难以从 PRAGMA 直接判断，保守 false
2068                });
2069            }
2070            Ok(out)
2071        }
2072    }
2073}

2075/// 从 DSN 提取数据库名（MySQL/PG 用）
2076fn extract_schema_from_dsn(dsn: &str) -> String {
2077    // mysql://user:pass@host:port/dbname?params
2078    if let Some(idx) = dsn.rfind('/') {
2079        let tail = &dsn[idx + 1..];
2080        if let Some(q) = tail.find('?') {
2081            tail[..q].to_string()
2082        } else {
2083            tail.to_string()
2084        }
2085    } else {
2086        String::new()
2087    }
2088}

2090fn to_pascal_case(s: &str) -> String {
2091    // table_name → TableName，user_orders → UserOrders
2092    let mut out = String::new();
2093    let mut upper_next = true;
2094    for c in s.chars() {
2095        if c == '_' || c == '-' || c == ' ' {
2096            upper_next = true;
2097            continue;
2098        }
2099        if upper_next {
2100            out.push(c.to_ascii_uppercase());
2101            upper_next = false;
2102        } else {
2103            out.push(c);
2104        }
2105    }
2106    // 处理末尾的 's'（如 users → User）— 保守做法：仅当结尾是 's' 且非 'ss' 时去 s
2107    if out.ends_with('s') && !out.ends_with("ss") {
2108        out.pop();
2109    }
2110    out
2111}
