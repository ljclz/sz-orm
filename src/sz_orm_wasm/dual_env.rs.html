1//! # WASM 双环境适配（v6.9.0 REQ-BND-WASM）
2//!
3//! 支持 `wasm-pack build --target web`（浏览器）和 `--target nodejs`（Node.js）双目标构建。
4//! 提供运行时环境检测、构建配置生成、沙箱隔离验证。
5
6use serde::{Deserialize, Serialize};
7
8/// WASM 构建目标
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
10pub enum WasmBuildTarget {
11    /// 浏览器目标（`--target web`）
12    Web,
13    /// Node.js 目标（`--target nodejs`）
14    Nodejs,
15}
16
17impl WasmBuildTarget {
18    /// 返回 wasm-pack --target 参数值
19    pub fn target_arg(&self) -> &'static str {
20        match self {
21            Self::Web => "web",
22            Self::Nodejs => "nodejs",
23        }
24    }
25
26    /// 返回构建输出目录名
27    pub fn output_dir(&self) -> &'static str {
28        match self {
29            Self::Web => "pkg-web",
30            Self::Nodejs => "pkg-nodejs",
31        }
32    }
33}
34
35/// 运行时环境
36#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
37pub enum RuntimeEnv {
38    /// 浏览器环境
39    Browser,
40    /// Node.js 环境
41    Nodejs,
42    /// 未知环境（原生或其他）
43    Unknown,
44}
45
46/// 双环境构建配置
47#[derive(Debug, Clone, Serialize, Deserialize)]
48pub struct DualEnvConfig {
49    /// 浏览器构建目标配置
50    pub web: BuildTargetConfig,
51    /// Node.js 构建目标配置
52    pub nodejs: BuildTargetConfig,
53}
54
55impl Default for DualEnvConfig {
56    fn default() -> Self {
57        Self::new()
58    }
59}
60
61impl DualEnvConfig {
62    /// 创建默认双环境配置
63    pub fn new() -> Self {
64        Self {
65            web: BuildTargetConfig::for_target(WasmBuildTarget::Web),
66            nodejs: BuildTargetConfig::for_target(WasmBuildTarget::Nodejs),
67        }
68    }
69
70    /// 获取指定目标的配置
71    pub fn get(&self, target: WasmBuildTarget) -> &BuildTargetConfig {
72        match target {
73            WasmBuildTarget::Web => &self.web,
74            WasmBuildTarget::Nodejs => &self.nodejs,
75        }
76    }
77
78    /// 生成 wasm-pack 构建命令
79    pub fn build_command(&self, target: WasmBuildTarget) -> String {
80        let cfg = self.get(target);
81        format!(
82            "wasm-pack build --target {} --out-dir {} {}",
83            target.target_arg(),
84            cfg.output_dir,
85            if cfg.release { "--release" } else { "" }
86        )
87    }
88}

90/// 单目标构建配置
91#[derive(Debug, Clone, Serialize, Deserialize)]
92pub struct BuildTargetConfig {
93    /// 构建目标
94    pub target: WasmBuildTarget,
95    /// 输出目录
96    pub output_dir: String,
97    /// 是否 release 构建
98    pub release: bool,
99    /// 启用的 feature 列表
100    pub features: Vec<String>,
101}
102
103impl BuildTargetConfig {
104    /// 为指定目标创建默认配置
105    pub fn for_target(target: WasmBuildTarget) -> Self {
106        let features = match target {
107            WasmBuildTarget::Web => vec!["js".to_string()],
108            WasmBuildTarget::Nodejs => vec!["js".to_string()],
109        };
110        Self {
111            target,
112            output_dir: target.output_dir().to_string(),
113            release: true,
114            features,
115        }
116    }
117
118    /// 生成 feature 参数字符串
119    pub fn feature_args(&self) -> String {
120        if self.features.is_empty() {
121            String::new()
122        } else {
123            format!("--features {}", self.features.join(","))
124        }
125    }
126}

137/// 沙箱隔离验证结果
138#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
139pub struct SandboxVerificationResult {
140    /// 被测试的路径
141    pub path: String,
142    /// 是否被沙箱拒绝
143    pub blocked: bool,
144    /// 拒绝原因
145    pub reason: String,
146}
147
148/// 验证沙箱是否阻止敏感路径访问
149///
150/// 检查 `/etc/passwd`、`/etc/shadow`、`~/.ssh` 等敏感路径
151/// 是否被 SandboxedFs 正确拒绝。
152pub fn verify_sandbox_isolation(sandbox: &crate::SandboxedFs) -> Vec<SandboxVerificationResult> {
153    let sensitive_paths = [
154        "/etc/passwd",
155        "/etc/shadow",
156        "~/.ssh/id_rsa",
157        "/root/.bashrc",
158    ];
159    sensitive_paths
160        .iter()
161        .map(|path| {
162            let result = sandbox.check_read(path);
163            SandboxVerificationResult {
164                path: path.to_string(),
165                blocked: result.is_err(),
166                reason: match result {
167                    Ok(()) => "access granted".to_string(),
168                    Err(e) => format!("{:?}", e),
169                },
170            }
171        })
172        .collect()
173}

244        let results = verify_sandbox_isolation(&sandbox);
245        assert!(
246            results.iter().all(|r| r.blocked),
