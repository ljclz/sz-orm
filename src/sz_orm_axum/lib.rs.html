42// ============================================================================
43// PoolState — 连接池的 axum State 包装
44// ============================================================================
45
46/// axum State wrapper for connection pool
47///
48/// `Pool` uses `Arc` internally for sharing, `PoolState` provides lightweight `Clone`,
49/// for easy use in `Router::with_state`.
50#[derive(Clone)]
51pub struct PoolState {
52    pool: Arc<Pool>,
53}
54
55impl PoolState {
56    /// Create PoolState
57    pub fn new(pool: Pool) -> Self {
58        Self {
59            pool: Arc::new(pool),
60        }
61    }
62
63    /// Create from Arc<Pool> (avoids duplicate Arc wrapping)
64    pub fn from_arc(pool: Arc<Pool>) -> Self {
65        Self { pool }
66    }
67
68    /// Get connection pool reference
69    pub fn pool(&self) -> &Pool {
70        &self.pool
71    }
72}
73
74/// Convert Pool directly to `Arc<Pool>` (for direct use as axum State)
75pub fn pool_into_state(pool: Pool) -> Arc<Pool> {
76    Arc::new(pool)
77}

79// ============================================================================
80// JsonRows — 查询结果的 JSON 响应
81// ============================================================================
82
83/// Wrap `QueryRows` implementing `IntoResponse`
84///
85/// `QueryRows = Vec<HashMap<String, Value>>`, since `Value` implements `Serialize`,
86/// directly serialized as JSON array.
87pub struct JsonRows(pub QueryRows);
88
89impl IntoResponse for JsonRows {
90    fn into_response(self) -> Response {
91        match serde_json::to_value(&self.0) {
92            Ok(v) => Json(v).into_response(),
93            Err(e) => (
94                StatusCode::INTERNAL_SERVER_ERROR,
95                format!("QueryRows 序列化失败: {}", e),
96            )
97                .into_response(),
98        }
99    }
100}

102// ============================================================================
103// JsonResp<T> — 通用 JSON 响应包装

106/// Generic JSON response wrapper
107///
108/// Provides `IntoResponse` implementation for any type implementing `Serialize`.
109pub struct JsonResp<T: Serialize>(pub T);
110
111impl<T: Serialize> IntoResponse for JsonResp<T> {
112    fn into_response(self) -> Response {
113        Json(self.0).into_response()
114    }
115}

117// ============================================================================
118// TransactionLayer — 事务中间件
119// ============================================================================
120
121/// Transaction middleware
122///
123/// Acquires connection from pool and starts transaction before request processing;
124/// After request processing:
125/// - Response status 2xx → commit transaction
126/// - Response status non-2xx → rollback transaction
127///
128/// # Usage
129///
130/// ```ignore
131/// use axum::middleware::from_fn_with_state;
132/// use sz_orm_axum::{PoolState, transaction_layer};
133///
134/// let app = Router::new()
135///     .route("/", get(handler))
136///     .layer(from_fn_with_state(state.clone(), transaction_layer))
137///     .with_state(state);
138/// ```
139pub async fn transaction_layer(
140    State(state): State<PoolState>,
141    req: Request<Body>,
142    next: Next,
143) -> Response {
144    // 1. 获取连接
145    let mut conn = match state.pool().acquire().await {
146        Ok(c) => c,
147        Err(e) => {
148            return (
149                StatusCode::INTERNAL_SERVER_ERROR,
150                format!("获取连接失败: {}", e),
151            )
152                .into_response()
153        }
154    };
155
156    // 2. 开启事务
157    if let Err(e) = conn.begin_transaction().await {
158        return (
159            StatusCode::INTERNAL_SERVER_ERROR,
160            format!("开启事务失败: {}", e),
161        )
162            .into_response();
163    }
164
165    // 3. 执行请求
166    let resp = next.run(req).await;
167
168    // 4. 根据响应状态提交或回滚
169    let tx_result = if resp.status().is_success() {
170        conn.commit().await
171    } else {
172        conn.rollback().await
173    };
174
175    if let Err(e) = tx_result {
176        return (
177            StatusCode::INTERNAL_SERVER_ERROR,
178            format!("事务结束失败: {}", e),
179        )
180            .into_response();
181    }
182
183    resp
184}
