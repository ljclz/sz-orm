10/// 全局 tokio runtime（PyO3 同步方法需要 block_on 执行异步池操作）
11fn runtime() -> &'static tokio::runtime::Runtime {
12    static RT: OnceLock<tokio::runtime::Runtime> = OnceLock::new();
13    RT.get_or_init(|| {
14        tokio::runtime::Builder::new_multi_thread()
15            .worker_threads(2)
16            .enable_all()
17            .build()
18            .expect("failed to build tokio runtime")
19    })
20}

37    #[new]
38    #[pyo3(signature = (db_type="mysql".to_string(), max_size=100, min_idle=0, acquire_timeout=30, idle_timeout=600, max_lifetime=1800))]
39    #[allow(clippy::too_many_arguments)]
40    fn new(
41        db_type: Option<String>,
42        max_size: Option<u32>,
43        min_idle: Option<u32>,
44        acquire_timeout: Option<u64>,
45        idle_timeout: Option<u64>,
46        max_lifetime: Option<u64>,
47    ) -> PyResult<Self> {
48        let dt = db_type.unwrap_or_else(|| "mysql".to_string());
49        let db_type = DbType::from_str(&dt)
50            .ok_or_else(|| PyValueError::new_err(format!("unknown DbType: {}", dt)))?;
51        Ok(Self {
52            db_type,
53            max_size: max_size.unwrap_or(100),
54            min_idle: min_idle.unwrap_or(0),
55            acquire_timeout_secs: acquire_timeout.unwrap_or(30),
56            idle_timeout_secs: idle_timeout.unwrap_or(600),
57            max_lifetime_secs: max_lifetime.unwrap_or(1800),
58            inner: None,
59        })
60    }

62    /// 建立真实连接（SQLite 后端）
63    ///
64    /// dsn 示例："sqlite::memory:" 或 "sqlite://path/to/db.sqlite"
65    #[pyo3(signature = (dsn="sqlite::memory:".to_string()))]
66    fn connect(&mut self, dsn: String) -> PyResult<()> {
67        if self.inner.is_some() {
68            return Ok(());
69        }
70        let cfg = PoolConfigBuilder::new()
71            .max_size(self.max_size.max(1))
72            .min_idle(self.min_idle.min(self.max_size.max(1)))
73            .acquire_timeout(self.acquire_timeout_secs)
74            .idle_timeout(self.idle_timeout_secs)
75            .build()
76            .map_err(|e| PyRuntimeError::new_err(format!("pool config: {e}")))?;
77
78        let pool = runtime()
79            .block_on(async {
80                let handle = Arc::new(
81                    sz_orm_sqlx::SqlitePoolHandle::connect(&dsn)
82                        .await
83                        .map_err(|e| PyRuntimeError::new_err(format!("connect: {e}")))?,
84                );
85                let factory = Arc::new(sz_orm_sqlx::SqlxSqliteConnectionFactory::new(handle));
86                Pool::new(cfg, factory).map_err(|e| PyRuntimeError::new_err(format!("pool: {e}")))
87            })
88            .map_err(|e: PyErr| e)?;
89
90        self.inner = Some(Arc::new(pool));
91        Ok(())
92    }

94    /// 执行写语句（INSERT/UPDATE/DELETE），返回影响行数
95    fn execute(&self, sql: &str) -> PyResult<u64> {
96        let pool = self
97            .inner
98            .as_ref()
99            .ok_or_else(|| PyRuntimeError::new_err("pool not connected, call connect() first"))?;
100        let pool = pool.clone();
101        runtime()
102            .block_on(async move {
103                let mut conn = pool
104                    .acquire()
105                    .await
106                    .map_err(|e| PyRuntimeError::new_err(format!("acquire: {e}")))?;
107                conn.execute(sql)
108                    .await
109                    .map_err(|e| PyRuntimeError::new_err(format!("execute: {e}")))
110            })
111            .map_err(|e: PyErr| e)
112    }

114    /// 执行查询，返回 JSON 行数组字符串
115    fn query(&self, sql: &str) -> PyResult<String> {
116        let pool = self
117            .inner
118            .as_ref()
119            .ok_or_else(|| PyRuntimeError::new_err("pool not connected, call connect() first"))?;
120        let pool = pool.clone();
121        runtime()
122            .block_on(async move {
123                let rows = pool
124                    .query_with_timeout(sql)
125                    .await
126                    .map_err(|e| PyRuntimeError::new_err(format!("query: {e}")))?;
127                serde_json::to_string(&rows)
128                    .map_err(|e| PyRuntimeError::new_err(format!("serialize: {e}")))
129            })
130            .map_err(|e: PyErr| e)
131    }

133    /// 健康检查（真实 acquire + ping）
134    fn ping(&self) -> PyResult<bool> {
135        let pool = self
136            .inner
137            .as_ref()
138            .ok_or_else(|| PyRuntimeError::new_err("pool not connected, call connect() first"))?;
139        let pool = pool.clone();
140        Ok(runtime().block_on(async move {
141            let mut conn = match pool.acquire().await {
142                Ok(c) => c,
143                Err(_) => return false,
144            };
145            conn.ping().await
146        }))
147    }

149    /// 关闭连接池
150    fn close(&mut self) {
151        if let Some(pool) = self.inner.take() {
152            runtime().block_on(async move {
153                pool.close_all().await;
154            });
155        }
156    }

207    /// 查询单行，返回 JSON 字符串（空行返回 "{}"）
208    fn query_one(&self, sql: &str) -> PyResult<String> {
209        let pool = self
210            .inner
211            .as_ref()
212            .ok_or_else(|| PyRuntimeError::new_err("pool not connected"))?;
213        let pool = pool.clone();
214        runtime()
215            .block_on(async move {
216                let rows = pool
217                    .query_with_timeout(sql)
218                    .await
219                    .map_err(|e| PyRuntimeError::new_err(format!("query: {e}")))?;
220                if rows.is_empty() {
221                    Ok("{}".to_string())
222                } else {
223                    serde_json::to_string(&rows[0])
224                        .map_err(|e| PyRuntimeError::new_err(format!("serialize: {e}")))
225                }
226            })
227            .map_err(|e: PyErr| e)
228    }

235    /// 检查表是否存在
236    fn table_exists(&self, table: &str) -> PyResult<bool> {
237        let pool = self
238            .inner
239            .as_ref()
240            .ok_or_else(|| PyRuntimeError::new_err("pool not connected"))?;
241        let pool = pool.clone();
242        let db_type = self.db_type;
243        runtime()
244            .block_on(async move {
245                let mut conn = pool
246                    .acquire()
247                    .await
248                    .map_err(|e| PyRuntimeError::new_err(format!("acquire: {e}")))?;
249                let check_sql = match db_type {
250                    DbType::Sqlite => format!("SELECT name FROM sqlite_master WHERE type='table' AND name='{table}'"),
251                    DbType::MySQL => format!("SHOW TABLES LIKE '{table}'"),
252                    _ => format!("SELECT 1 FROM information_schema.tables WHERE table_name = '{table}' LIMIT 1"),
253                };
254                let rows = conn
255                    .query(&check_sql)
256                    .await
257                    .map_err(|e| PyRuntimeError::new_err(format!("query: {e}")))?;
258                Ok(!rows.is_empty())
259            })
260            .map_err(|e: PyErr| e)
261    }

263    /// 统计表行数
264    fn count(&self, table: &str) -> PyResult<i64> {
265        let sql = format!("SELECT COUNT(*) AS cnt FROM {table}");
266        let json = self.query_one(&sql)?;
267        let v: serde_json::Value = serde_json::from_str(&json)
268            .map_err(|e| PyRuntimeError::new_err(format!("parse: {e}")))?;
269        Ok(v.get("cnt").and_then(|c| c.as_i64()).unwrap_or(0))
270    }

291    /// 开启事务（返回事务标识符，0 表示失败）
292    fn begin_transaction(&self) -> PyResult<bool> {
293        let pool = self
294            .inner
295            .as_ref()
296            .ok_or_else(|| PyRuntimeError::new_err("pool not connected"))?;
297        let pool = pool.clone();
298        runtime()
299            .block_on(async move {
300                let mut conn = pool
301                    .acquire()
302                    .await
303                    .map_err(|e| PyRuntimeError::new_err(format!("acquire: {e}")))?;
304                conn.execute("BEGIN")
305                    .await
306                    .map_err(|e| PyRuntimeError::new_err(format!("begin: {e}")))
307            })
308            .map(|_| true)
309            .map_err(|e: PyErr| e)
310    }

359    /// 插入记录（table, json_data）→ 影响行数
360    fn insert(&self, table: &str, data: &str) -> PyResult<u64> {
361        let v: serde_json::Value = serde_json::from_str(data)
362            .map_err(|e| PyRuntimeError::new_err(format!("parse data: {e}")))?;
363        let obj = v
364            .as_object()
365            .ok_or_else(|| PyRuntimeError::new_err("data must be a JSON object"))?;
366        let columns: Vec<&str> = obj.keys().map(|k| k.as_str()).collect();
367        let placeholders: Vec<&str> = columns.iter().map(|_| "?").collect();
368        let sql = format!(
369            "INSERT INTO {table} ({}) VALUES ({})",
370            columns.join(", "),
371            placeholders.join(", ")
372        );
373        self.execute(&sql)
374    }

380        let obj = v

383        let sets: Vec<String> = obj.keys().map(|k| format!("{k} = ?")).collect();
384        let sql = format!(
385            "UPDATE {table} SET {} WHERE {where_clause}",
386            sets.join(", ")
387        );
388        self.execute(&sql)

457    #[test]
458    fn test_pool_connect_and_e2e() {
459        init_python();
460        pyo3::Python::with_gil(|_py| {
461            let mut pool =
462                PyPool::new(Some("sqlite".to_string()), Some(4), None, None, None, None).unwrap();
463            assert!(!pool.is_connected());
464            pool.connect("sqlite::memory:".to_string()).unwrap();
465            assert!(pool.is_connected());
466            assert!(pool.ping().unwrap(), "pool should be healthy");
467
468            // 建表 + 插入（execute 返回 u64，无失败路径，仅断言调用成功）
469            let _ = pool
470                .execute("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)")
471                .unwrap();
472            let n = pool
473                .execute("INSERT INTO users (name) VALUES ('Alice'), ('Bob'), ('Carol')")
474                .unwrap();
475            assert_eq!(n, 3, "3 rows inserted");
476
477            // 查询 JSON
478            let json = pool
479                .query("SELECT id, name FROM users ORDER BY id")
480                .unwrap();
481            assert!(json.contains("Alice"), "missing Alice: {json}");
482            assert!(json.contains("Bob"), "missing Bob: {json}");
483            assert!(json.contains("Carol"), "missing Carol: {json}");
484
485            pool.close();
486            assert!(!pool.is_connected());
487        });
488    }
