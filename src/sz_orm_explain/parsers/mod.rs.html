13/// 从 EXPLAIN 文本中提取第一个 `rows=N` 形式的行数（PG/MySQL 均有此格式）
14pub(crate) fn extract_rows_from_kv(raw: &str) -> Option<u64> {
15    for line in raw.lines() {
16        if let Some(pos) = line.find("rows=") {
17            let rest = &line[pos + 5..];
18            let digits: String = rest.chars().take_while(|c| c.is_ascii_digit()).collect();
19            if !digits.is_empty() {
20                return digits.parse().ok();
21            }
22        }
23    }
24    None
25}
26
27/// 提取数字（过滤非数字字符），供表格型输出使用
28pub(crate) fn parse_u64(s: &str) -> Option<u64> {
29    let digits: String = s.chars().filter(|c| c.is_ascii_digit()).collect();
30    if digits.is_empty() {
31        None
32    } else {
33        digits.parse().ok()
34    }
35}
