23/// 工作负载摘要
24///
25/// 包含查询日志、时间范围和总查询数，用于驱动索引建议。
26#[derive(Debug, Clone, Serialize, Deserialize)]
27pub struct WorkloadSummary {
28    /// 慢查询日志
29    pub query_logs: Vec<SlowQueryLog>,
30    /// 时间范围
31    pub time_range: TimeRange,
32    /// 总查询数（含非慢查询）
33    pub total_queries: u64,
34}
35
36impl WorkloadSummary {
37    /// 创建工作负载摘要
38    pub fn new(query_logs: Vec<SlowQueryLog>, time_range: TimeRange, total_queries: u64) -> Self {
39        Self {
40            query_logs,
41            time_range,
42            total_queries,
43        }
44    }
45
46    /// 样本量是否充足（>= 100）
47    pub fn is_sufficient(&self) -> bool {
48        self.query_logs.len() >= 100
49    }
50
51    /// 获取 Top N 慢查询（按执行时间降序）
52    pub fn top_n_slow_queries(&self, n: usize) -> Vec<SlowQueryLog> {
53        let mut logs = self.query_logs.clone();
54        logs.sort_by_key(|a| std::cmp::Reverse(a.execution_time_ms));
55        logs.into_iter().take(n).collect()
56    }
57
58    /// 从慢查询日志提取查询模式
59    pub fn extract_patterns(&self) -> Vec<QueryPattern> {
60        let mut pattern_map: std::collections::HashMap<String, (u64, Vec<String>)> =
61            std::collections::HashMap::new();
62
63        for log in &self.query_logs {
64            let template = Self::sql_to_template(&log.sql);
65            let entry = pattern_map
66                .entry(template.clone())
67                .or_insert((0, Vec::new()));
68            entry.0 += 1;
69            // 提取列名（简化：从 WHERE 子句提取）
70            let cols = Self::extract_columns(&log.sql);
71            for col in cols {
72                if !entry.1.contains(&col) {
73                    entry.1.push(col);
74                }
75            }
76        }
77
78        pattern_map
79            .into_iter()
80            .map(|(template, (freq, cols))| QueryPattern {
81                sql_template: template,
82                frequency: freq,
83                columns_accessed: cols,
84            })
85            .collect()
86    }
87
88    /// 将 SQL 转为参数化模板（简化实现）
89    fn sql_to_template(sql: &str) -> String {
90        let lower = sql.to_lowercase();
91        lower
92            .replace("0x[0-9a-f]+", "?")
93            .replace("'-?'", "?")
94            .replace("'-?\\d+'", "?")
95    }
96
97    /// 从 SQL 提取列名（简化实现：从 WHERE/ORDER BY 子句提取）
98    fn extract_columns(sql: &str) -> Vec<String> {
99        let lower = sql.to_lowercase();
100        let mut cols = Vec::new();
101
102        if let Some(where_pos) = lower.find("where") {
103            let after_where = &sql[where_pos + 5..];
104            // 简化：按 AND/OR 分割，提取列名
105            for cond in after_where.split([' ', ')']) {
106                if cond.contains('=') || cond.contains('>') || cond.contains('<') {
107                    if let Some(col) = cond.split(['=', '>', '<', '!']).next() {
108                        let col = col.trim().trim_matches(|c: char| c.is_whitespace());
109                        if !col.is_empty()
110                            && !col.starts_with('\'')
111                            && !col.chars().all(|c| c.is_ascii_digit())
112                            && !cols.contains(&col.to_string())
113                        {
114                            cols.push(col.to_string());
115                        }
116                    }
117                }
118            }
119        }
120
121        cols
122    }
123}

127/// 工作负载驱动索引建议器
128///
129/// 从工作负载（查询日志 + 频率 + 耗时）分析高频查询模式，
130/// 生成覆盖 Top N 慢查询的复合索引建议。
131pub struct WorkloadDrivenIndexAdvisor {
132    /// 内部索引建议器
133    index_advisor: IndexAdvisor,
134    /// 组合优化器
135    combination_optimizer: IndexCombinationOptimizer,
136    /// Top N 慢查询数量（默认 10）
137    top_n: usize,
138}
139
140impl Default for WorkloadDrivenIndexAdvisor {
141    fn default() -> Self {
142        Self::new()
143    }
144}
145
146impl WorkloadDrivenIndexAdvisor {
147    /// 创建工作负载驱动索引建议器
148    pub fn new() -> Self {
149        Self {
150            index_advisor: IndexAdvisor::new(),
151            combination_optimizer: IndexCombinationOptimizer::new(),
152            top_n: 10,
153        }
154    }
155
156    /// 启用 LLM 增强
157    pub fn with_llm(mut self) -> Self {
158        self.index_advisor = self.index_advisor.with_llm();
159        self
160    }
161
162    /// 设置 Top N
163    pub fn with_top_n(mut self, n: usize) -> Self {
164        self.top_n = n;
165        self
166    }
167
168    /// 设置最大索引数
169    pub fn with_max_indexes(mut self, max: usize) -> Self {
170        self.combination_optimizer = self.combination_optimizer.with_max_indexes(max);
171        self
172    }

174    /// 基于工作负载生成索引建议
175    ///
176    /// 分析工作负载中的高频查询模式，生成覆盖 Top N 慢查询的复合索引建议。
177    /// 样本量 < 100 时返回建议但标注"样本量不足，建议置信度低"。
178    pub async fn advise_for_workload(
179        &self,
180        workload: &WorkloadSummary,
181    ) -> Result<WorkloadAdviceResult, IndexError> {
182        let is_sufficient = workload.is_sufficient();
183
184        // 提取查询模式
185        let patterns = workload.extract_patterns();
186        if patterns.is_empty() {
187            return Err(IndexError::NoQueryPatterns);
188        }
189
190        // 获取 Top N 慢查询
191        let top_slow = workload.top_n_slow_queries(self.top_n);
192
193        // 生成候选索引建议
194        let candidates = self.index_advisor.suggest(&patterns, &top_slow).await?;
195
196        // 组合优化
197        let optimized = self.combination_optimizer.optimize(candidates);
198
199        // 样本量不足时降低置信度
200        let suggestions = if is_sufficient {
201            optimized
202        } else {
203            optimized
204                .into_iter()
205                .map(|mut s| {
206                    s.expected_benefit.confidence *= 0.5;
207                    s.expected_benefit.uncertain = true;
208                    s
209                })
210                .collect()
211        };
212
213        Ok(WorkloadAdviceResult {
214            suggestions,
215            is_sufficient,
216            sample_size: workload.query_logs.len(),
217            top_n_analyzed: top_slow.len(),
218        })
219    }
220}

235// ==================== 索引组合优化器 ====================
236
237/// 索引组合优化器
238///
239/// 分析索引间互斥/互补关系，输出最优索引组合。
240/// 识别冗余索引（被覆盖的索引）并建议删除。
241pub struct IndexCombinationOptimizer {
242    /// 最大索引数（默认 5）
243    max_indexes: usize,
244}
245
246impl Default for IndexCombinationOptimizer {
247    fn default() -> Self {
248        Self::new()
249    }
250}
251
252impl IndexCombinationOptimizer {
253    /// 创建组合优化器
254    pub fn new() -> Self {
255        Self { max_indexes: 5 }
256    }
257
258    /// 设置最大索引数
259    pub fn with_max_indexes(mut self, max: usize) -> Self {
260        self.max_indexes = max;
261        self
262    }
263
264    /// 优化索引组合
265    ///
266    /// 分析索引间互斥/互补关系，输出最优索引组合。
267    /// 10 候选 → max_indexes 最优，附理由。
268    pub fn optimize(&self, candidates: Vec<IndexSuggestion>) -> Vec<IndexSuggestion> {
269        // 1. 检测冗余索引（被覆盖的索引）
270        let mut filtered = self.remove_redundant(candidates);
271
272        if filtered.len() <= self.max_indexes {
273            // 仍需按综合评分排序
274            filtered.sort_by(|a, b| {
275                b.expected_benefit
276                    .composite_score()
277                    .partial_cmp(&a.expected_benefit.composite_score())
278                    .unwrap_or(std::cmp::Ordering::Equal)
279            });
280            return filtered;
281        }
282
283        // 2. 按综合评分排序
284        filtered.sort_by(|a, b| {
285            b.expected_benefit
286                .composite_score()
287                .partial_cmp(&a.expected_benefit.composite_score())
288                .unwrap_or(std::cmp::Ordering::Equal)
289        });
290
291        // 3. 选取 Top N
292        filtered.into_iter().take(self.max_indexes).collect()
293    }
294
295    /// 移除冗余索引
296    ///
297    /// 如果索引 A 的列是索引 B 的列的前缀，则 A 被 B 覆盖，可移除。
298    /// 例如：idx(a) 被 idx(a, b) 覆盖。
299    fn remove_redundant(&self, candidates: Vec<IndexSuggestion>) -> Vec<IndexSuggestion> {
300        let mut result = Vec::new();
301
302        for (i, candidate) in candidates.iter().enumerate() {
303            let mut is_redundant = false;
304
305            for (j, other) in candidates.iter().enumerate() {
306                if i == j {
307                    continue;
308                }
309                // 检查 candidate 的列是否是 other 的列的前缀
310                if Self::is_prefix(&candidate.index_columns, &other.index_columns)
311                    && candidate.index_columns.len() < other.index_columns.len()
312                {
313                    is_redundant = true;
314                    break;
315                }
316            }
317
318            if !is_redundant {
319                result.push(candidate.clone());
320            }
321        }
322
323        result
324    }
325
326    /// 检查 cols_a 是否是 cols_b 的前缀
327    fn is_prefix(cols_a: &[String], cols_b: &[String]) -> bool {
328        if cols_a.len() > cols_b.len() {
329            return false;
330        }
331        cols_a
332            .iter()
333            .zip(cols_b.iter())
334            .all(|(a, b)| a.eq_ignore_ascii_case(b))
335    }
336
337    /// 量化收益
338    ///
339    /// 量化查询加速比 / 写入开销 / 存储开销。
340    pub fn quantify_benefit(
341        &self,
342        suggestion: &IndexSuggestion,
343        table_row_count: u64,
344        column_selectivity: f64,
345    ) -> BenefitEstimate {
346        // 加速比：基于列选择性估算
347        // 选择性越低（越接近 0），索引收益越高
348        let selectivity = column_selectivity.clamp(0.001, 1.0);
349        let speedup = 1.0 + (1.0 / selectivity).ln().max(0.0);
350
351        // 写入开销：索引列数越多，写入开销越大
352        let write_overhead = suggestion.index_columns.len() as f64 * 0.1;
353
354        // 存储开销：基于表行数 + 索引列数估算
355        // 假设每行每列 8 字节，MB = rows * cols * 8 / 1024 / 1024
356        let storage_cost_mb =
357            (table_row_count as f64 * suggestion.index_columns.len() as f64 * 8.0)
358                / (1024.0 * 1024.0);
359
360        BenefitEstimate::certain(speedup, suggestion.expected_benefit.confidence)
361            .with_write_overhead(write_overhead)
362            .with_storage_cost(storage_cost_mb)
363    }

365    /// 检测冗余索引并生成删除建议
366    ///
367    /// 返回 (保留的索引, 建议删除的索引)
368    pub fn detect_redundant(
369        &self,
370        existing_indexes: Vec<Vec<String>>,
371    ) -> (Vec<Vec<String>>, Vec<RedundantIndexInfo>) {
372        let mut keep = Vec::new();
373        let mut redundant = Vec::new();
374
375        for (i, idx) in existing_indexes.iter().enumerate() {
376            let mut covered_by: Option<usize> = None;
377
378            for (j, other) in existing_indexes.iter().enumerate() {
379                if i == j {
380                    continue;
381                }
382                if Self::is_prefix(idx, other) && idx.len() < other.len() {
383                    covered_by = Some(j);
384                    break;
385                }
386            }
387
388            if let Some(by) = covered_by {
389                redundant.push(RedundantIndexInfo {
390                    redundant_columns: idx.clone(),
391                    covered_by_columns: existing_indexes[by].clone(),
392                    reason: format!(
393                        "索引 ({}) 被复合索引 ({}) 覆盖，可删除以减少写入开销",
394                        idx.join(", "),
395                        existing_indexes[by].join(", ")
396                    ),
397                });
398            } else {
399                keep.push(idx.clone());
400            }
401        }
402
403        (keep, redundant)
404    }
405}

418// ==================== IndexUsageStats ====================
419
420/// 索引使用统计
421///
422/// 用于检测冗余索引和未使用索引。
423#[derive(Debug, Clone, Serialize, Deserialize)]
424pub struct IndexUsageStats {
425    /// 索引名
426    pub index_name: String,
427    /// 索引列
428    pub columns: Vec<String>,
429    /// 使用次数
430    pub usage_count: u64,
431    /// 最后使用时间（Unix 时间戳）
432    pub last_used: i64,
433}
434
435impl IndexUsageStats {
436    /// 索引是否未使用（usage_count == 0）
437    pub fn is_unused(&self) -> bool {
438        self.usage_count == 0
439    }
440
441    /// 索引是否低使用率（usage_count < threshold）
442    pub fn is_low_usage(&self, threshold: u64) -> bool {
443        self.usage_count < threshold
444    }
445}
446
447/// 检测未使用索引
448pub fn detect_unused_indexes(stats: &[IndexUsageStats]) -> Vec<&IndexUsageStats> {
449    stats.iter().filter(|s| s.is_unused()).collect()
450}
451
452/// 检测低使用率索引
453pub fn detect_low_usage_indexes(
454    stats: &[IndexUsageStats],
455    threshold: u64,
456) -> Vec<&IndexUsageStats> {
457    stats.iter().filter(|s| s.is_low_usage(threshold)).collect()
458}
