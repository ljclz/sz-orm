1//! v6.5.0 prepared_cache demo
2//!
3//! 演示 PreparedStatementCache：重复查询同一 SQL 模板 100 次，打印命中率。
4
5use sz_orm_core::prepared_cache::PreparedStatementCache;
6use sz_orm_core::Value;
7
8#[tokio::main]
9async fn main() {
10    println!("=== prepared_cache demo ===");
11
12    let cache = PreparedStatementCache::new(256);
13    let conn_id = 1;
14    let sql = "SELECT * FROM users WHERE id = ?";
15
16    for i in 1..=100 {
17        let _ = cache.get_or_prepare(conn_id, sql, &[Value::I64(i)]).await;
18    }
19
20    let stats = cache.stats();
21    println!("查询次数: 100");
22    println!("命中: {}", stats.hits);
23    println!("未命中: {}", stats.misses);
24    println!("命中率: {:.2}%", stats.hit_rate * 100.0);
25    println!("=== demo 完成 ===");
26}
