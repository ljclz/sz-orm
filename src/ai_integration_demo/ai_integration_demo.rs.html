82/// 演示 1：查询改写（规则路径）
83fn demo_rewrite() -> AiDecision {
84    println!("\n=== 1. 查询改写（规则路径）===");
85    let engine = RewriteEngine::new();
86
87    // 含子查询的 SQL，SubqueryFlatteningRule 应命中
88    let sql = "SELECT * FROM users WHERE id IN (SELECT user_id FROM orders WHERE amount > 100)";
89    println!("输入 SQL: {}", sql);
90
91    let result = engine.rewrite(sql);
92    match &result.suggestion {
93        Some(s) => {
94            println!("改写后 SQL: {}", s.rewritten_sql);
95            println!("命中规则: {}", s.transform_type.name());
96            println!("延迟: {}ms", result.latency_ms);
97            AiDecision::new(
98                AiDecisionType::Rewrite,
99                sql,
100                &s.rewritten_sql,
101                s.transform_type.name(),
102                0.9,
103                result.latency_ms,
104            )
105        }
106        None => {
107            let reason = result.fallback_reason.as_deref().unwrap_or("无规则命中");
108            println!("未改写: {}", reason);
109            AiDecision::new(
110                AiDecisionType::Rewrite,
111                sql,
112                sql,
113                reason,
114                0.0,
115                result.latency_ms,
116            )
117        }
118    }
119}

121/// 演示 2：索引推荐（负载建模 + 收益预估）
122async fn demo_index_advisor() -> AiDecision {
123    println!("\n=== 2. 索引推荐（负载建模 + 收益预估）===");
124    let advisor = IndexAdvisor::new();
125
126    let workload = WorkloadModel {
127        patterns: vec![QueryPattern {
128            sql_template: "SELECT * FROM users WHERE age > $1".to_string(),
129            frequency: 100,
130            columns_accessed: vec!["age".to_string()],
131        }],
132        table_stats: {
133            let mut stats = HashMap::new();
134            stats.insert(
135                "users".to_string(),
136                TableStats {
137                    row_count: 100_000,
138                    existing_indexes: vec!["id".to_string()],
139                },
140            );
141            stats
142        },
143    };
144
145    println!("负载: 1 个查询模式，users 表 100000 行");
146    let recommendations = advisor.recommend(&workload).await.unwrap();
147
148    if recommendations.is_empty() {
149        println!("无索引推荐");
150        return AiDecision::new(
151            AiDecisionType::IndexRecommend,
152            "workload with 1 pattern",
153            "无推荐",
154            "无候选索引",
155            0.0,
156            0,
157        );
158    }
159
160    let rec = &recommendations[0];
161    println!("推荐索引: {}", rec.suggestion.ddl_text);
162    println!(
163        "扫描降低: {:.1}%, 延迟降低: {:.1}%",
164        rec.estimated_scan_reduction * 100.0,
165        rec.estimated_latency_reduction * 100.0
166    );
167    println!("理由: {}", rec.reason);
168
169    AiDecision::new(
170        AiDecisionType::IndexRecommend,
171        "workload: SELECT users WHERE age",
172        &rec.suggestion.ddl_text,
173        &rec.reason,
174        0.85,
175        0,
176    )
177}

179/// 演示 3：NL2SQL 多轮对话（意图分析 + 注入防护）
180async fn demo_nl2sql() -> AiDecision {
181    println!("\n=== 3. NL2SQL 多轮对话（意图分析 + 注入防护）===");
182    let engine = SimpleNl2SqlEngine::new();
183    let schema = build_schema();
184    let mut ctx = MultiTurnContext::new(schema);
185
186    // 第一轮
187    let nl1 = "show all users";
188    println!("第一轮: \"{}\"", nl1);
189    let result1 = engine.nl2sql(nl1, &ctx).await.unwrap();
190    println!("  生成 SQL: {}", result1.sql.sql);
191    println!(
192        "  意图: {}, 实体: {:?}",
193        result1.intent.intent, result1.intent.entities
194    );
195    ctx.add_turn(nl1, result1.sql);
196
197    // 第二轮（带上下文）
198    let nl2 = "show users where age > 25";
199    println!("第二轮: \"{}\"", nl2);
200    let result2 = engine.nl2sql(nl2, &ctx).await.unwrap();
201    println!("  生成 SQL: {}", result2.sql.sql);
202    println!(
203        "  意图: {}, 实体: {:?}",
204        result2.intent.intent, result2.intent.entities
205    );
206    println!("  延迟: {}ms", result2.latency_ms);
207
208    AiDecision::new(
209        AiDecisionType::Nl2sql,
210        nl2,
211        &result2.sql.sql,
212        format!("intent={}", result2.intent.intent),
213        result2.sql.confidence as f64,
214        result2.latency_ms,
215    )
216}

218/// 演示 4：向量 ANN 检索（租户隔离 + 召回率）
219async fn demo_vector_ann() -> AiDecision {
220    println!("\n=== 4. 向量 ANN 检索（租户隔离 + 召回率）===");
221    let store = InMemoryVectorStore::new();
222    store.create_collection("docs", 3, None).await.unwrap();
223
224    // 插入多租户向量
225    let records = vec![
226        make_tenant_record("a1", vec![1.0, 0.0, 0.0], "tenant_a"),
227        make_tenant_record("a2", vec![0.9, 0.1, 0.0], "tenant_a"),
228        make_tenant_record("a3", vec![0.8, 0.2, 0.0], "tenant_a"),
229        make_tenant_record("b1", vec![1.0, 0.0, 0.0], "tenant_b"),
230        make_tenant_record("b2", vec![0.95, 0.05, 0.0], "tenant_b"),
231    ];
232    store.insert("docs", records).await.unwrap();
233
234    // tenant_a 检索
235    let result = store
236        .ann_search("docs", &[1.0, 0.0, 0.0], 5, Some("tenant_a"))
237        .await
238        .unwrap();
239
240    println!("查询: tenant_a 的向量检索");
241    println!("返回结果数: {}", result.records.len());
242    println!("召回率: {:.1}%", result.recall_rate * 100.0);
243    println!("延迟: {}ms", result.latency_ms);
244    for r in &result.records {
245        let tid = r
246            .metadata
247            .as_ref()
248            .and_then(|m| m.get("tenant_id"))
249            .and_then(|v| v.as_str())
250            .unwrap_or("");
251        println!("  - id={}, score={:.4}, tenant={}", r.id, r.score, tid);
252    }
253
254    AiDecision::new(
255        AiDecisionType::VectorSearch,
256        "ann_search docs tenant_a",
257        format!(
258            "{} results, recall={:.1}%",
259            result.records.len(),
260            result.recall_rate * 100.0
261        ),
262        "InMemoryVectorStore AnnAccelerated",
263        result.recall_rate,
264        result.latency_ms,
265    )
266}

268/// 构造带租户元数据的向量记录
269fn make_tenant_record(id: &str, vector: Vec<f32>, tenant_id: &str) -> VectorRecord {
270    let mut meta = HashMap::new();
271    meta.insert(
272        "tenant_id".to_string(),
273        serde_json::Value::String(tenant_id.to_string()),
274    );
275    VectorRecord::new(id, vector).with_metadata(meta)
276}

278/// 打印 AiConfig 配置
279fn print_config(config: &AiConfig) {
280    println!("\n=== AiConfig 配置 ===");
281    println!(
282        "  查询改写: {} (路径: {})",
283        config.rewrite_enabled,
284        config.rewrite_path.as_str()
285    );
286    println!("  索引推荐: {}", config.index_advisor_enabled);
287    println!(
288        "  NL2SQL: {} (多轮: {})",
289        config.nl2sql_enabled, config.nl2sql_multi_turn
290    );
291    println!("  向量 ANN 索引: {}", config.vector_ann_index.as_str());
292    println!(
293        "  召回率阈值: {:.1}%",
294        config.vector_recall_threshold * 100.0
295    );
296    println!("  配置合法: {}", config.validate().is_ok());
297    println!("  任一能力启用: {}", config.any_enabled());
298}

300#[tokio::main]
301async fn main() {
302    println!("========================================");
303    println!("sz-orm v7.3.0 AI 深度集成端到端演示");
304    println!("========================================");
305
306    // 1. 构造 AiConfig，启用四项 AI 能力
307    let config = AiConfig {
308        rewrite_enabled: true,
309        rewrite_path: RewritePath::Rule,
310        index_advisor_enabled: true,
311        nl2sql_enabled: true,
312        nl2sql_multi_turn: true,
313        vector_ann_index: AnnIndexType::Hnsw,
314        vector_recall_threshold: 0.9,
315    };
316    print_config(&config);
317
318    // 2. 执行四项 AI 能力，收集 AiDecision 审计记录
319    let mut decisions: Vec<AiDecision> = Vec::new();
320
321    // 2.1 查询改写
322    decisions.push(demo_rewrite());
323
324    // 2.2 索引推荐
325    decisions.push(demo_index_advisor().await);
326
327    // 2.3 NL2SQL 多轮对话
328    decisions.push(demo_nl2sql().await);
329
330    // 2.4 向量 ANN 检索
331    decisions.push(demo_vector_ann().await);
332
333    // 3. 汇总打印审计记录
334    println!("\n========================================");
335    println!("AiDecision 审计记录汇总（{} 条）", decisions.len());
336    println!("========================================");
337    for (i, d) in decisions.iter().enumerate() {
338        println!("\n[{}] 类型: {}", i + 1, d.decision_type.as_str());
339        println!("  输入: {}", d.input_summary);
340        println!("  输出: {}", d.output);
341        println!("  原因: {}", d.reason);
342        println!("  置信度: {:.2}, 延迟: {}ms", d.confidence, d.latency_ms);
343    }
344
345    println!("\n演示完成。");
346}
