1//! wiring_smoke — v4.7.0 接线冒烟示例（生产入口）
2//!
3//! 目的：为门禁 15 的接线断言（W1~W2 与符号断言）提供**可运行的
4//! 生产入口证据**——从真实二进制调用 v4.7.0 接线组件：
5//!   1. 租户配额：`Pool::acquire_with_tenant` / `release_with_tenant`
6//!      （QuotaEnforcer 接入连接池路径）
7//!   2. 缓存预热：`ProcessL1Cache::warmup`（CacheWarmer 接入）
8//!   3. 查询缓存：`QueryBuilder::execute_with_cache`（cache_ttl → L2Cache）
9//!   4. 观测导出：`Pool::metrics_snapshot_json`（运行时指标 JSON）
10//!
11//! 运行：`cargo run -p sz-orm-examples --bin wiring_smoke`
12//! 说明：使用内存 Mock 连接工厂，无需真实数据库。
13//! 输出：各接线点的成功标志 + Pool 指标 JSON（观测闭环演示）。
14

108// ============ 接线冒烟 ============
109
110#[tokio::main]
111async fn main() -> Result<(), Box<dyn std::error::Error>> {
112    println!("== v4.7.0 接线冒烟（wiring_smoke）==");
113
114    // 1. 租户配额接线：QuotaEnforcer → Pool::acquire_with_tenant / release_with_tenant
115    let config = PoolConfigBuilder::new().max_size(5).build()?;
116    let pool = Pool::new(config, Arc::new(MockFactory))?;
117    let enforcer = Arc::new(QuotaEnforcer::new());
118    enforcer.set_quota(TenantResourceQuota::new("t1").with_max_connections(3));
119    pool.set_quota_enforcer(Some(enforcer.clone()));
120
121    let conn = pool.acquire_with_tenant("t1").await?;
122    assert!(conn.is_connected());
123    let usage_after_acquire = enforcer.current_usage("t1", QuotaResource::Connection);
124    assert_eq!(usage_after_acquire, 1);
125    pool.release_with_tenant("t1", conn).await;
126    let usage_after_release = enforcer.current_usage("t1", QuotaResource::Connection);
127    assert_eq!(
128        usage_after_release, 0,
129        "release 必须递减配额（P0 修复回归）"
130    );
131    println!("✅ 1. 租户配额接线：acquire +1 / release -1（使用量 {usage_after_acquire}→{usage_after_release}）");
132
133    // 2. 缓存预热接线：CacheWarmer → ProcessL1Cache::warmup
134    use sz_orm_core::cache_warmup_protection::{WarmupConfig, WarmupStrategy};
135    use sz_orm_core::process_l1_cache::{ProcessL1Cache, ProcessL1Config};
136    let cache: Arc<ProcessL1Cache<String>> = Arc::new(ProcessL1Cache::new(
137        ProcessL1Config::new().with_capacity(100),
138    ));
139    let wconfig = WarmupConfig::new("users", WarmupStrategy::HotspotKey(vec!["1".to_string()]));
140    let result = cache
141        .warmup(&wconfig, |_| async {
142            Ok(vec![(Value::I64(1), "Alice".to_string())])
143        })
144        .await;
145    assert!(result.is_ok(), "预热应成功");
146    assert!(
147        cache.get("users", &Value::I64(1)).await.is_some(),
148        "预热后 key 应命中"
149    );
150    println!("✅ 2. 缓存预热接线：warmup 后 L1 命中");
151
152    // 3. 查询缓存接线：cache_ttl → L2Cache（execute_with_cache）
153    use sz_orm_core::dialect::MySqlDialect;
154    use sz_orm_core::l2_cache::L2Cache;
155    use sz_orm_core::QueryBuilder;
156    let l2 = L2Cache::new();
157    let qb = QueryBuilder::<SmokeUser>::new(Box::new(MySqlDialect))
158        .table("users")
159        .cache_ttl(std::time::Duration::from_secs(300));
160    let rows = qb
161        .execute_with_cache(&l2, "users", || async { Ok(vec![]) })
162        .await;
163    assert!(rows.is_ok(), "execute_with_cache 应成功");
164    println!("✅ 3. 查询缓存接线：execute_with_cache 消费 cache_ttl");
165
166    // 4. 组合防护接线：CacheProtection（穿透 + 击穿）进入缓存加载路径
167    use sz_orm_core::cache_warmup_protection::CacheProtection;
168    let protection = CacheProtection::new(Arc::clone(&cache), 1000);
169    // 加载路径：get_or_load 加载并注册布隆
170    let loaded = protection
171        .get_or_load("users", Value::I64(2), || async {
172            Ok(Some("Bob".to_string()))
173        })
174        .await
175        .unwrap();
176    assert_eq!(loaded.unwrap().as_str(), "Bob");
177    // 查询热路径：已注册 key 命中；未注册 key 被穿透防护拦截（不查 DB）
178    assert!(protection.get("users", &Value::I64(2)).await.is_some());
179    assert!(
180        protection.get("users", &Value::I64(999)).await.is_none(),
181        "未注册 key 必须被穿透防护拦截"
182    );
183    println!("✅ 4. 组合防护接线：CacheProtection 加载/命中/穿透拦截");
184
185    // 5. 配额审计接线：QuotaEnforcer 超限事件写入 TenantAuditLogger
186    use sz_orm_core::tenant_quota_rls::TenantAuditLogger;
187    let audit_logger = Arc::new(TenantAuditLogger::new());
188    let audit_enforcer = Arc::new(QuotaEnforcer::new());
189    audit_enforcer.set_audit_logger(Some(Arc::clone(&audit_logger)));
190    audit_enforcer.set_quota(TenantResourceQuota::new("t2").with_max_connections(1));
191    let exceeded = audit_enforcer.check_and_record("t2", QuotaResource::Connection, 2);
192    assert!(exceeded.is_err(), "超限应拒绝");
193    let audit_logs = audit_logger.filter_by_operation("t2", "quota_exceeded");
194    assert_eq!(audit_logs.len(), 1, "超限事件必须写入审计日志");
195    println!("✅ 5. 配额审计接线：超限拒绝 → 审计日志 1 条");
196
197    // 6. 观测导出：Pool 指标 JSON（观测闭环演示——monitoring/grafana 数据源）
198    let snapshot = pool.metrics_snapshot_json();
199    assert!(snapshot.contains("acquire_count"), "JSON 快照应含指标字段");
200    println!("✅ 6. 观测导出：metrics_snapshot_json = {snapshot}");
201
202    println!("== 接线冒烟全部通过（配额/预热/查询缓存/组合防护/审计/观测）==");
203    Ok(())
204}
