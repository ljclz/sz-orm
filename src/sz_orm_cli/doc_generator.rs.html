3use crate::entity_generator::{EntityDefinition, RelationType};

6#[derive(Debug, Clone, Serialize, Deserialize)]
7pub struct DocOutput {
8    pub markdown: String,
9    pub plantuml: String,
10}
11
12pub struct DocGenerator;
13
14impl DocGenerator {
15    pub fn new() -> Self {
16        Self
17    }

19    pub fn generate_doc(&self, entities: &[EntityDefinition]) -> DocOutput {
20        let markdown = self.generate_markdown(entities);
21        let plantuml = self.generate_er_diagram(entities);
22        DocOutput { markdown, plantuml }
23    }
24
25    fn generate_markdown(&self, entities: &[EntityDefinition]) -> String {
26        let mut md = String::new();
27        md.push_str("# Database Schema Documentation\n\n");
28        md.push_str(&format!("Generated entities: {}\n\n", entities.len()));
29        md.push_str("---\n\n");
30
31        for entity in entities {
32            md.push_str(&format!("## {}\n\n", entity.name));
33            md.push_str(&format!("**Table**: `{}`\n\n", entity.table_name));
34            md.push_str("### Fields\n\n");
35            md.push_str("| Name | Type | PK | Nullable | DB Type |\n");
36            md.push_str("|------|------|----|---------|--------|\n");
37            for field in &entity.fields {
38                md.push_str(&format!(
39                    "| `{}` | `{}` | {} | {} | `{}` |\n",
40                    field.name,
41                    field.rust_type,
42                    if field.is_primary_key { "✓" } else { "" },
43                    if field.nullable { "✓" } else { "" },
44                    field.db_type
45                ));
46            }
47            md.push('\n');
48            if !entity.relations.is_empty() {
49                md.push_str("### Relations\n\n");
50                for rel in &entity.relations {
51                    let rel_type = match rel.relation_type {
52                        RelationType::HasOne => "has one",
53                        RelationType::HasMany => "has many",
54                        RelationType::BelongsTo => "belongs to",
55                    };
56                    md.push_str(&format!(
57                        "- **{}** `{}` ({} → {})\n",
58                        rel_type, rel.target_entity, rel.local_field, rel.target_field
59                    ));
60                }
61                md.push('\n');
62            }
63            md.push_str("---\n\n");
64        }
65        md
66    }
67
68    fn generate_er_diagram(&self, entities: &[EntityDefinition]) -> String {
69        let mut puml = String::new();
70        puml.push_str("@startuml\n\n");
71        for entity in entities {
72            puml.push_str(&format!("entity {} {{\n", entity.name));
73            for field in &entity.fields {
74                if field.is_primary_key {
75                    puml.push_str(&format!("  * {} : {}\n", field.name, field.rust_type));
76                } else {
77                    puml.push_str(&format!("  {} : {}\n", field.name, field.rust_type));
78                }
79            }
80            puml.push_str("}\n\n");
81        }
82        for entity in entities {
83            for rel in &entity.relations {
84                let (from, to, arrow) = match rel.relation_type {
85                    RelationType::HasOne => (&entity.name, &rel.target_entity, "||--o|"),
86                    RelationType::HasMany => (&entity.name, &rel.target_entity, "||--|{"),
87                    RelationType::BelongsTo => (&rel.target_entity, &entity.name, "||--|{"),
88                };
89                puml.push_str(&format!(
90                    "{} {} {} : {}\n",
91                    from, arrow, to, rel.local_field
92                ));
93            }
94        }
95        puml.push_str("@enduml\n");
96        puml
97    }
98}
