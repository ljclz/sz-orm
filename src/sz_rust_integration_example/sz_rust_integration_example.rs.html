1//! sz-rust 协同集成示例 — 多后端透明访问
2//!
3//! 演示 sz-rust 透明适配层仅依赖 sz-orm 公开 API（AnyBackend/AnyPool/UnifiedPool）
4//! 完成统一访问，不触碰 sz-orm 内部实现。
5//!
6//! 本示例使用 SQLite 内存数据库，无需外部数据库连接。
7//!
8//! 运行：`cargo run -p sz-orm-examples --bin sz_rust_integration_example`
9
10use std::sync::Arc;
11
12use sz_orm_sqlx::any_driver::AnyBackend;
13use sz_orm_sqlx::UnifiedPool;
14
15/// 模拟 sz-rust 的 AppState — 持有单一类型 `Arc<UnifiedPool>`
16///
17/// 这是 sz-rust 集成 sz-orm 的推荐方式：
18/// - 业务代码无需感知后端类型
19/// - 通过 DSN 自动识别后端（mysql/postgres/sqlite/oracle/mssql）
20/// - 所有方法委托内部 Pool，零能力丢失
21struct AppState {
22    pool: Arc<UnifiedPool>,
23}
24
25impl AppState {
26    /// 从 DSN 创建 AppState
27    ///
28    /// sz-rust 只需传入配置中的 DSN 字符串，sz-orm 自动识别后端类型。
29    async fn from_dsn(dsn: &str) -> Result<Self, sz_orm_core::DbError> {
30        let pool = UnifiedPool::connect(dsn).await?;
31        Ok(Self {
32            pool: Arc::new(pool),
33        })
34    }
35
36    /// 获取后端类型（用于日志/调试）
37    fn backend(&self) -> AnyBackend {
38        self.pool.backend()
39    }
40}
41
42#[tokio::main]
43async fn main() -> Result<(), Box<dyn std::error::Error>> {
44    println!("=== sz-rust 协同集成示例 ===\n");
45
46    // ===== 1. DSN 自动识别后端 =====
47    println!("1. DSN 自动识别后端");
48
49    let test_dsns = [
50        ("mysql://root:pass@127.0.0.1/db", AnyBackend::MySql),
51        ("postgres://user:pass@127.0.0.1/db", AnyBackend::Postgres),
52        ("sqlite::memory:", AnyBackend::Sqlite),
53        ("oracle://user:pass@127.0.0.1/service", AnyBackend::Oracle),
54        ("mssql://user:pass@127.0.0.1/db", AnyBackend::Mssql),
55    ];
56
57    for (dsn, expected) in &test_dsns {
58        let detected = AnyBackend::from_dsn(dsn)?;
59        assert_eq!(detected, *expected);
60        println!("   {} → {:?} ✓", dsn, detected);
61    }
62    println!();
63
64    // ===== 2. 创建 AppState（使用 SQLite 内存数据库）=====
65    println!("2. 创建 AppState（SQLite 内存数据库）");
66
67    let state = AppState::from_dsn("sqlite::memory:").await?;
68    println!("   后端类型: {:?}", state.backend());
69    println!();
70
71    // ===== 3. CRUD 操作（透明访问）=====
72    println!("3. CRUD 操作（业务代码无需感知后端）");
73
74    let mut conn = state.pool.acquire().await?;
75
76    // CREATE TABLE
77    conn.execute("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT NOT NULL, email TEXT)")
78        .await?;
79    println!("   CREATE TABLE users ✓");
80
81    // INSERT
82    conn.execute("INSERT INTO users (id, name, email) VALUES (1, 'Alice', 'alice@example.com')")
83        .await?;
84    conn.execute("INSERT INTO users (id, name, email) VALUES (2, 'Bob', 'bob@example.com')")
85        .await?;
86    println!("   INSERT 2 rows ✓");
87
88    // SELECT
89    let rows = conn
90        .query("SELECT id, name, email FROM users ORDER BY id")
91        .await?;
92    println!("   SELECT: {} rows", rows.len());
93    for row in rows.iter() {
94        println!("     {:?}", row);
95    }
96
97    // UPDATE
98    conn.execute("UPDATE users SET email = 'alice@updated.com' WHERE id = 1")
99        .await?;
100    println!("   UPDATE ✓");
101
102    // SELECT 验证更新
103    let rows = conn.query("SELECT email FROM users WHERE id = 1").await?;
104    println!("   验证更新: {:?}", rows.first());
105    println!();
106
107    // ===== 4. 方言特性查询 =====
108    println!("4. 方言特性查询（运行时感知后端能力）");
109
110    let dialect = state.pool.dialect();
111    println!("   supports_returning: {}", dialect.supports_returning());
112    println!("   supports_if_exists: {}", dialect.supports_if_exists());
113    println!(
114        "   supports_if_not_exists: {}",
115        dialect.supports_if_not_exists()
116    );
117    println!(
118        "   supports_lock_for_update: {}",
119        dialect.supports_lock_for_update()
120    );
121    println!(
122        "   auto_increment_keyword: {}",
123        dialect.auto_increment_keyword()
124    );
125    println!("   last_insert_id_sql: {:?}", dialect.last_insert_id_sql());
126    println!();
127
128    // ===== 5. 连接池状态 =====
129    println!("5. 连接池状态");
130
131    let status = state.pool.status().await;
132    println!("   max: {}", status.max);
133    println!("   idle: {}", status.idle);
134    println!();
135
136    // ===== 6. 事务操作 =====
137    println!("6. 事务操作");
138
139    conn.begin_transaction().await?;
140    conn.execute(
141        "INSERT INTO users (id, name, email) VALUES (3, 'Charlie', 'charlie@example.com')",
142    )
143    .await?;
144    println!("   BEGIN + INSERT ✓");
145
146    conn.commit().await?;
147    println!("   COMMIT ✓");
148
149    let rows = conn.query("SELECT COUNT(*) as count FROM users").await?;
150    println!("   事务后行数: {:?}", rows.first());
151    println!();
152
153    // ===== 7. 事务回滚验证 =====
154    println!("7. 事务回滚验证");
155
156    conn.begin_transaction().await?;
157    conn.execute("INSERT INTO users (id, name, email) VALUES (4, 'Dave', 'dave@example.com')")
158        .await?;
159    conn.rollback().await?;
160    println!("   BEGIN + INSERT + ROLLBACK ✓");
161
162    let rows = conn.query("SELECT COUNT(*) as count FROM users").await?;
163    println!("   回滚后行数: {:?}", rows.first());
164    println!();
165
166    // ===== 8. 零成本迁移路径 =====
167    println!("8. 零成本迁移路径（from_pool）");
168    println!("   sz-rust 可从 Arc<Pool> 迁移到 Arc<UnifiedPool>，行为完全一致");
169    println!("   UnifiedPool::from_pool(existing_pool, AnyBackend::Sqlite)");
170    println!();
171
172    // ===== 9. 运行时切换后端 =====
173    println!("9. 运行时切换后端（DSN 驱动）");
174    println!("   同一份业务代码，只需改变 DSN 即可切换数据库后端：");
175    println!("   - 开发环境: sqlite::memory:");
176    println!("   - 测试环境: postgres://user:pass@test-db/db");
177    println!("   - 生产环境: mysql://user:pass@prod-db/db");
178    println!();
179
180    // ===== 10. 清理 =====
181    println!("10. 清理");
182
183    state.pool.close_all().await;
184    println!("   连接池已关闭 ✓");
185
186    println!("\n=== 示例完成 ===");
187    println!(
188        "\n结论：sz-orm 公开 API（AnyBackend + UnifiedPool）已满足 sz-rust P2-1 多后端 ORM 需求。"
189    );
190    println!("sz-rust 透明适配层无需触碰 sz-orm 内部实现，仅依赖公开 API 即可完成统一访问。");
191
192    Ok(())
193}
