1//! SZ-ORM Microsoft SQL Server 数据库适配器
2//!
3//! 基于 `tiberius` crate (纯 Rust TDS 协议实现) 实现 sz-orm-core 的 `Connection` trait,
4//! 支持 SQL Server 2008+ (TDS 7.3+)。
5//!
6//! # 设计说明
7//!
8//! `tiberius` 是纯 Rust 异步库,基于 TDS (Tabular Data Stream) 协议直接与 SQL Server
9//! 通信,无需原生客户端库 (如 ODBC/OLEDB)。
10//!
11//! tiberius 0.12 使用 `futures_io` traits,需通过 `tokio-util::compat` 包装 tokio TcpStream。
12//!
13//! SQL Server 占位符使用 `@P1, @P2, ...` 格式,本适配器自动将 SQL 中的 `?`
14//! 转换为 `@PN`。
15//!
16//! # 用法
17//!
18//! ```ignore
19//! use sz_orm_core::{Pool, PoolConfigBuilder};
20//! use sz_orm_mssql::{MssqlPoolHandle, MssqlConnectionFactory};
21//! use std::sync::Arc;
22//!
23//! # #[tokio::main]
24//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
25//! let handle = Arc::new(MssqlPoolHandle::connect("server=tcp:localhost,1433;user=sa;password=P@ssw0rd;database=test").await?);
26//! let factory = Arc::new(MssqlConnectionFactory::new(handle));
27//! let config = PoolConfigBuilder::new().max_size(10).build()?;
28//! let pool = Pool::new(config, factory)?;
29//! # Ok(())
30//! # }
31//! ```

51// 连接池内部状态用 `std::sync::Mutex` 保护：只在 `acquire()`（弹出空闲连接/更新
52// 计数）与 `Drop`（归还连接）时短暂持锁，不在锁内执行任何 `.await`，因此用同步
53// 互斥锁而非 `tokio::sync::Mutex`，避免跨 await 持锁且开销更低。

55/// 错误转换: tiberius::error::Error → DbError
56fn map_tiberius_error(e: tiberius::error::Error) -> DbError {
57    let msg = e.to_string();
58    if msg.contains("Connection")
59        || msg.contains("connection")
60        || msg.contains("Broken pipe")
61        || msg.contains("Login failed")
62    {
63        DbError::ConnectionError(msg)
64    } else if msg.contains("timeout") || msg.contains("Timeout") {
65        DbError::ConnectionTimeout(msg)
66    } else if msg.contains("already exists") || msg.contains("2627") || msg.contains("2601") {
67        // SQL Server 2627=唯一约束违反, 2601=重复键索引
68        DbError::UniqueViolation(msg)
69    } else if msg.contains("547") {
70        // SQL Server 547=外键约束违反
71        DbError::ForeignKeyViolation(msg)
72    } else if msg.contains("515") {
73        DbError::NullValue(msg)
74    } else if msg.contains("208") || msg.contains("Invalid object name") {
75        DbError::NotFound(msg)
76    } else if msg.contains("Config") || msg.contains("config") {
77        DbError::ConfigError(msg)
78    } else {
79        DbError::QueryError(msg)
80    }
81}

83/// 判断 SQL 是否需要走 simple_query 路径 (DDL/DCL/事务控制语句不走 prepared statement)
84fn needs_simple_query(sql: &str) -> bool {
85    let upper = sql.trim_start().to_uppercase();
86    upper.starts_with("CREATE ")
87        || upper.starts_with("ALTER ")
88        || upper.starts_with("DROP ")
89        || upper.starts_with("TRUNCATE ")
90        || upper.starts_with("GRANT ")
91        || upper.starts_with("REVOKE ")
92        || upper.starts_with("USE ")
93        || upper.starts_with("SET ")
94        || upper.starts_with("BEGIN TRAN")
95        || upper.starts_with("COMMIT")
96        || upper.starts_with("ROLLBACK")
97        || upper.starts_with("SAVE TRAN")
98        || upper.starts_with("EXEC ")
99        || upper.starts_with("EXECUTE ")
100}

102/// 将 SQL 中的 `?` 占位符转换为 SQL Server 的 `@PN` 格式
103///
104/// 跳过字符串字面量内的 `?`,仅转换参数占位符。
105fn convert_placeholders(sql: &str) -> String {
106    let mut result = String::with_capacity(sql.len() + 16);
107    let mut in_single_quote = false;
108    let mut param_index = 1u32;
109
110    for ch in sql.chars() {
111        match ch {
112            '\'' if !in_single_quote => {
113                in_single_quote = true;
114                result.push(ch);
115            }
116            '\'' if in_single_quote => {
117                in_single_quote = false;
118                result.push(ch);
119            }
120            '?' if !in_single_quote => {
121                result.push_str(&format!("@P{}", param_index));
122                param_index += 1;
123            }
124            _ => {
125                result.push(ch);
126            }
127        }
128    }
129    result
130}

132/// 拥有所有权的参数枚举,实现 tiberius::ToSql 用于 prepared statement 绑定。
133///
134/// tiberius 0.12 的 `Client::execute/query` 接受 `&[&dyn ToSql]`,
135/// 通过为 `MssqlParamOwned` 实现 `ToSql`,可直接作为参数传入。
136enum MssqlParamOwned {
137    Null,
138    Bool(Option<bool>),
139    I16(Option<i16>),
140    I32(Option<i32>),
141    I64(Option<i64>),
142    U8(Option<u8>),
143    F32(Option<f32>),
144    F64(Option<f64>),
145    String(String),
146    Bytes(Vec<u8>),
147}
148
149impl ToSql for MssqlParamOwned {
150    fn to_sql(&self) -> ColumnData<'_> {
151        match self {
152            MssqlParamOwned::Null => ColumnData::I32(None),
153            MssqlParamOwned::Bool(b) => ColumnData::Bit(*b),
154            MssqlParamOwned::I16(n) => ColumnData::I16(*n),
155            MssqlParamOwned::I32(n) => ColumnData::I32(*n),
156            MssqlParamOwned::I64(n) => ColumnData::I64(*n),
157            MssqlParamOwned::U8(n) => ColumnData::U8(*n),
158            MssqlParamOwned::F32(f) => ColumnData::F32(*f),
159            MssqlParamOwned::F64(f) => ColumnData::F64(*f),
160            MssqlParamOwned::String(s) => {
161                ColumnData::String(Some(std::borrow::Cow::Borrowed(s.as_str())))
162            }
163            MssqlParamOwned::Bytes(b) => {
164                ColumnData::Binary(Some(std::borrow::Cow::Borrowed(b.as_slice())))
165            }
166        }
167    }
168}

170/// 将 sz-orm Value 转换为 MssqlParamOwned (拥有所有权的参数)
171///
172/// # 性能说明
173///
174/// `String`/`Bytes` 分支会 clone 底层数据。由于 `Value` 本身持有所有权，
175/// 而 `MssqlParamOwned` 也需要 owned 数据以跨越 async 边界（tiberius 的
176/// `Client::execute/query` 异步执行），此处的 clone 无法避免。
177///
178/// 如需降低单次调用的 clone 开销，建议：
179/// - 使用批量操作（`sz-orm-batch`）摊薄每次参数构造的成本；
180/// - 在循环外预构造参数并复用；
181/// - 对大文本/二进制字段考虑分页或流式处理。
182fn value_to_mssql_param_owned(value: &Value) -> MssqlParamOwned {
183    match value {
184        Value::Null => MssqlParamOwned::Null,
185        Value::Bool(b) => MssqlParamOwned::Bool(Some(*b)),
186        Value::I8(n) => MssqlParamOwned::I32(Some(*n as i32)),
187        Value::I16(n) => MssqlParamOwned::I16(Some(*n)),
188        Value::I32(n) => MssqlParamOwned::I32(Some(*n)),
189        Value::I64(n) => MssqlParamOwned::I64(Some(*n)),
190        Value::U8(n) => MssqlParamOwned::U8(Some(*n)),
191        Value::U16(n) => MssqlParamOwned::I32(Some(*n as i32)),
192        Value::U32(n) => MssqlParamOwned::I64(Some(*n as i64)),
193        Value::U64(n) => MssqlParamOwned::I64(Some(*n as i64)),
194        Value::F32(f) => MssqlParamOwned::F32(Some(*f)),
195        Value::F64(f) => MssqlParamOwned::F64(Some(*f)),
196        Value::String(s) => MssqlParamOwned::String(s.clone()),
197        Value::Bytes(b) => MssqlParamOwned::Bytes(b.clone()),
198        // DECIMAL/NUMERIC 以字符串绑定，SQL Server 隐式转换为 DECIMAL 列
199        Value::Decimal(s) => MssqlParamOwned::String(s.clone()),
200        Value::Date(s) | Value::DateTime(s) | Value::Time(s) => MssqlParamOwned::String(s.clone()),
201        Value::Uuid(s) => MssqlParamOwned::String(s.clone()),
202        Value::Json(s) => MssqlParamOwned::String(s.clone()),
203        Value::Array(arr) => {
204            let json = serde_json::to_string(arr).unwrap_or_else(|_| "[]".to_string());
205            MssqlParamOwned::String(json)
206        }
207        Value::Object(obj) => {
208            let json = serde_json::to_string(obj).unwrap_or_else(|_| "{}".to_string());
209            MssqlParamOwned::String(json)
210        }
211        // 兼容 #[non_exhaustive] 的 Value：未知变体降级为 Null
212        _ => MssqlParamOwned::Null,
213    }
214}

216/// 将 tiberius Row 中指定列转换为 sz-orm Value
217///
218/// tiberius 0.12 的 `FromSql` 为 `&str`/`&[u8]` 实现 (借用),`String`/`Vec<u8>` 仅实现 `FromSqlOwned`。
219/// `try_get` 需要 `FromSql<'a>`,因此使用 `&str`/`&[u8]` 并转为 owned。
220///
221/// DECIMAL/NUMERIC/MONEY/SMALLMONEY 优先以 `rust_decimal::Decimal` 解码为
222/// `Value::Decimal`(字符串形式),避免 f64 精度丢失。
223fn row_to_value(row: &MssqlRow, idx: usize) -> Value {
224    // 1. 尝试 i64 (覆盖 BIT/TINYINT/SMALLINT/INT/BIGINT)
225    if let Ok(v) = row.try_get::<i64, usize>(idx) {
226        return v.map(Value::I64).unwrap_or(Value::Null);
227    }
228    // 2. 尝试 rust_decimal::Decimal (覆盖 DECIMAL/NUMERIC/MONEY/SMALLMONEY，保留精度)
229    if let Ok(v) = row.try_get::<rust_decimal::Decimal, usize>(idx) {
230        return v
231            .map(|d| Value::Decimal(d.to_string()))
232            .unwrap_or(Value::Null);
233    }
234    // 3. 尝试 f64 (覆盖 REAL/FLOAT)
235    if let Ok(v) = row.try_get::<f64, usize>(idx) {
236        return v.map(Value::F64).unwrap_or(Value::Null);
237    }
238    // 4. 尝试 bool (覆盖 BIT)
239    if let Ok(v) = row.try_get::<bool, usize>(idx) {
240        return v.map(Value::Bool).unwrap_or(Value::Null);
241    }
242    // 5. 尝试 &[u8] (覆盖 BINARY/VARBINARY/IMAGE)
243    if let Ok(v) = row.try_get::<&[u8], usize>(idx) {
244        return v.map(|b| Value::Bytes(b.to_vec())).unwrap_or(Value::Null);
245    }
246    // 6. 尝试 &str (覆盖 CHAR/VARCHAR/NCHAR/NVARCHAR/TEXT/NTEXT/DATE/TIME/DATETIME 等)
247    if let Ok(v) = row.try_get::<&str, usize>(idx) {
248        return v
249            .map(|s| Value::String(s.to_string()))
250            .unwrap_or(Value::Null);
251    }
252    Value::Null
253}

255/// MSSQL 连接池内部状态
256struct MssqlPoolInner {
257    /// 空闲连接列表：`acquire()` 从尾部弹出，`Drop` 归还时推入尾部
258    idle: Vec<tiberius::Client<CompatTcpStream>>,
259    /// 已创建连接总数（空闲 + 在用），用于约束池大小上限
260    total: usize,
261}
262
263/// SQL Server 连接池句柄
264///
265/// v1.2.0 修复 P0：从单连接伪池（`Mutex<Option<Client>>`，并发度=1）改为
266/// 真正的连接池（`Mutex<MssqlPoolInner>` 持有多个空闲连接 + `Notify` 等待
267/// 机制），并发度 = `max_size`（默认 10）。
268///
269/// # 设计要点
270///
271/// - **多连接并发**：`idle` 持有多个空闲连接，`acquire()` 弹出一个供调用方
272///   独占使用，查询结束后通过 `MssqlClientGuard::drop` 归还
273/// - **池大小约束**：`total` 跟踪已创建连接数（空闲+在用），`acquire()` 在
274///   `total < max_size` 时创建新连接，否则在 `Notify` 上等待归还
275/// - **锁粒度**：`std::sync::Mutex` 仅在弹出/归还/计数更新时短暂持有，不跨
276///   `.await`；`tiberius::Client::connect` 是异步建连，在锁外执行
277/// - **连接复用**：连接在 `Drop` 时归还到 `idle`，不关闭，避免反复建连开销
278pub struct MssqlPoolHandle {
279    conn_str: String,
280    /// 连接池内部状态（空闲连接 + 总数）
281    inner: Mutex<MssqlPoolInner>,
282    /// 连接归还通知：池满时 `acquire()` 在此 `notified().await`，`Drop` 时 `notify_one`
283    notify: Notify,
284    /// 连接池大小上限（默认 10）
285    max_size: usize,
286}
287
288impl MssqlPoolHandle {
289    /// 创建新的 SQL Server 连接池（连接池上限默认 10）
290    pub async fn connect(conn_str: &str) -> Result<Self, DbError> {
291        Self::connect_with_max_size(conn_str, 10).await
292    }
293
294    /// 创建新的 SQL Server 连接池（自定义连接池大小上限）
295    ///
296    /// # 参数
297    ///
298    /// - `max_size`：连接池大小上限。传入 0 会被替换为默认值 10
299    pub async fn connect_with_max_size(conn_str: &str, max_size: usize) -> Result<Self, DbError> {
300        let config = tiberius::Config::from_ado_string(conn_str)
301            .map_err(|e| DbError::ConfigError(e.to_string()))?;
302        let tcp = tokio::net::TcpStream::connect(config.get_addr())
303            .await
304            .map_err(|e| DbError::ConnectionError(e.to_string()))?;
305        tcp.set_nodelay(true).ok();
306        let client = tiberius::Client::connect(config, tcp.compat())
307            .await
308            .map_err(map_tiberius_error)?;
309        let max_size = if max_size == 0 { 10 } else { max_size };
310        Ok(Self {
311            conn_str: conn_str.to_string(),
312            inner: Mutex::new(MssqlPoolInner {
313                idle: vec![client],
314                total: 1,
315            }),
316            notify: Notify::new(),
317            max_size,
318        })
319    }
320
321    /// 从池中获取一个空闲连接（独占使用，`Drop` 时自动归还）
322    ///
323    /// # 获取策略
324    ///
325    /// 1. 优先从 `idle` 弹出一个空闲连接
326    /// 2. 若 `idle` 为空且 `total < max_size`，创建新连接并递增 `total`
327    /// 3. 若 `total >= max_size`，在 `notify` 上等待，直到有连接归还
328    ///
329    /// # 锁粒度
330    ///
331    /// `std::sync::Mutex` 仅在弹出连接/更新计数时短暂持有，异步建连
332    /// （`TcpStream::connect` / `Client::connect`）在锁外 `.await`，
333    /// 不阻塞其他调用方的 `acquire()`。
334    pub async fn acquire(&self) -> Result<MssqlClientGuard<'_>, DbError> {
335        loop {
336            // 短暂持锁：弹出空闲连接或决定是否创建新连接
337            let create_new = {
338                let mut inner = self
339                    .inner
340                    .lock()
341                    .map_err(|e| DbError::Internal(format!("MSSQL pool mutex poisoned: {}", e)))?;
342                if let Some(client) = inner.idle.pop() {
343                    return Ok(MssqlClientGuard {
344                        client: Some(client),
345                        pool: self,
346                    });
347                }
348                if inner.total < self.max_size {
349                    // 预占一个名额，锁外再建连
350                    inner.total += 1;
351                    true
352                } else {
353                    false
354                }
355            };
356
357            if create_new {
358                // 锁外异步建连：不阻塞其他 acquire()
359                match self.build_client().await {
360                    Ok(client) => {
361                        return Ok(MssqlClientGuard {
362                            client: Some(client),
363                            pool: self,
364                        });
365                    }
366                    Err(e) => {
367                        // 建连失败：回滚 total 并唤醒一个等待者，避免其空等
368                        let mut inner = self.inner.lock().map_err(|e| {
369                            DbError::Internal(format!("MSSQL pool mutex poisoned: {}", e))
370                        })?;
371                        inner.total -= 1;
372                        drop(inner);
373                        self.notify.notify_one();
374                        return Err(e);
375                    }
376                }
377            } else {
378                // 池满：等待连接归还
379                self.notify.notified().await;
380            }
381        }
382    }
383
384    /// 建立一个新的 tiberius 客户端（私有辅助方法）
385    async fn build_client(&self) -> Result<tiberius::Client<CompatTcpStream>, DbError> {
386        let config = tiberius::Config::from_ado_string(&self.conn_str)
387            .map_err(|e| DbError::ConfigError(e.to_string()))?;
388        let tcp = tokio::net::TcpStream::connect(config.get_addr())
389            .await
390            .map_err(|e| DbError::ConnectionError(e.to_string()))?;
391        tcp.set_nodelay(true).ok();
392        tiberius::Client::connect(config, tcp.compat())
393            .await
394            .map_err(map_tiberius_error)
395    }
396
397    pub fn conn_str(&self) -> &str {
398        &self.conn_str
399    }

401    /// 获取连接池大小上限
402    pub fn max_size(&self) -> usize {
403        self.max_size
404    }

407/// SQL Server 连接的独占守卫
408///
409/// 持有从池中取出的客户端，`Drop` 时自动归还到池的 `idle` 列表并通知一个等待者。
410/// 通过 `Deref` / `DerefMut` 透明访问底层 `tiberius::Client`。
411pub struct MssqlClientGuard<'a> {
412    /// 当前持有的客户端；`Drop` 时取出并归还，正常使用期间始终为 `Some`
413    client: Option<tiberius::Client<CompatTcpStream>>,
414    /// 池的引用，用于归还连接
415    pool: &'a MssqlPoolHandle,
416}
417
418impl std::ops::Deref for MssqlClientGuard<'_> {
419    type Target = tiberius::Client<CompatTcpStream>;
420    fn deref(&self) -> &tiberius::Client<CompatTcpStream> {
421        self.client.as_ref().expect("connection must exist")
422    }
423}
424
425impl std::ops::DerefMut for MssqlClientGuard<'_> {
426    fn deref_mut(&mut self) -> &mut tiberius::Client<CompatTcpStream> {
427        self.client.as_mut().expect("connection must exist")
428    }
429}
430
431impl Drop for MssqlClientGuard<'_> {
432    fn drop(&mut self) {
433        if let Some(client) = self.client.take() {
434            // 即使 mutex poisoned 也强制归还，避免连接泄漏
435            let mut inner = self.pool.inner.lock().unwrap_or_else(|e| e.into_inner());
436            inner.idle.push(client);
437            drop(inner);
438            // 唤醒一个等待 acquire() 的任务
439            self.pool.notify.notify_one();
440        }
441    }
442}

483impl Connection for MssqlConnection {
484    fn execute<'a>(
485        &'a mut self,
486        sql: &'a str,
487    ) -> Pin<Box<dyn Future<Output = Result<u64, DbError>> + Send + 'a>> {
488        Box::pin(async move {
489            if !self.connected {
490                return Err(DbError::ConnectionError("connection closed".to_string()));
491            }
492            let result = {
493                let mut guard = self.handle.acquire().await?;
494                let client = &mut *guard;
495                if needs_simple_query(sql) {
496                    // DDL/事务控制：simple_query 不返回 rows_affected，消费流即可
497                    let mut stream = client.simple_query(sql).await.map_err(map_tiberius_error)?;
498                    while stream.next().await.is_some() {}
499                    Ok(1u64)
500                } else {
501                    // DML 无参数：用 execute 获取 rows_affected
502                    let exec_result = client.execute(sql, &[]).await.map_err(map_tiberius_error)?;
503                    Ok(exec_result.total())
504                }
505            };
506            match result {
507                Ok(n) => Ok(n),
508                Err(e) => {
509                    self.mark_connection_error_after(&e);
510                    Err(e)
511                }
512            }
513        })
514    }

516    fn query<'a>(
517        &'a mut self,
518        sql: &'a str,
519    ) -> Pin<Box<dyn Future<Output = Result<QueryRows, DbError>> + Send + 'a>> {
520        Box::pin(async move {
521            if !self.connected {
522                return Err(DbError::ConnectionError("connection closed".to_string()));
523            }
524            let result = {
525                let mut guard = self.handle.acquire().await?;
526                let client = &mut *guard;
527                let mut stream = client.simple_query(sql).await.map_err(map_tiberius_error)?;
528                let mut result_rows: QueryRows = Vec::new();
529                let mut col_names: Vec<String> = Vec::new();
530                let mut col_set = false;
531                while let Some(item_result) = stream.next().await {
532                    match item_result {
533                        Ok(tiberius::QueryItem::Row(row)) => {
534                            if !col_set {
535                                for col in row.columns() {
536                                    col_names.push(col.name().to_string());
537                                }
538                                col_set = true;
539                            }
540                            let mut row_map: HashMap<String, Value> = HashMap::new();
541                            for (i, col_name) in col_names.iter().enumerate() {
542                                let value = row_to_value(&row, i);
543                                row_map.insert(col_name.clone(), value);
544                            }
545                            result_rows.push(row_map);
546                        }
547                        Ok(tiberius::QueryItem::Metadata(_)) => {
548                            // 元数据项，忽略（列信息从第一行提取）
549                        }
550                        Err(e) => {
551                            return Err(map_tiberius_error(e));
552                        }
553                    }
554                }
555                Ok(result_rows)
556            };
557            match result {
558                Ok(rows) => Ok(rows),
559                Err(e) => {
560                    self.mark_connection_error_after(&e);
561                    Err(e)
562                }
563            }
564        })
565    }

567    /// G-SX-4：MSSQL 原生游标流式查询
568    ///
569    /// tiberius `simple_query` 返回 `QueryItem::Row` 的异步流，本方法直接
570    /// yield 每一行，避免默认实现将全部行收集到 Vec 再逐行返回。
571    /// 适用于大结果集场景，显著降低内存峰值。
572    fn query_stream<'a>(
573        &'a mut self,
574        sql: &'a str,
575    ) -> Pin<Box<dyn futures::Stream<Item = QueryStreamItem> + Send + 'a>> {
576        Box::pin(async_stream::try_stream! {
577            if !self.connected {
578                Err(DbError::ConnectionError("connection closed".to_string()))?;
579            }
580            let mut guard = self.handle.acquire().await?;
581            let client = &mut *guard;
582            let mut stream = client.simple_query(sql).await.map_err(map_tiberius_error)?;
583            let mut col_names: Vec<String> = Vec::new();
584            let mut col_set = false;
585            while let Some(item_result) = stream.next().await {
586                match item_result {
587                    Ok(tiberius::QueryItem::Row(row)) => {
588                        if !col_set {
589                            for col in row.columns() {
590                                col_names.push(col.name().to_string());
591                            }
592                            col_set = true;
593                        }
594                        let mut row_map: HashMap<String, Value> = HashMap::new();
595                        for (i, col_name) in col_names.iter().enumerate() {
596                            let value = row_to_value(&row, i);
597                            row_map.insert(col_name.clone(), value);
598                        }
599                        yield row_map;
600                    }
601                    Ok(tiberius::QueryItem::Metadata(_)) => {
602                        // 元数据项，忽略（列信息从第一行提取）
603                    }
604                    Err(e) => {
605                        Err(map_tiberius_error(e))?;
606                    }
607                }
608            }
609        })
610    }

613        &'a mut self,
614        sql: &'a str,
615        params: &'a [Value],
616    ) -> Pin<Box<dyn Future<Output = Result<u64, DbError>> + Send + 'a>> {
617        Box::pin(async move {

620            }

622                return self.execute(sql).await;
623            }
624            let sql_converted = convert_placeholders(sql);
625            let params_owned: Vec<MssqlParamOwned> =
626                params.iter().map(value_to_mssql_param_owned).collect();

630            let result = {

632                let client = &mut *guard;
633                let exec_result = client
634                    .execute(&sql_converted, &params_ref)
635                    .await
636                    .map_err(map_tiberius_error)?;
637                Ok(exec_result.total())
638            };
639            match result {
640                Ok(n) => Ok(n),
641                Err(e) => {
642                    self.mark_connection_error_after(&e);
643                    Err(e)
644                }
645            }
646        })
647    }

710    /// 位置式查询（SELECT，无参数）：返回 `(列名, 按列顺序的值矩阵)`
711    ///
712    /// 绕过 `HashMap<String, Value>` 行映射，列名仅 `to_string` 一次后复用，
713    /// 每行值按列序号直接 `Vec::push`，无哈希计算与字符串克隆。
714    /// 适用于 SELECT ALL 大结果集场景，相比 [`query`](Connection::query) 可获得 30%~50% 性能提升。
715    fn query_values<'a>(
716        &'a mut self,
717        sql: &'a str,
718    ) -> Pin<Box<dyn Future<Output = Result<QueryValues, DbError>> + Send + 'a>> {
719        Box::pin(async move {
720            if !self.connected {
721                return Err(DbError::ConnectionError("connection closed".to_string()));
722            }
723            let result = {
724                let mut guard = self.handle.acquire().await?;
725                let client = &mut *guard;
726                let mut stream = client.simple_query(sql).await.map_err(map_tiberius_error)?;
727                let mut col_names: Vec<String> = Vec::new();
728                let mut result_rows: Vec<Vec<Value>> = Vec::new();
729                let mut col_set = false;
730                while let Some(item_result) = stream.next().await {
731                    match item_result {
732                        Ok(tiberius::QueryItem::Row(row)) => {
733                            if !col_set {
734                                for col in row.columns() {
735                                    col_names.push(col.name().to_string());
736                                }
737                                col_set = true;
738                            }
739                            let mut row_values: Vec<Value> = Vec::with_capacity(col_names.len());
740                            for (i, _) in col_names.iter().enumerate() {
741                                row_values.push(row_to_value(&row, i));
742                            }
743                            result_rows.push(row_values);
744                        }
745                        Ok(tiberius::QueryItem::Metadata(_)) => {
746                            // 元数据项，忽略（列信息从第一行提取）
747                        }
748                        Err(e) => {
749                            return Err(map_tiberius_error(e));
750                        }
751                    }
752                }
753                Ok::<QueryValues, DbError>((col_names, result_rows))
754            };
755            match result {
756                Ok(values) => Ok(values),
757                Err(e) => {
758                    self.mark_connection_error_after(&e);
759                    Err(e)
760                }
761            }
762        })
763    }

832        &'a mut self,
833    ) -> Pin<Box<dyn Future<Output = Result<(), DbError>> + Send + 'a>> {
834        Box::pin(async move {
835            if !self.connected {
836                return Err(DbError::ConnectionError("connection closed".to_string()));
837            }
838            let result = {
839                let mut guard = self.handle.acquire().await?;
840                let client = &mut *guard;
841                client
842                    .simple_query("BEGIN TRANSACTION")
843                    .await
844                    .map_err(map_tiberius_error)?;
845                Ok::<(), DbError>(())
846            };
847            match result {
848                Ok(_) => {
849                    self.in_transaction = true;
850                    Ok(())
851                }
852                Err(e) => {
853                    self.mark_connection_error_after(&e);
854                    Err(e)
855                }
856            }
857        })
858    }
