1//! SZ-ORM Microsoft SQL Server database adapter
2//!
3//! Implements the `Connection` trait of sz-orm-core based on the `tiberius` crate
4//! (pure Rust TDS protocol implementation), supporting SQL Server 2008+ (TDS 7.3+).
5//!
6//! # Design notes
7//!
8//! `tiberius` is a pure Rust asynchronous library that communicates directly with
9//! SQL Server via the TDS (Tabular Data Stream) protocol, without requiring native
10//! client libraries (such as ODBC/OLEDB).
11//!
12//! tiberius 0.12 uses `futures_io` traits, so tokio TcpStream must be wrapped via
13//! `tokio-util::compat`.
14//!
15//! SQL Server placeholders use the `@P1, @P2, ...` format; this adapter automatically
16//! converts `?` in SQL to `@PN`.
17//!
18//! # Usage
19//!
20//! ```ignore
21//! use sz_orm_core::{Pool, PoolConfigBuilder};
22//! use sz_orm_mssql::{MssqlPoolHandle, MssqlConnectionFactory};
23//! use std::sync::Arc;
24//!
25//! # #[tokio::main]
26//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
27//! let handle = Arc::new(MssqlPoolHandle::connect("server=tcp:localhost,1433;user=sa;password=P@ssw0rd;database=test").await?);
28//! let factory = Arc::new(MssqlConnectionFactory::new(handle));
29//! let config = PoolConfigBuilder::new().max_size(10).build()?;
30//! let pool = Pool::new(config, factory)?;
31//! # Ok(())
32//! # }
33//! ```

77/// Error conversion: tiberius::error::Error → DbError
78fn map_tiberius_error(e: tiberius::error::Error) -> DbError {
79    let msg = e.to_string();
80    if msg.contains("Connection")
81        || msg.contains("connection")
82        || msg.contains("Broken pipe")
83        || msg.contains("Login failed")
84    {
85        DbError::ConnectionError(msg)
86    } else if msg.contains("timeout") || msg.contains("Timeout") {
87        DbError::ConnectionTimeout(msg)
88    } else if msg.contains("already exists") || msg.contains("2627") || msg.contains("2601") {
89        // SQL Server 2627=唯一约束违反, 2601=重复键索引
90        DbError::UniqueViolation(msg)
91    } else if msg.contains("547") {
92        // SQL Server 547=外键约束违反
93        DbError::ForeignKeyViolation(msg)
94    } else if msg.contains("515") {
95        DbError::NullValue(msg)
96    } else if msg.contains("208") || msg.contains("Invalid object name") {
97        DbError::NotFound(msg)
98    } else if msg.contains("Config") || msg.contains("config") {
99        DbError::ConfigError(msg)
100    } else {
101        DbError::QueryError(msg)
102    }
103}

105/// Determines whether the SQL needs to go through the simple_query path
106/// (DDL/DCL/transaction control statements do not use prepared statements)
107fn needs_simple_query(sql: &str) -> bool {
108    let upper = sql.trim_start().to_uppercase();
109    upper.starts_with("CREATE ")
110        || upper.starts_with("ALTER ")
111        || upper.starts_with("DROP ")
112        || upper.starts_with("TRUNCATE ")
113        || upper.starts_with("GRANT ")
114        || upper.starts_with("REVOKE ")
115        || upper.starts_with("USE ")
116        || upper.starts_with("SET ")
117        || upper.starts_with("BEGIN TRAN")
118        || upper.starts_with("COMMIT")
119        || upper.starts_with("ROLLBACK")
120        || upper.starts_with("SAVE TRAN")
121        || upper.starts_with("EXEC ")
122        || upper.starts_with("EXECUTE ")
123}

125/// Converts `?` placeholders in SQL to SQL Server's `@PN` format
126///
127/// Skips `?` inside string literals; only converts parameter placeholders.
128fn convert_placeholders(sql: &str) -> String {
129    let mut result = String::with_capacity(sql.len() + 16);
130    let mut in_single_quote = false;
131    let mut param_index = 1u32;
132
133    for ch in sql.chars() {
134        match ch {
135            '\'' if !in_single_quote => {
136                in_single_quote = true;
137                result.push(ch);
138            }
139            '\'' if in_single_quote => {
140                in_single_quote = false;
141                result.push(ch);
142            }
143            '?' if !in_single_quote => {
144                result.push_str(&format!("@P{}", param_index));
145                param_index += 1;
146            }
147            _ => {
148                result.push(ch);
149            }
150        }
151    }
152    result
153}

155/// Owned parameter enum that implements tiberius::ToSql for prepared statement binding.
156///
157/// tiberius 0.12's `Client::execute/query` accepts `&[&dyn ToSql]`.
158/// By implementing `ToSql` for `MssqlParamOwned`, it can be passed directly as a parameter.
159enum MssqlParamOwned {
160    Null,
161    Bool(Option<bool>),
162    I16(Option<i16>),
163    I32(Option<i32>),
164    I64(Option<i64>),
165    U8(Option<u8>),
166    F32(Option<f32>),
167    F64(Option<f64>),
168    String(String),
169    Bytes(Vec<u8>),
170}

172impl ToSql for MssqlParamOwned {
173    fn to_sql(&self) -> ColumnData<'_> {
174        match self {
175            MssqlParamOwned::Null => ColumnData::I32(None),
176            MssqlParamOwned::Bool(b) => ColumnData::Bit(*b),
177            MssqlParamOwned::I16(n) => ColumnData::I16(*n),
178            MssqlParamOwned::I32(n) => ColumnData::I32(*n),
179            MssqlParamOwned::I64(n) => ColumnData::I64(*n),
180            MssqlParamOwned::U8(n) => ColumnData::U8(*n),
181            MssqlParamOwned::F32(f) => ColumnData::F32(*f),
182            MssqlParamOwned::F64(f) => ColumnData::F64(*f),
183            MssqlParamOwned::String(s) => {
184                ColumnData::String(Some(std::borrow::Cow::Borrowed(s.as_str())))
185            }
186            MssqlParamOwned::Bytes(b) => {
187                ColumnData::Binary(Some(std::borrow::Cow::Borrowed(b.as_slice())))
188            }
189        }
190    }

193/// Converts a sz-orm Value into MssqlParamOwned (an owned parameter)
194///
195/// # Performance notes
196///
197/// The `String`/`Bytes` branches clone the underlying data. Because `Value` itself
198/// holds ownership and `MssqlParamOwned` also needs owned data to cross the async
199/// boundary (tiberius's `Client::execute/query` executes asynchronously), this clone
200/// is unavoidable.
201///
202/// To reduce the per-call clone overhead, consider:
203/// - Using batch operations (`sz-orm-batch`) to amortize the cost of parameter construction;
204/// - Pre-constructing parameters outside the loop and reusing them;
205/// - Using pagination or streaming for large text/binary fields.
206fn value_to_mssql_param_owned(value: &Value) -> MssqlParamOwned {
207    match value {
208        Value::Null => MssqlParamOwned::Null,
209        Value::Bool(b) => MssqlParamOwned::Bool(Some(*b)),
210        Value::I8(n) => MssqlParamOwned::I32(Some(*n as i32)),
211        Value::I16(n) => MssqlParamOwned::I16(Some(*n)),
212        Value::I32(n) => MssqlParamOwned::I32(Some(*n)),
213        Value::I64(n) => MssqlParamOwned::I64(Some(*n)),
214        Value::U8(n) => MssqlParamOwned::U8(Some(*n)),
215        Value::U16(n) => MssqlParamOwned::I32(Some(*n as i32)),
216        Value::U32(n) => MssqlParamOwned::I64(Some(*n as i64)),
217        Value::U64(n) => MssqlParamOwned::I64(Some(*n as i64)),
218        Value::F32(f) => MssqlParamOwned::F32(Some(*f)),
219        Value::F64(f) => MssqlParamOwned::F64(Some(*f)),
220        Value::String(s) => MssqlParamOwned::String(s.clone()),
221        Value::Bytes(b) => MssqlParamOwned::Bytes(b.clone()),
222        // DECIMAL/NUMERIC 以字符串绑定，SQL Server 隐式转换为 DECIMAL 列
223        Value::Decimal(s) => MssqlParamOwned::String(s.clone()),
224        Value::Date(s) | Value::DateTime(s) | Value::Time(s) => MssqlParamOwned::String(s.clone()),
225        Value::Uuid(s) => MssqlParamOwned::String(s.clone()),
226        Value::Json(s) => MssqlParamOwned::String(s.clone()),
227        Value::Array(arr) => {
228            let json = serde_json::to_string(arr).unwrap_or_else(|_| "[]".to_string());
229            MssqlParamOwned::String(json)
230        }
231        Value::Object(obj) => {
232            let json = serde_json::to_string(obj).unwrap_or_else(|_| "{}".to_string());
233            MssqlParamOwned::String(json)
234        }
235        // 兼容 #[non_exhaustive] 的 Value：未知变体降级为 Null
236        _ => MssqlParamOwned::Null,
237    }
238}

240/// Converts the specified column of a tiberius Row into a sz-orm Value
241///
242/// tiberius 0.12's `FromSql` is implemented for `&str`/`&[u8]` (borrowed);
243/// `String`/`Vec<u8>` only implement `FromSqlOwned`.
244/// `try_get` requires `FromSql<'a>`, so `&str`/`&[u8]` are used and then converted to owned.
245///
246/// DECIMAL/NUMERIC/MONEY/SMALLMONEY are preferentially decoded as
247/// `rust_decimal::Decimal` into `Value::Decimal` (string form) to avoid f64 precision loss.
248fn row_to_value(row: &MssqlRow, idx: usize) -> Value {
249    // 1. 尝试 i64 (覆盖 BIT/TINYINT/SMALLINT/INT/BIGINT)
250    if let Ok(v) = row.try_get::<i64, usize>(idx) {
251        return v.map(Value::I64).unwrap_or(Value::Null);
252    }
253    // 2. 尝试 rust_decimal::Decimal (覆盖 DECIMAL/NUMERIC/MONEY/SMALLMONEY，保留精度)
254    if let Ok(v) = row.try_get::<rust_decimal::Decimal, usize>(idx) {
255        return v
256            .map(|d| Value::Decimal(d.to_string()))
257            .unwrap_or(Value::Null);
258    }
259    // 3. 尝试 f64 (覆盖 REAL/FLOAT)
260    if let Ok(v) = row.try_get::<f64, usize>(idx) {
261        return v.map(Value::F64).unwrap_or(Value::Null);
262    }
263    // 4. 尝试 bool (覆盖 BIT)
264    if let Ok(v) = row.try_get::<bool, usize>(idx) {
265        return v.map(Value::Bool).unwrap_or(Value::Null);
266    }
267    // 5. 尝试 &[u8] (覆盖 BINARY/VARBINARY/IMAGE)
268    if let Ok(v) = row.try_get::<&[u8], usize>(idx) {
269        return v.map(|b| Value::Bytes(b.to_vec())).unwrap_or(Value::Null);
270    }
271    // 6. 尝试 &str (覆盖 CHAR/VARCHAR/NCHAR/NVARCHAR/TEXT/NTEXT/DATE/TIME/DATETIME 等)
272    if let Ok(v) = row.try_get::<&str, usize>(idx) {
273        return v
274            .map(|s| Value::String(s.to_string()))
275            .unwrap_or(Value::Null);
276    }
277    Value::Null
278}

288/// SQL Server connection pool handle
289///
290/// v1.2.0 fixes P0: replaces the single-connection pseudo-pool (`Mutex<Option<Client>>`,
291/// concurrency=1) with a real connection pool (`Mutex<MssqlPoolInner>` holding multiple
292/// idle connections + a `Notify` wait mechanism), with concurrency = `max_size` (default 10).
293///
294/// # Design highlights
295///
296/// - **Multi-connection concurrency**: `idle` holds multiple idle connections; `acquire()`
297///   pops one for the caller's exclusive use, and it is returned via `MssqlClientGuard::drop`
298///   after the query completes
299/// - **Pool size constraint**: `total` tracks the number of created connections (idle + in-use);
300///   `acquire()` creates a new connection when `total < max_size`, otherwise waits on `Notify`
301///   for a return
302/// - **Lock granularity**: `std::sync::Mutex` is held only briefly during pop/return/counter
303///   updates, never across `.await`; `tiberius::Client::connect` performs async connection
304///   establishment outside the lock
305/// - **Connection reuse**: connections are returned to `idle` on `Drop` without being closed,
306///   avoiding repeated connection establishment overhead
307pub struct MssqlPoolHandle {
308    conn_str: String,
309    /// Connection pool internal state (idle connections + total count)
310    inner: Mutex<MssqlPoolInner>,
311    /// Connection return notification: when the pool is full, `acquire()` awaits `notified().await` here; `Drop` calls `notify_one`
312    notify: Notify,
313    /// Connection pool size upper bound (default 10)
314    max_size: usize,
315}

317impl MssqlPoolHandle {
318    /// Creates a new SQL Server connection pool (pool size upper bound defaults to 10)
319    pub async fn connect(conn_str: &str) -> Result<Self, DbError> {
320        Self::connect_with_max_size(conn_str, 10).await
321    }

323    /// Creates a new SQL Server connection pool (custom pool size upper bound)
324    ///
325    /// # Parameters
326    ///
327    /// - `max_size`: Pool size upper bound. Passing 0 is replaced with the default value 10
328    pub async fn connect_with_max_size(conn_str: &str, max_size: usize) -> Result<Self, DbError> {
329        let config = tiberius::Config::from_ado_string(conn_str)
330            .map_err(|e| DbError::ConfigError(e.to_string()))?;
331        let tcp = tokio::net::TcpStream::connect(config.get_addr())
332            .await
333            .map_err(|e| DbError::ConnectionError(e.to_string()))?;
334        tcp.set_nodelay(true).ok();
335        let client = tiberius::Client::connect(config, tcp.compat())
336            .await
337            .map_err(map_tiberius_error)?;
338        let max_size = if max_size == 0 { 10 } else { max_size };
339        Ok(Self {
340            conn_str: conn_str.to_string(),
341            inner: Mutex::new(MssqlPoolInner {
342                idle: vec![client],
343                total: 1,
344            }),
345            notify: Notify::new(),
346            max_size,
347        })
348    }

350    /// Acquires an idle connection from the pool (for exclusive use; automatically returned on `Drop`)
351    ///
352    /// # Acquire strategy
353    ///
354    /// 1. First pops an idle connection from `idle`
355    /// 2. If `idle` is empty and `total < max_size`, creates a new connection and increments `total`
356    /// 3. If `total >= max_size`, waits on `notify` until a connection is returned
357    ///
358    /// # Lock granularity
359    ///
360    /// `std::sync::Mutex` is held only briefly when popping a connection / updating counters;
361    /// async connection establishment (`TcpStream::connect` / `Client::connect`) is performed
362    /// outside the lock via `.await`, without blocking other callers' `acquire()`.
363    pub async fn acquire(&self) -> Result<MssqlClientGuard<'_>, DbError> {
364        loop {
365            // 短暂持锁：弹出空闲连接或决定是否创建新连接
366            let create_new = {
367                let mut inner = self
368                    .inner
369                    .lock()
370                    .map_err(|e| DbError::Internal(format!("MSSQL pool mutex poisoned: {}", e)))?;
371                if let Some(client) = inner.idle.pop() {
372                    return Ok(MssqlClientGuard {
373                        client: Some(client),
374                        pool: self,
375                    });
376                }
377                if inner.total < self.max_size {
378                    // 预占一个名额，锁外再建连
379                    inner.total += 1;
380                    true
381                } else {
382                    false
383                }
384            };
385
386            if create_new {
387                // 锁外异步建连：不阻塞其他 acquire()
388                match self.build_client().await {
389                    Ok(client) => {
390                        return Ok(MssqlClientGuard {
391                            client: Some(client),
392                            pool: self,
393                        });
394                    }
395                    Err(e) => {
396                        // 建连失败：回滚 total 并唤醒一个等待者，避免其空等
397                        let mut inner = self.inner.lock().map_err(|e| {
398                            DbError::Internal(format!("MSSQL pool mutex poisoned: {}", e))
399                        })?;
400                        inner.total -= 1;
401                        drop(inner);
402                        self.notify.notify_one();
403                        return Err(e);
404                    }
405                }
406            } else {
407                // 池满：等待连接归还
408                self.notify.notified().await;
409            }
410        }
411    }

436/// Exclusive guard for a SQL Server connection
437///
438/// Holds the client taken from the pool; on `Drop` it is automatically returned to the
439/// pool's `idle` list and one waiter is notified. The underlying `tiberius::Client` is
440/// accessed transparently via `Deref` / `DerefMut`.
441pub struct MssqlClientGuard<'a> {
442    /// The currently held client; taken out and returned on `Drop`, always `Some` during normal use
443    client: Option<tiberius::Client<CompatTcpStream>>,
444    /// Reference to the pool, used to return the connection
445    pool: &'a MssqlPoolHandle,
446}
447
448impl std::ops::Deref for MssqlClientGuard<'_> {
449    type Target = tiberius::Client<CompatTcpStream>;
450    fn deref(&self) -> &tiberius::Client<CompatTcpStream> {
451        self.client.as_ref().expect("connection must exist")
452    }
453}
454
455impl std::ops::DerefMut for MssqlClientGuard<'_> {
456    fn deref_mut(&mut self) -> &mut tiberius::Client<CompatTcpStream> {
457        self.client.as_mut().expect("connection must exist")
458    }
459}
460
461impl Drop for MssqlClientGuard<'_> {
462    fn drop(&mut self) {
463        if let Some(client) = self.client.take() {
464            // 即使 mutex poisoned 也强制归还，避免连接泄漏
465            let mut inner = self.pool.inner.lock().unwrap_or_else(|e| e.into_inner());
466            inner.idle.push(client);
467            drop(inner);
468            // 唤醒一个等待 acquire() 的任务
469            self.pool.notify.notify_one();
470        }
471    }
472}

513impl Connection for MssqlConnection {
514    fn execute<'a>(
515        &'a mut self,
516        sql: &'a str,
517    ) -> Pin<Box<dyn Future<Output = Result<u64, DbError>> + Send + 'a>> {
518        Box::pin(async move {
519            if !self.connected {
520                return Err(DbError::ConnectionError("connection closed".to_string()));
521            }
522            let result = {
523                let mut guard = self.handle.acquire().await?;
524                let client = &mut *guard;
525                if needs_simple_query(sql) {
526                    // DDL/事务控制：simple_query 不返回 rows_affected，消费流即可
527                    let mut stream = client.simple_query(sql).await.map_err(map_tiberius_error)?;
528                    while stream.next().await.is_some() {}
529                    Ok(1u64)
530                } else {
531                    // DML 无参数：用 execute 获取 rows_affected
532                    let exec_result = client.execute(sql, &[]).await.map_err(map_tiberius_error)?;
533                    Ok(exec_result.total())
534                }
535            };
536            match result {
537                Ok(n) => Ok(n),
538                Err(e) => {
539                    self.mark_connection_error_after(&e);
540                    Err(e)
541                }
542            }
543        })
544    }
545
546    fn query<'a>(
547        &'a mut self,
548        sql: &'a str,
549    ) -> Pin<Box<dyn Future<Output = Result<QueryRows, DbError>> + Send + 'a>> {
550        Box::pin(async move {
551            if !self.connected {
552                return Err(DbError::ConnectionError("connection closed".to_string()));
553            }
554            let result = {
555                let mut guard = self.handle.acquire().await?;
556                let client = &mut *guard;
557                let mut stream = client.simple_query(sql).await.map_err(map_tiberius_error)?;
558                let mut result_rows: QueryRows = Vec::new();
559                let mut col_names: Vec<String> = Vec::new();
560                let mut col_set = false;

1025/// Parses a SQL Server connection string
1026///
1027/// Supported formats: `server=host;port=1433;database=db;user=sa;password=pwd`
1028/// or `server=host,port=1433;database=db;user=sa;password=pwd`
1029///
1030/// # Errors
1031///
1032/// Returns `DbError::Internal` if the `server` or `database` field is missing.
1033pub fn parse_conn_str(conn_str: &str) -> Result<MssqlConnInfo, DbError> {
1034    let mut server = String::new();
1035    let mut port: u16 = 1433;
1036    let mut database = String::new();
1037    let mut user = String::new();
1038    let mut password = String::new();
1039
1040    for part in conn_str
1041        .split(';')
1042        .map(|s| s.trim())
1043        .filter(|s| !s.is_empty())
1044    {
1045        let (key, val) = part
1046            .split_once('=')
1047            .or_else(|| part.split_once(','))
1048            .ok_or_else(|| DbError::Internal(format!("invalid conn_str segment: {part}")))?;
1049        let key = key.trim().to_lowercase();
1050        let val = val.trim();
1051        match key.as_str() {
1052            "server" => server = val.to_string(),
1053            "port" => port = val.parse().unwrap_or(1433),
1054            "database" | "db" => database = val.to_string(),
1055            "user" | "uid" => user = val.to_string(),
1056            "password" | "pwd" => password = val.to_string(),
1057            _ => {}
1058        }
1059    }
1060
1061    if server.is_empty() {
1062        return Err(DbError::Internal("missing server in conn_str".to_string()));
1063    }
1064    if database.is_empty() {
1065        return Err(DbError::Internal(
1066            "missing database in conn_str".to_string(),
1067        ));
1068    }
1069
1070    Ok(MssqlConnInfo {
1071        server,
1072        port,
1073        database,
1074        user,
1075        password,
1076    })
1077}

1378/// SQL Server error categories
1379#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1380pub enum MssqlErrorCategory {
1381    /// Primary/unique key conflict (error codes 2627/2601)
1382    DuplicateKey,
1383    /// Constraint violation (error code 547)
1384    ConstraintViolation,
1385    /// NULL value violates a non-null constraint (error code 515)
1386    NullViolation,
1387    /// Object does not exist (error code 208)
1388    InvalidObject,
1389    /// Deadlock (error code 1205)
1390    Deadlock,
1391    /// Timeout (error code -2)
1392    Timeout,
1393    /// Other errors
1394    Other,
1395}
1396
1397impl MssqlErrorCategory {
1398    /// Categorizes from a SQL Server error code
1399    #[must_use]
1400    pub fn from_code(code: i32) -> Self {
1401        match code {
1402            2627 | 2601 => MssqlErrorCategory::DuplicateKey,
1403            547 => MssqlErrorCategory::ConstraintViolation,
1404            515 => MssqlErrorCategory::NullViolation,
1405            208 => MssqlErrorCategory::InvalidObject,
1406            1205 => MssqlErrorCategory::Deadlock,
1407            -2 => MssqlErrorCategory::Timeout,
1408            _ => MssqlErrorCategory::Other,
1409        }
1410    }
1411
1412    /// Error description
1413    #[must_use]
1414    pub fn description(&self) -> &'static str {
1415        match self {
1416            MssqlErrorCategory::DuplicateKey => "duplicate key violation",
1417            MssqlErrorCategory::ConstraintViolation => "constraint violation",
1418            MssqlErrorCategory::NullViolation => "NULL value in non-null column",
1419            MssqlErrorCategory::InvalidObject => "invalid object name",
1420            MssqlErrorCategory::Deadlock => "deadlock detected",
1421            MssqlErrorCategory::Timeout => "query timeout",
1422            MssqlErrorCategory::Other => "unknown error",
1423        }
1424    }
1425
1426    /// Whether it is retriable (deadlock/timeout)
1427    #[must_use]
1428    pub fn is_retriable(&self) -> bool {
1429        matches!(
1430            self,
1431            MssqlErrorCategory::Deadlock | MssqlErrorCategory::Timeout
1432        )
1433    }
1434}
