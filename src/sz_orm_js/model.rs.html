1//! Model — 通用模型包装器
2
3use napi_derive::napi;
4use std::collections::HashMap;
5use sz_orm_core::Value;
6
7#[napi]
8pub struct Model {
9    table_name: String,
10    pk_name: String,
11    fields: HashMap<String, Value>,
12}
13
14#[napi]
15impl Model {
16    #[napi(constructor)]
17    pub fn new(table_name: String, pk_name: Option<String>) -> Self {
18        Self {
19            table_name,
20            pk_name: pk_name.unwrap_or_else(|| "id".to_string()),
21            fields: HashMap::new(),
22        }
23    }
24
25    #[napi(getter)]
26    pub fn table_name(&self) -> String {
27        self.table_name.clone()
28    }
29
30    #[napi(getter)]
31    pub fn pk_name(&self) -> String {
32        self.pk_name.clone()
33    }
34
35    #[napi]
36    pub fn set_str(&mut self, key: String, value: String) {
37        self.fields.insert(key, Value::String(value));
38    }
39
40    #[napi]
41    pub fn set_i64(&mut self, key: String, value: i64) {
42        self.fields.insert(key, Value::I64(value));
43    }
44
45    #[napi]
46    pub fn set_f64(&mut self, key: String, value: f64) {
47        self.fields.insert(key, Value::F64(value));
48    }
49
50    #[napi]
51    pub fn set_bool(&mut self, key: String, value: bool) {
52        self.fields.insert(key, Value::Bool(value));
53    }
54
55    #[napi]
56    pub fn set_null(&mut self, key: String) {
57        self.fields.insert(key, Value::Null);
58    }
59
60    #[napi]
61    pub fn to_json_string(&self) -> String {
62        let obj: HashMap<String, Value> = self.fields.clone();
63        serde_json::to_string(&crate::types::value_to_json(&Value::Object(obj)))
64            .unwrap_or_else(|_| "{}".to_string())
65    }
66
67    #[napi]
68    pub fn status(&self) -> String {
69        format!(
70            "Model(table={}, pk={}, fields={})",
71            self.table_name,
72            self.pk_name,
73            self.fields.len()
74        )
75    }
76}

82    #[test]
83    fn model_default_pk() {
84        let m = Model::new("users".to_string(), None);
85        assert_eq!(m.table_name(), "users");
86        assert_eq!(m.pk_name(), "id");
87    }

92        assert_eq!(m.pk_name(), "order_id");
93    }

95    #[test]
96    fn model_set_and_json() {
97        let mut m = Model::new("users".to_string(), None);
98        m.set_str("name".to_string(), "Alice".to_string());
99        m.set_i64("age".to_string(), 30);
100        m.set_bool("vip".to_string(), true);
101        let json = m.to_json_string();
102        assert!(json.contains("Alice"), "json should contain name: {json}");
103        assert!(json.contains("30"), "json should contain age: {json}");
104        assert!(json.contains("true"), "json should contain vip: {json}");
105    }

113    }
