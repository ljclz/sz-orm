1//! v6.5.0 parallel_queries demo
2//!
3//! 演示多表并行查询 API：对 3 张表并行查询，打印结果对齐验证。
4
5use std::future::Future;
6use std::pin::Pin;
7
8use sz_orm_parallel::{config::ParallelQueryConfig, outcome::QueryOutcome, parallel_queries};
9
10#[tokio::main]
11async fn main() {
12    println!("=== parallel_queries demo ===");
13
14    let queries: Vec<
15        Box<
16            dyn FnOnce() -> Pin<
17                    Box<dyn Future<Output = Result<QueryOutcome<Vec<String>>, String>> + Send>,
18                > + Send,
19        >,
20    > = vec![
21        Box::new(|| {
22            Box::pin(async {
23                Ok(QueryOutcome::new(
24                    vec!["user1".into(), "user2".into()],
25                    2,
26                    10,
27                ))
28            })
29        }),
30        Box::new(|| Box::pin(async { Ok(QueryOutcome::new(vec!["order1".into()], 1, 15)) })),
31        Box::new(|| {
32            Box::pin(async {
33                Ok(QueryOutcome::new(
34                    vec!["log1".into(), "log2".into(), "log3".into()],
35                    3,
36                    20,
37                ))
38            })
39        }),
40    ];
41
42    let results = parallel_queries(queries, ParallelQueryConfig::default())
43        .await
44        .unwrap();
45
46    println!("结果数量: {}", results.len());
47    for (i, result) in results.iter().enumerate() {
48        match result {
49            Ok(rows) => println!("  query[{}]: {} rows", i, rows.len()),
50            Err(e) => println!("  query[{}]: error = {}", i, e),
51        }
52    }
53
54    println!("=== demo 完成 ===");
55}
