1//! v9.0.0 性能优化示例
2//!
3//! 演示 known_good 快速路径、clonecheap、reap_idle_selective 三项优化。
4
5use sz_orm_core::Value;
6
7#[tokio::main]
8async fn main() {
9    println!("=== sz-orm v9.0.0 性能优化示例 ===\n");
10
11    // 1. known_good 快速路径
12    println!("1. known_good 快速路径");
13    println!("   连接释放后 known_good=true，再次 acquire 时跳过 is_connected() 检查");
14    println!("   透明优化，无需修改业务代码\n");
15
16    // 2. clonecheap
17    println!("2. clonecheap 零堆分配克隆");
18    let val = Value::I64(42);
19    let cloned = val.clonecheap();
20    println!(
21        "   原值: {:?}, 克隆: {:?}（Copy 变体零堆分配）\n",
22        val, cloned
23    );
24
25    let str_val = Value::String("hello".into());
26    let str_cloned = str_val.clonecheap();
27    println!(
28        "   字符串值: {:?}, 克隆: {:?}（非 Copy 变体退化为 clone）\n",
29        str_val, str_cloned
30    );
31
32    // 3. reap_idle_selective
33    println!("3. reap_idle_selective 选择性回收");
34    println!("   逐个 pop 检查，遇到未过期连接即停止");
35    println!("   减少不必要的 pop/push 循环\n");
36
37    println!("=== 示例完成 ===");
38}
