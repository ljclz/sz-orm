1//! v7.3.0 任务 4.3：源 ORM 解析与迁移报告生成
2//!
3//! 提供 SourceOrmParser trait 与 Diesel/SeaORM/SQLx 三种解析器实现。
4//! 基于文件只读解析，识别模型定义或 schema，生成 sz-orm 等价定义与迁移 SQL + 报告。
5//!
6//! # 设计
7//!
8//! - `SourceOrmParser` trait：parse(&self, project_path) -> Result<SourceSchema, MigrationError>
9//! - `SourceSchema`：tables + relations
10//! - `OrmMigrator`：migrate(&self, source_path, dry_run) -> Result<MigrationReport, MigrationError>
11//! - 只读解析：source_unchanged = true，不修改源项目文件
12//! - 危险 DDL：含 DROP/TRUNCATE 时 has_dangerous_ddl = true，不自动执行
13//!
14//! # 生产调用点
15//!
16//! - DieselParser::parse：source_orm_parser.rs:DieselParser::parse
17//! - SeaOrmParser::parse：source_orm_parser.rs:SeaOrmParser::parse
18//! - SqlxParser::parse：source_orm_parser.rs:SqlxParser::parse
19//! - OrmMigrator::migrate：source_orm_parser.rs:OrmMigrator::migrate

102// ============================================================================
103// Diesel 解析器
104// ============================================================================
105
106/// Diesel 源 ORM 解析器
107///
108/// 识别 Diesel 模型定义：`#[derive(Selectable)]` + `#[diesel(table_name = ...)]` 结构体。
109/// 基于文件只读解析（不修改源文件）。
110pub struct DieselParser;
111
112impl SourceOrmParser for DieselParser {
113    fn parse(&self, project_path: &str) -> Result<SourceSchema, MigrationError> {
114        let mut schema = SourceSchema::default();
115        let path = Path::new(project_path);
116        if !path.exists() {
117            return Err(MigrationError::Llm(format!(
118                "源项目路径不存在: {}",
119                project_path
120            )));
121        }
122        // 只读遍历 src 目录，识别 Diesel 模型定义
123        let src = path.join("src");
124        let scan_root = if src.exists() { &src } else { path };
125        Self::scan_rust_files(scan_root, &mut schema, OrmMarker::Diesel)?;
126        Ok(schema)
127    }
128
129    fn orm_type(&self) -> SourceOrm {
130        SourceOrm::Diesel
131    }
132}
133
134impl DieselParser {
135    fn scan_rust_files(
136        dir: &Path,
137        schema: &mut SourceSchema,
138        marker: OrmMarker,
139    ) -> Result<(), MigrationError> {
140        let entries = fs::read_dir(dir).map_err(|e| MigrationError::Llm(e.to_string()))?;
141        for entry in entries.flatten() {
142            let p = entry.path();
143            if p.is_dir() {
144                Self::scan_rust_files(&p, schema, marker)?;
145            } else if p.extension().and_then(|s| s.to_str()) == Some("rs") {
146                let content =
147                    fs::read_to_string(&p).map_err(|e| MigrationError::Llm(e.to_string()))?;
148                Self::parse_rust_file(&content, schema, marker);
149            }
150        }
151        Ok(())
152    }
153
154    fn parse_rust_file(content: &str, schema: &mut SourceSchema, marker: OrmMarker) {
155        // 识别 Diesel: #[diesel(table_name = xxx)] 或 sea_orm::DeriveEntityModel
156        let marker_str = match marker {
157            OrmMarker::Diesel => "diesel(table_name",
158            OrmMarker::SeaOrm => "DeriveEntityModel",
159            OrmMarker::Sqlx => "FromRow",
160        };
161        if !content.contains(marker_str) {
162            return;
163        }
164        // 简化解析：提取 struct 名作为表名，字段作为列
165        let mut iter = content.lines().peekable();
166        while let Some(line) = iter.next() {
167            let trimmed = line.trim();
168            if trimmed.starts_with("pub struct ") || trimmed.starts_with("struct ") {
169                let after = trimmed
170                    .strip_prefix("pub struct ")
171                    .or_else(|| trimmed.strip_prefix("struct "))
172                    .unwrap_or("");
173                let name = after
174                    .split(|c: char| !c.is_alphanumeric() && c != '_')
175                    .next()
176                    .unwrap_or("");
177                if name.is_empty() {
178                    continue;
179                }
180                let table_name = name.to_lowercase();
181                let mut columns = Vec::new();
182                // 收集字段直到 }
183                for field_line in iter.by_ref() {
184                    let ft = field_line.trim();
185                    if ft.starts_with('}') || ft.is_empty() {
186                        break;
187                    }
188                    if let Some(col) = Self::parse_field(ft) {
189                        columns.push(col);
190                    }
191                }
192                if !columns.is_empty() {
193                    schema.tables.push(TableDef {
194                        name: table_name,
195                        columns,
196                    });
197                }
198            }
199        }
200    }
201
202    fn parse_field(line: &str) -> Option<ColumnDef> {
203        // pub field_name: Type 或 field_name: Type
204        let line = line.trim().trim_end_matches(',');
205        let line = line.strip_prefix("pub ").unwrap_or(line);
206        let colon_pos = line.find(':')?;
207        let name = line[..colon_pos].trim().to_string();
208        if name.is_empty() || name.starts_with("//") {
209            return None;
210        }
211        let ty = line[colon_pos + 1..].trim();
212        let nullable = ty.contains("Option<") || ty.contains("Nullable");
213        let sql_type = Self::rust_type_to_sql(ty);
214        Some(ColumnDef {
215            name,
216            sql_type,
217            nullable,
218        })
219    }
220
221    fn rust_type_to_sql(ty: &str) -> String {
222        let ty = ty.trim();
223        if ty.contains("i64") || ty.contains("i32") {
224            "BIGINT".to_string()
225        } else if ty.contains("String") || ty.contains("str") {
226            "VARCHAR(255)".to_string()
227        } else if ty.contains("f64") || ty.contains("f32") {
228            "DOUBLE".to_string()
229        } else if ty.contains("bool") {
230            "BOOLEAN".to_string()
231        } else if ty.contains("DateTime") || ty.contains("NaiveDateTime") {
232            "TIMESTAMP".to_string()
233        } else if ty.contains("Uuid") {
234            "UUID".to_string()
235        } else {
236            "TEXT".to_string()
237        }
238    }
239}

314pub struct MigrationMapping {

332/// 迁移报告（v7.3.0 任务 4.3）
333#[derive(Debug, Clone, Serialize, Deserialize)]
334pub struct MigrationReport {
335    /// 源 ORM 类型
336    pub source_orm: SourceOrm,
337    /// 源项目路径
338    pub source_path: String,
339    /// 映射数量
340    pub mapping_count: usize,
341    /// 不可映射数量
342    pub unmappable_count: usize,
343    /// 风险数量
344    pub risk_count: usize,
345    /// 是否含危险 DDL（DROP/TRUNCATE）
346    pub has_dangerous_ddl: bool,
347    /// 源项目是否未修改（只读解析，必须 true）
348    pub source_unchanged: bool,
349    /// 迁移映射列表
350    pub mappings: Vec<MigrationMapping>,
351    /// 不可映射列表
352    pub unmappable: Vec<UnmappableItem>,
353}

355/// 源 ORM 迁移器（v7.3.0 任务 4.3）
356pub struct OrmMigrator {
357    parser: Box<dyn SourceOrmParser>,
358}
359
360impl OrmMigrator {
361    /// 创建 OrmMigrator
362    pub fn new(parser: Box<dyn SourceOrmParser>) -> Self {
363        Self { parser }
364    }
365
366    /// 创建 Diesel 迁移器
367    pub fn diesel() -> Self {
368        Self::new(Box::new(DieselParser))
369    }
370
371    /// 创建 SeaORM 迁移器
372    pub fn sea_orm() -> Self {
373        Self::new(Box::new(SeaOrmParser))
374    }
375
376    /// 创建 SQLx 迁移器
377    pub fn sqlx() -> Self {
378        Self::new(Box::new(SqlxParser))
379    }
380
381    /// 执行迁移：解析 → 转换为 sz-orm 等价定义 → 生成迁移 SQL + 报告
382    ///
383    /// - `dry_run=true`：仅生成报告，不执行 DDL
384    /// - `dry_run=false`：仍不执行 DDL（本模块不连库），仅标记
385    /// - 源项目文件不修改（只读解析）：source_unchanged = true
386    /// - 含 DROP/TRUNCATE 时 has_dangerous_ddl = true，不自动执行
387    pub fn migrate(
388        &self,
389        source_path: &str,
390        dry_run: bool,
391    ) -> Result<MigrationReport, MigrationError> {
392        let _ = dry_run; // dry_run 仅影响报告标记，本模块不连库执行
393        let schema = self.parser.parse(source_path)?;
394        let source_orm = self.parser.orm_type();
395
396        let mut mappings = Vec::new();
397        let mut unmappable = Vec::new();
398        let has_dangerous_ddl = false;
399        let mut risk_count = 0;
400
401        for table in &schema.tables {
402            // 空表不可映射
403            if table.columns.is_empty() {
404                unmappable.push(UnmappableItem {
405                    source_table: table.name.clone(),
406                    reason: "表无列定义".to_string(),
407                });
408                continue;
409            }
410
411            // 生成 CREATE TABLE 迁移 SQL（正向，安全 DDL）
412            let mut sql = format!("CREATE TABLE {} (\n", table.name);
413            for (i, col) in table.columns.iter().enumerate() {
414                let null_str = if col.nullable { " NULL" } else { " NOT NULL" };
415                if i > 0 {
416                    sql.push_str(",\n");
417                }
418                sql.push_str(&format!("  {} {}{}", col.name, col.sql_type, null_str));
419            }
420            sql.push_str("\n)");
421            mappings.push(MigrationMapping {
422                source_table: table.name.clone(),
423                target_table: table.name.clone(),
424                migration_sql: sql,
425            });
426        }
427
428        // 危险 DDL 检测：本模块仅生成 CREATE TABLE（安全），不生成 DROP/TRUNCATE
429        // has_dangerous_ddl 保持 false（除非未来扩展反向迁移）
430        if has_dangerous_ddl {
431            risk_count += 1;
432        }
433
434        Ok(MigrationReport {
435            source_orm,
436            source_path: source_path.to_string(),
437            mapping_count: mappings.len(),
438            unmappable_count: unmappable.len(),
439            risk_count,
440            has_dangerous_ddl,
441            source_unchanged: true,
442            mappings,
443            unmappable,
444        })
445    }
446}

454    /// 创建临时源项目目录
455    fn make_temp_project(content: &str) -> std::path::PathBuf {
456        let dir = std::env::temp_dir().join(format!(
457            "sz_orm_test_{}_{}",
458            std::process::id(),
459            std::time::SystemTime::now()
460                .duration_since(std::time::UNIX_EPOCH)
461                .unwrap()
462                .as_nanos()
463        ));
464        let src = dir.join("src");
465        fs::create_dir_all(&src).unwrap();
466        let mut f = fs::File::create(src.join("models.rs")).unwrap();
467        f.write_all(content.as_bytes()).unwrap();
468        dir
469    }
