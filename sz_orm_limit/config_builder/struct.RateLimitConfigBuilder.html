限流配置构建器

链式 API 构建限流配置。

§示例

use sz_orm_limit::config_builder::{RateLimitConfigBuilder, LimitAlgorithm};

let config = RateLimitConfigBuilder::new()
    .algorithm(LimitAlgorithm::TokenBucket)
    .capacity(100)
    .rate(10.0)
    .max_keys(5000)
    .build();
